async function infer(prompt, { maxTokens } = {}) {
  const r = await inferOn(ollamaExec, { messages: [{ role: "user", content: String(prompt) }], temperature: 0, maxTokens: maxTokens ?? 900 });
  return r.text;
}

async function gateFor(checks) {
  const { mkdtempSync, rmSync } = await import("node:fs");
  const { tmpdir } = await import("node:os");
  const { pathToFileURL } = await import("node:url");
  return async function gate(code) {
    const dir = mkdtempSync(path.join(tmpdir(), "ember-gate-"));
    try {
      const file = path.join(dir, "m.mjs");
      writeFileSync(file, String(code ?? ""), "utf8");
      let mod;
      try { mod = await import(pathToFileURL(file).href); } catch (e) { return { ok: false, why: "does not import: " + String(e?.message ?? e).slice(0, 140) }; }
      const bad = [];
      for (const c of checks) {
        const fn = typeof c === "function" ? c : c.run;
        const r = await fn(mod);
        if (!r.ok) bad.push(`${r.name}: ${JSON.stringify(r.got)}, want ${JSON.stringify(r.want)}`);
      }
      return { ok: bad.length === 0, why: bad.join("; ") };
    } finally { rmSync(dir, { recursive: true, force: true }); }
  };
}
