// (penelope code-fix over the real mouth door), metering every draw.
//
// The mechanism: each module is drafted through runDrawDoor (box → mouth →
// bridge → local lane), the FALSIFYING GATE decides, a refused draft is
// reason-linted (janus + khora) into a sharpened atom and re-drawn. The
// ledger records prompt/eval tokens per draw; frontier tokens used and the
// frontier-equivalent estimates follow heimdall's dispatch meter conventions
// (exact external counts; estimates marked as estimates).

const ollamaExec = emptyExecutor({ executor: "ollama:qwen2.5-coder:1.5b", endpoint: "http://127.0.0.1:11434", model: "qwen2.5-coder:1.5b", provider: "ollama", location: "local/LAN", authClass: "local_open", privacyClass: "local-raw" });
async function infer(prompt, { kind, model, maxTokens } = {}) {
  const r = await inferOn(ollamaExec, { messages: [{ role: "user", content: String(prompt) }], temperature: 0, maxTokens: maxTokens ?? 900 });
  ledger.push({
    at: new Date().toISOString(),
    kind, model: model ?? MODEL, winner: "local",
    promptTokens: 0, evalTokens: r.tokens ?? 0,
    ok: r.status === 200, ms: r.ms ?? null, prompt: String(prompt ?? "").slice(0, 70),
  });
  return r.text;
}

// ── a gate that imports the candidate module and runs the checks ───────────
async function makeGate(checks, label) {
  return async function gate(code) {
    const dir = mkdtempSync(path.join(tmpdir(), "ember-gate-"));
    try {
      const file = path.join(dir, "m.mjs");
      writeFileSync(file, String(code ?? ""), "utf8");
      let mod;
      try {
        mod = await import(pathToFileURL(file).href);
      } catch (e) {
        return { ok: false, why: `candidate does not import: ${String(e?.message ?? e).slice(0, 160)}` };
      }
      const rows = [];
      for (const c of checks) {
        const fn = typeof c === "function" ? c : c.run;
        let got, want, ok;
        try { ({ got, want, ok } = await fn(mod)); } catch (e) { got = "threw: " + String(e?.message ?? e).slice(0, 80); want = "no throw"; ok = false; }
        rows.push({ name: c.name, got, want, ok });
      }
      const bad = rows.filter((r) => !r.ok);
      return {
        ok: bad.length === 0,
        why: bad.length ? bad.map((r) => `${r.name} = ${JSON.stringify(r.got)}, want ${JSON.stringify(r.want)}`).join("; ") : "exit 0",
        actual: bad.length ? { end1: bad[0].name, end2: JSON.stringify(bad[0].got) } : null,
        expected: bad.length ? { end1: bad[0].name, end2: JSON.stringify(bad[0].want) } : null,
        label,
      };
    } finally {
      rmSync(dir, { recursive: true, force: true });
    }
  };
}

async function buildModule(spec) {
  const result = await fixCode({
    brief: spec.brief,
    model: MODEL,
    goals: spec.goals,
    gate: await makeGate(spec.checks.map((c) => ({ name: c.name, run: c.run })), spec.file),
    infer,
    maxAttempts: 4,
    maxTokens: 900,
  });
  if (result.ok) writeFileSync(path.join(SRC, spec.file), result.code, "utf8");
  return { file: spec.file, ok: result.ok, attempts: result.attempts, reason: result.reason, why: result.why, scars: result.scars };
}

mkdirSync(SRC, { recursive: true });
const results = [];
for (const spec of MODULES) results.push(await buildModule(spec));

// ── the frontier token monitor ─────────────────────────────────────────────
const promptTotal = ledger.reduce((a, d) => a + (d.promptTokens || 0), 0);
const evalTotal = ledger.reduce((a, d) => a + (d.evalTokens || 0), 0);
const localTotal = promptTotal + evalTotal;
const frontierModels = (d) => /anthropic|openai|groq|claude|sonnet|opus/i.test(d.model ?? "");
const frontierUsed = ledger.filter(frontierModels);
const monitor = {
  schema: "TokenLedger@1",
  model: MODEL,
  draws: ledger.length,
  modules: results.map((r) => ({ file: r.file, ok: r.ok, attempts: r.attempts?.length ?? 0 })),
  localTokens: { prompt: promptTotal, eval: evalTotal, total: localTotal, exact: true },
  frontierTokens: {
    used: frontierUsed.reduce((a, d) => a + (d.promptTokens || 0) + (d.evalTokens || 0), 0),
    draws: frontierUsed.length,
    note: frontierUsed.length ? "frontier lanes served these draws" : "no frontier lane configured — every draw routed to the local lane",
  },
  estimated: {
    frontierEverything: Math.round(localTotal),
    conventionalRawContext: Math.round(localTotal * 22.7),
    note: "frontier-everything (1x) and conventional raw-context (22.7x) are estimates per heimdall dispatch meter; localTokens is exact",
  },
  perDraw: ledger,
};

writeFileSync(path.join(OUT, "TOKENS.json"), JSON.stringify(monitor, null, 2), "utf8");
