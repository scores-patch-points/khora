batch_18.py - Nashville CaseLink Eviction Case Scraper

An open-source tool for scraping eviction case data from Nashville's

WHAT IT DOES:
    1. Logs into Nashville's CaseLink system (requires an account)
    2. Searches for DETAINER WARRANT filings (Tennessee's eviction case type)
    3. Extracts case metadata from the virtual grid (docket, parties, dates)
    4. Geocodes addresses using OpenStreetMap's free Nominatim API
    5. Looks up Nashville Metro Council districts via the city's public GIS
    6. Saves everything to Google Drive via n8n webhook (no local state files)

HOW CASELINK WORKS:
    CaseLink is an Oracle APEX application rendered inside nested iframes.
    
    Login form fields:
        input#OPERCODE  = username
        input#PASSWD    = password  
        button#LogInSub = submit (also: xSubmit() JS function)
    
    After login, the URL changes from containing "#login" to just the base URL.
    The search page loads in an iframe with a basic search field (P_21) and
    a collapsible "Advanced Search" accordion.
    
    Advanced Search fields (hidden until accordion expanded):
        P_26   = File Date from (MM/DD/YYYY)
        P_27   = File Date to (MM/DD/YYYY)
        P_31   = Case Type dropdown (value "2" = DETAINER WARRANT)
        WTKCB_20 = Search button
    
    The accordion is expanded by calling: Accordion('SECTION_1', 1)
    
    Search results appear in a "virtual grid" using hidden <input> elements
    with class "list-input" and a "pfx" attribute encoding field + row:
        P_102 = Docket Number (e.g., "26GT1234")
        P_101 = Office
        P_103 = Status
        P_104 = File Date
        P_105 = Case Type
        P_106 = Plaintiff name
        P_107 = Defendant name
        P_108 = Attorney
        P_109 = Defendant Attorney

#
# These MUST be the key names the case records actually use. They were not:
# the set carried "Status", "File_Date", "lat", "lon", "defendant_address",

# Hashing

def canonical(obj) -> str:
    """Deterministic JSON for hashing. Sorted keys, no whitespace."""
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False)

def sha256_hex(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()

def hash_jsonl(jsonl: str) -> str:
    return "sha256:" + sha256_hex(jsonl)

# ===========================================================================
# Field passthrough
# ===========================================================================
# Defendant names and other sensitive fields are already encrypted by the
# scraper (ENC: prefix, Fernet over PBKDF2-derived key). The ciphertext is
# safe to publish — anyone with the password decrypts, anyone without it
# can't. We pass values through unchanged so the chain stays decryptable
# later by anyone holding the password.

def scrub(value):
    return value

def scrub_case(case: dict) -> dict:
    return dict(case)

# ===========================================================================
# Transformation construction
# ===========================================================================

def diff_case(old: Optional[dict], new: dict) -> list[dict]:
    """EO ops for one docket. Empty list means truly nothing changed."""
    if old is None:
        return [{"op": "INS", "data": scrub_case(new)}]
    changes = {}
    for k in DEF_FIELDS:
        if old.get(k) != new.get(k):
            changes[k] = {"old": scrub(old.get(k)), "new": scrub(new.get(k))}
    if not changes:
        return []
    ops: list[dict] = [{"op": "DEF", "changes": changes}]
    # lowercase 'status' is the real key; "Status" was always None on both
    # sides, so this SEG op could never fire.
    new_status = (new.get("status") or "").upper()
    old_status = (old.get("status") or "").upper()
    if new_status in TERMINAL_STATUSES and old_status not in TERMINAL_STATUSES:
        ops.append({"op": "SEG", "to_status": new_status})
    if new.get("enriched") and not old.get("enriched"):
        ops.append({"op": "CON", "via": new.get("geocode_source", "unknown")})
    return ops


def transformations_from_diff(
    prev_cases: dict,
    new_cases: dict,
    checked_dockets: Iterable[str] = (),
    extra_ops: Optional[list[dict]] = None,
) -> list[dict]:
    """Build the transformation list for a cycle.

    - INS for newly-seen dockets
    - DEF / SEG / CON for changed dockets
    - One batched NUL op listing dockets checked but unchanged
    - Extra ops (EVA, REC events from the scraper) appended as-is
    """
    out: list[dict] = []
    changed: set[str] = set()
    for docket, new in new_cases.items():
        ops = diff_case(prev_cases.get(docket), new)
        for op in ops:
            out.append({"docket": docket, **op})
            changed.add(docket)

    checked = set(checked_dockets)
    unchanged = sorted(d for d in checked if d not in changed and d in new_cases)
    if unchanged:
        # NUL is an OBSERVATION -- "looked at these at time T, nothing had
        # changed" -- not the absence of one. Without `at`, replay had no way
        # to record that the look happened, and every "still open" confirmation
        # from the sweep was discarded at the next restart: 4,246 open cases
        # read as unchecked for 90+ days after two full passes had seen them.
        out.append({"op": "NUL", "dockets": unchanged, "count": len(unchanged),
                    "at": datetime.now(CT).isoformat()})

    for ev in (extra_ops or []):
        if ev.get("op") in ("EVA", "REC", "SIG", "INS", "SEG", "CON", "DEF", "NUL"):
            out.append(ev)

    return out

# Block construction

def build_block_jsonl(
    *,
    block_type: str,
    block_index: int,
    prev_archive_id: Optional[str],
    prev_hash: Optional[str],
    transformations: list[dict],
    extra_header: Optional[dict] = None,
    block_date: Optional[str] = None,
) -> tuple[str, dict]:
    """Build the JSONL for one block. Returns (jsonl_text, header_dict).
    block_date overrides the header date, used when sealing yesterday's
    accumulated ops into a daily block."""
    now = datetime.now(timezone.utc)
    counts: dict[str, int] = {}
    for t in transformations:
        op = t.get("op", "?")
        counts[op] = counts.get(op, 0) + 1

    header = {
        "_block": True,
        "schema_version": 1,
        "chain": CHAIN_SUBJECT,
        "chain_version": 2,
        "block_type": block_type,
        "block_index": block_index,
        "date": block_date or now.strftime("%Y-%m-%d"),
        "ts": now.isoformat(),
        "prev_archive_id": prev_archive_id,
        "prev_hash": prev_hash,
        "transformation_count": len(transformations),
        "operator_counts": counts,
    }
    if extra_header:
        header.update(extra_header)

    lines = [canonical(header)] + [canonical(t) for t in transformations]
    jsonl = "\n".join(lines) + "\n"
    return jsonl, header

# ===========================================================================
# Upload via the archiveo n8n webhook
# ===========================================================================

async def upload_block(
    session: aiohttp.ClientSession,
    jsonl: str,
    header: dict,
    compress: Optional[bool] = None,
) -> dict:
    """Upload a block as a JSONL file directly to archive.org's S3
    endpoint. A 2xx response means archive.org accepted the bytes; there is
    no intermediary that can swallow failures.

    compress=None (default) auto-gzips blocks larger than COMPRESS_THRESHOLD.
    Filename extension signals the encoding: .jsonl plain, .jsonl.gz gzipped.
    Hashes (prev_hash, filename short hash) are always over the uncompressed
    JSONL.

    Raises on non-2xx."""
    block_type  = header["block_type"]
    block_index = header["block_index"]
    day         = header["date"]
    short       = sha256_hex(jsonl)[:10]

    # Identifier is generated client-side so we know it before upload.
    # Daily blocks (daily_fresh + same-day incrementals) share one per-day
    # item so a day's activity is a single archive.org item, not one per
    # cycle. Genesis/aggregate are rare and keep a UUID identifier.
    if block_type in ("daily_fresh", "incremental"):
        identifier = f"nashville-evictions-{day}"
    else:
        identifier = str(_uuid.uuid4())

    raw = jsonl.encode("utf-8")
    if compress is None:
        compress = len(raw) > COMPRESS_THRESHOLD

    if compress:
        payload  = gzip.compress(raw, compresslevel=9)
        filename = (f"evictions-block-{day}-{block_type}-"
                    f"{block_index:06d}-{short}.jsonl.gz")
        mime     = "application/gzip"
    else:
        payload  = raw
        filename = (f"evictions-block-{day}-{block_type}-"
                    f"{block_index:06d}-{short}.jsonl")
        mime     = "application/x-ndjson"

    tags = [
        CHAIN_SUBJECT,
        f"block-type:{block_type}",
        f"block-index:{block_index}",
        f"block-date:{day}",
    ]
    if block_type == "aggregate":
        tags.append(AGGREGATE_SUBJECT)
    if block_type == "genesis":
        tags.append(GENESIS_SUBJECT)

    # Title and description are not sent as HTTP headers — n8n's templating
    # of long strings into archive.org's S3 headers is fragile. The block's
    # JSONL line 1 carries this metadata authoritatively. archive.org will
    # use the filename as the item title, which is descriptive enough.

    # mime and payload were set above based on compression.
    archive_mediatype = "data"

    # Direct S3-style PUT. Header values must stay ASCII.
    put_url = f"{ARCHIVE_S3_ENDPOINT}/{identifier}/{filename}"
    put_headers = {
        "Authorization":               f"LOW {ARCHIVE_S3_ACCESS}:{ARCHIVE_S3_SECRET}",
        "Content-Type":                mime,
        "x-archive-auto-make-bucket":  "1",
        "x-archive-queue-derive":      "0",
        "x-archive-meta-mediatype":    archive_mediatype,
        "x-archive-meta-subject":      ";".join(tags),
        "x-archive-meta-date":         day,
    }

    async with session.put(
        put_url,
        data=payload,
        headers=put_headers,
        timeout=aiohttp.ClientTimeout(total=600),
    ) as r:
        text = await r.text()
        if r.status >= 300:
            raise RuntimeError(
                f"archive.org upload failed {r.status}: {text[:300]}")
        status = r.status

    return {
        "success": True,
        "archive": {
            "identifier": identifier,
            "url":        f"https://archive.org/details/{identifier}",
            "filename":   filename,
            "mime":       mime,
            "s3_status":  status,
        },
    }

async def search_chain(
    session: aiohttp.ClientSession,
    *,
    subject: str = CHAIN_SUBJECT,
    rows: int = 200,
    sort_desc: bool = True,
) -> list[dict]:
    """Search archive.org for blocks with the given subject tag."""
    params = [
        ("q", f'subject:"{subject}"'),
        ("fl[]", "identifier"),
        ("fl[]", "date"),
        ("fl[]", "subject"),
        ("fl[]", "title"),
        ("fl[]", "addeddate"),
        ("sort[]", f"addeddate {'desc' if sort_desc else 'asc'}"),
        ("rows", str(rows)),
        ("output", "json"),
    ]
    async with session.get(
        ARCHIVE_SEARCH_URL,
        params=params,
        timeout=aiohttp.ClientTimeout(total=60),
    ) as r:
        r.raise_for_status()
        data = await r.json()
    return data.get("response", {}).get("docs", [])


async def fetch_block(
    session: aiohttp.ClientSession,
    identifier: str,
    filename: Optional[str] = None,
) -> tuple[dict, list[dict]]:
    """Download a block's JSONL (plain or gzipped). Returns (header, ops)."""
    if filename is None:
        async with session.get(
            f"{ARCHIVE_METADATA_BASE}/{identifier}",
            timeout=aiohttp.ClientTimeout(total=60),
        ) as r:
            r.raise_for_status()
            meta = await r.json()
        files = meta.get("files", [])
        jsonl_files = [f for f in files
                       if f.get("name", "").endswith((".jsonl", ".jsonl.gz"))]
        if not jsonl_files:
            raise RuntimeError(f"{identifier}: no .jsonl(.gz) file")
        filename = jsonl_files[0]["name"]

    url = f"{ARCHIVE_DOWNLOAD_BASE}/{identifier}/{filename}"
    async with session.get(url, timeout=aiohttp.ClientTimeout(total=300)) as r:
        r.raise_for_status()
        raw = await r.read()

    if filename.endswith(".gz"):
        raw = gzip.decompress(raw)
    text = raw.decode("utf-8")

    lines = [ln for ln in text.splitlines() if ln.strip()]
    if not lines:
        raise RuntimeError(f"{identifier}: empty block")
    header = json.loads(lines[0])
    ops = [json.loads(ln) for ln in lines[1:]]
    return header, ops

# ===========================================================================
# State replay
# ===========================================================================

def apply_transformations(state: dict, ops: list[dict]) -> dict:
    src = state.get("cases") if isinstance(state, dict) else None
    if not isinstance(src, dict):
        src = state if isinstance(state, dict) and "cases" not in state else {}
    cases = {k: v for k, v in src.items() if isinstance(v, dict)}
    last_eva = None
    last_def_params = None

    for t in ops:
        op = t.get("op")
        docket = t.get("docket")

        if op == "INS" and docket:
            cases[docket] = dict(t.get("data", {}))
        elif op == "DEF" and docket:
            case = cases.setdefault(docket, {})
            for field, change in (t.get("changes") or {}).items():
                case[field] = change.get("new")
        elif op == "SEG" and docket:
            case = cases.setdefault(docket, {})
            # lowercase `status` is the real key; "Status" was written here for
            # years and read by nothing.
            case["status"] = t.get("to_status", case.get("status"))
        elif op == "CON" and docket:
            case = cases.setdefault(docket, {})
            case["enriched"] = True
        elif op == "EVA":
            last_eva = t
        elif op == "REC":
            if "def_snapshot" in t:
                last_def_params = t["def_snapshot"]
        elif op == "NUL":
            # Deliberately not a mutation of any FIELD -- but the observation
            # itself is state: these dockets were confirmed unchanged at `at`.
            at = t.get("at")
            if at:
                for d in t.get("dockets") or []:
                    case = cases.get(d)
                    if case is not None and at > (case.get("last_checked") or ""):
                        case["last_checked"] = at

    return {
        "cases": cases,
        "last_eva": last_eva,
        "last_def_params": last_def_params,
    }

# ===========================================================================
# Hydration — reconstruct current state from the chain
# ===========================================================================

async def hydrate_state(session: aiohttp.ClientSession) -> dict:
    """Load latest aggregate + replay all newer chain blocks. Returns
    {cases, last_eva, last_def_params, head_id, head_hash, next_index}."""

    state = {"cases": {}, "last_eva": None, "last_def_params": None}
    cutoff_addeddate: Optional[str] = None
    head_id: Optional[str] = None
    head_hash: Optional[str] = None
    next_index = 0

    # Direct-download the v2 genesis by id first — independent of the
    # search index. Gives us the full ~70k baseline immediately on a fresh
    # restart. The aggregate search + chain replay below still run and fold
    # any newer daily blocks on top.
    try:
        g_header, g_ops = await fetch_block(session, V2_GENESIS_ID)
        state = apply_transformations(state, g_ops)
        head_id = V2_GENESIS_ID
        next_index = int(g_header.get("block_index", 0)) + 1
        _gn = len(state.get("cases", {}))
        logger.info(f"  hydrate: v2 genesis loaded by id ({_gn} cases)")
    except Exception as e:
        logger.warning(f"  hydrate: direct v2 genesis fetch failed: {e}")

    aggregates = await search_chain(session, subject=AGGREGATE_SUBJECT, rows=1)

    if aggregates:
        agg = aggregates[0]
        logger.info(f"  hydrate: latest aggregate = {agg['identifier']}")
        header, ops = await fetch_block(session, agg["identifier"])
        state = apply_transformations(state, ops)
        cutoff_addeddate = agg.get("addeddate")
        head_id   = agg["identifier"]
        next_index = int(header.get("block_index", 0)) + 1
    else:
        logger.info("  hydrate: no aggregate yet, replaying from start")

    blocks = await search_chain(session, subject=CHAIN_SUBJECT, rows=1000)
    blocks.sort(key=lambda d: d.get("addeddate", ""))

    sealed_daily_dates: set = set()
    replayed = 0
    last_header: Optional[dict] = None
    for b in blocks:
        if cutoff_addeddate and (b.get("addeddate") or "") <= cutoff_addeddate:
            continue
        try:
            header, ops = await fetch_block(session, b["identifier"])
        except Exception as e:
            logger.warning(f"  hydrate: skipped {b['identifier']}: {e}")
            continue
        state = apply_transformations(state, ops)
        last_header = header
        head_id     = b["identifier"]
        replayed   += 1
        if header.get("block_type") == "daily":
            sealed_daily_dates.add(header.get("date"))

    if last_header:
        next_index = int(last_header.get("block_index", 0)) + 1

    state.update({
        "head_id":    head_id,
        "head_hash":  head_hash,
        "next_index": next_index,
    })
    logger.info(
        f"  hydrate: replayed {replayed} blocks, "
        f"{len(state.get('cases', {}))} cases, next_index={next_index}"
    )
    return state

# ===========================================================================
# Aggregate fold
# ===========================================================================

def should_aggregate(blocks: list[dict],
                     window_days: int = AGGREGATE_WINDOW_DAYS) -> bool:
    """True when at least window_days of non-aggregate blocks have accumulated
    since the most recent aggregate."""
    if not blocks:
        return False

    def has_agg(b):
        subj = b.get("subject")
        if isinstance(subj, list):
            return AGGREGATE_SUBJECT in subj
        return AGGREGATE_SUBJECT in (subj or "")

    aggregates = [b for b in blocks if has_agg(b)]
    today = datetime.now(timezone.utc).date()

    if not aggregates:
        dates = sorted({(b.get("date") or "")[:10] for b in blocks if b.get("date")})
        return len(dates) >= window_days

    last = max((a.get("date") or "")[:10] for a in aggregates)
    if not last:
        return False
    last_d = datetime.strptime(last, "%Y-%m-%d").date()
    return (today - last_d).days >= window_days


def build_aggregate_block(
    *,
    block_index: int,
    prev_archive_id: Optional[str],
    prev_hash: Optional[str],
    resolved_state: dict,
    folded_block_ids: list[str],
    operator_totals: dict,
    def_params: Optional[dict] = None,
) -> tuple[str, dict]:
    """Build an aggregate block. Body is a sequence of INS ops, one per case
    in resolved_state. A client loading this aggregate gets the full case set
    in one fetch and skips replaying every prior block."""
    ops: list[dict] = [
        {"docket": docket, "op": "INS", "data": scrub_case(case)}
        for docket, case in resolved_state.items()
    ]
    extra = {
        "folded_block_count": len(folded_block_ids),
        "folded_block_ids":   folded_block_ids,
        "folded_operator_totals": operator_totals,
    }
    if def_params:
        extra["def_params"] = def_params

    return build_block_jsonl(
        block_type="aggregate",
        block_index=block_index,
        prev_archive_id=prev_archive_id,
        prev_hash=prev_hash,
        transformations=ops,
        extra_header=extra,
    )

# ===========================================================================
# Chain head (the only thing kept between scraper runs)
# ===========================================================================

class ChainHead:
    """Tiny pointer file. About 200 bytes."""

    def __init__(self, path: str):
        self.path = Path(path)
        self.data: dict = {
            "head_id":         None,
            "head_hash":       None,
            "next_index":      0,
            "last_block_date": None,
            "last_aggregate":  None,
        }
        if self.path.exists():
            try:
                self.data.update(json.loads(self.path.read_text()))
            except Exception as e:
                logger.warning(f"  chain head read failed: {e}")

    def save(self):
        self.path.parent.mkdir(parents=True, exist_ok=True)
        tmp = str(self.path) + ".tmp"
        with open(tmp, "w") as f:
            json.dump(self.data, f, indent=2)
        os.replace(tmp, self.path)

    def advance(self, archive_id: str, jsonl: str, *, block_date: str,
                is_aggregate: bool = False):
        self.data["head_id"]         = archive_id
        self.data["head_hash"]       = hash_jsonl(jsonl)
        self.data["next_index"]      = int(self.data.get("next_index") or 0) + 1
        self.data["last_block_date"] = block_date
        if is_aggregate:
            self.data["last_aggregate"] = archive_id
        self.save()

# ===========================================================================
# Local cache purge
# ===========================================================================

def purge_local_cache(cache_dir: str, keep: Optional[set] = None) -> int:
    """Delete every file in cache_dir except those in `keep`."""
    p = Path(cache_dir)
    if not p.exists():
        return 0
    keep = keep or set()
    removed = 0
    for child in p.iterdir():
        if child.name in keep or not child.is_file():
            continue
        try:
            child.unlink()
            removed += 1
        except Exception as e:
            logger.warning(f"  purge: {child}: {e}")
    return removed


def choose_block_type(head: ChainHead) -> str:
    """genesis if chain is empty, daily_fresh if first block of today,
    incremental otherwise."""
    if not head.data.get("head_id"):
        return "genesis"
    today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    last  = head.data.get("last_block_date")
    return "daily_fresh" if last != today else "incremental"

def _encrypt_ops_inplace(ops):
    """Encrypt ENCRYPT_FIELDS inside a list of EO ops so nothing cleartext
    reaches archive.org. Idempotent: values already ENC: are left alone.
    INS carries data{...}; DEF carries changes{field:{old,new}}."""
    for t in ops:
        op = t.get("op")
        if op == "INS":
            data = t.get("data") or {}
            for fld in ENCRYPT_FIELDS:
                v = data.get(fld, "")
                if v and not str(v).startswith("ENC:"):
                    data[fld] = encrypt_name(v)
        elif op == "DEF":
            for fld, chg in (t.get("changes") or {}).items():
                if fld in ENCRYPT_FIELDS:
                    for side in ("old", "new"):
                        v = chg.get(side, "")
                        if v and not str(v).startswith("ENC:"):
                            chg[side] = encrypt_name(v)
    return ops


def write_snapshot(cases: dict):
    """Write the full current state as plain JSONL — one INS line per case —
    with defendant fields ENC: encrypted. Atomic temp-rename, 0644 so n8n
    (same user) can read it. This is the served artifact."""

    encrypt_state({"cases": snap})          # idempotent; guarantees ciphertext
    os.makedirs(LOCAL_STATE_DIR, exist_ok=True)

        for docket, case in snap.items():
            f.write(canonical({"docket": docket, "op": "INS", "data": case}) + "\n")
    os.replace(tmp, SNAPSHOT_FILE)
    try:
        os.chmod(SNAPSHOT_FILE, 0o644)
    except OSError:
        pass

_fernet = None

def get_fernet(password=None):
    global _fernet
    if _fernet is not None and password is None:
        return _fernet
    pwd = (password or ENCRYPTION_PASSWORD).encode()
    kdf = PBKDF2HMAC(algorithm=hashes.SHA256(), length=32,
                      salt=b'caselink-eviction-scraper', iterations=480000)
    key = base64.urlsafe_b64encode(kdf.derive(pwd))
    f = Fernet(key)
    if password is None:
        _fernet = f
    return f

def encrypt_name(plaintext, password=None):
    if not plaintext or not plaintext.strip():
        return ''
    return 'ENC:' + get_fernet(password).encrypt(plaintext.encode()).decode()

def decrypt_name(ciphertext, password=None):
    if not ciphertext or not ciphertext.startswith('ENC:'):
        return ciphertext or ''
    return get_fernet(password).decrypt(ciphertext[4:].encode()).decode()

def encrypt_state(state, password=None):
    for case in state.get('cases', {}).values():
        if not isinstance(case, dict):
            continue
        for f in ENCRYPT_FIELDS:
            v = case.get(f, '')
            if v and not v.startswith('ENC:'):
                case[f] = encrypt_name(v, password)

def decrypt_state(state, password=None):
    for case in state.get('cases', {}).values():
        if not isinstance(case, dict):
            continue
        for f in ENCRYPT_FIELDS:
            v = case.get(f, '')
            if v and v.startswith('ENC:'):
                try:
                    case[f] = decrypt_name(v, password)
                except Exception:
                    pass

def run_rec(proc):
    d = proc['def']
    e = proc['eva']
    adjustments = []
    now = datetime.now(CT).isoformat()

    # ── Rule rollback: check if last REC made things worse ──
    pre_snap = e.get('_pre_rec_snapshot')
    if pre_snap:
        prev_errors = pre_snap.get('errors_at_snapshot', 0)
        curr_errors = e.get('errors_this_cycle', 0)
        prev_enriched = pre_snap.get('enriched_at_snapshot', 0)
        # EVA entry from the history that recorded the snapshot cycle
        # If errors doubled AND enrichment dropped, revert
        if (curr_errors > prev_errors * 2 and curr_errors > 3
                and e.get('last_enriched_count', 0) < prev_enriched):
            # Walk backwards — restore pre-REC DEF
            old_def = pre_snap.get('def_snapshot')
            if old_def:
                for k, v in old_def.items():
                    d[k] = v
                adjustments.append(
                    f"ROLLBACK: errors {prev_errors}→{curr_errors},"
                    f" reverting last REC")
                e.pop('_pre_rec_snapshot', None)
                # Don't apply further REC this cycle — just revert
                if adjustments:
                    proc['history'].append(
                        {'ts': now, 'op': 'REC',
                         'adjustments': adjustments})
                    proc['history'].append(
                        {'ts': now, 'op': 'DEF',
                         'params': copy.deepcopy(d)})
                return adjustments
        # Snapshot cycle passed — clear it
        e.pop('_pre_rec_snapshot', None)

    # ── Enrichment waits tuning ──
    es = e.get('enrichment_stats', {})
    attempted = es.get('attempted', 0)
    succeeded = es.get('succeeded', 0)
    if attempted > 10:
        fail_rate = 1 - (succeeded / attempted)
        if fail_rate > 0.15:
            for k in ['short', 'medium', 'long', 'grid']:
                d['waits'][k] = int(d['waits'][k] * 1.3)
            e['consecutive_clean'] = 0
            adjustments.append(f"fail_rate {fail_rate:.0%}>15%: slowed waits")
        elif fail_rate < 0.03 and e.get('consecutive_clean', 0) >= 5:
            floors = {'short': 200, 'medium': 800, 'long': 1500, 'grid': 800}
            for k in floors:
                d['waits'][k] = max(floors[k], int(d['waits'][k] * 0.85))
            adjustments.append(f"clean x{e['consecutive_clean']}: sped up")

    # ── Job 2 learning ──
    j2 = e.get('job2', {})
    j2_cycle_attempts = j2.get('cycle_attempts', 0)
    j2_cycle_successes = j2.get('cycle_successes', 0)
    j2_fail_streak = j2.get('fail_streak', 0)

    if j2_cycle_attempts > 0 and j2_cycle_successes == 0:
        j2['fail_streak'] = j2_fail_streak + 1
    elif j2_cycle_successes > 0:
        j2['fail_streak'] = 0

    fail_streak = j2.get('fail_streak', 0)
    current_budget = d.get('pending_budget', 50)

    if fail_streak >= 3 and current_budget > 0:
        old_budget = current_budget
        d['pending_budget'] = 0
        j2['disabled_at'] = now
        j2['budget_before_disable'] = old_budget
        adjustments.append(
            f"job2 failed {fail_streak} cycles: budget {old_budget}→0")
    elif current_budget == 0 and j2.get('disabled_at'):
        try:
            disabled_dt = datetime.fromisoformat(j2['disabled_at'])
            hours_disabled = (
                datetime.now(CT) - disabled_dt).total_seconds() / 3600
        except Exception:
            hours_disabled = 0
        if hours_disabled > 24:
            d['pending_budget'] = 3
            adjustments.append(
                f"job2 probe: budget 0→3 after {hours_disabled:.0f}h")
    elif fail_streak == 0 and j2.get('budget_before_disable') and current_budget < 50:
        restored = j2.pop('budget_before_disable', 50)
        d['pending_budget'] = restored
        j2.pop('disabled_at', None)
        adjustments.append(f"job2 recovered: budget→{restored}")

    j2['cycle_attempts'] = 0
    j2['cycle_successes'] = 0
    e['job2'] = j2

    # ── Pending budget growth (only if job2 is working) ──
    ps = e.get('pending_stats', {})
    if ps.get('checks', 0) > 100 and d.get('pending_budget', 0) > 0:
        hit_rate = ps['hits'] / ps['checks'] if ps['checks'] else 0
        if hit_rate < 0.02:
            d['pending_budget'] = min(100, d.get('pending_budget', 50) + 10)
            adjustments.append(
                f"pending low hit {hit_rate:.0%}:"
                f" budget->{d['pending_budget']}")

    # ── Geocode learning ──
    geo = e.get('geocode', {})
    geo_prev = e.get('_geo_prev', {})
    if geo.get('success', 0) > 0:
        total_with_addr = (geo.get('success', 0) + geo.get('unfindable', 0)
                           + geo.get('pending', 0))
        if total_with_addr > 0:
            rate = geo['success'] / total_with_addr
            census_share = (geo.get('census', 0)
                            / geo['success'] if geo['success'] else 0)
            # Log source distribution
            if geo != geo_prev:
                adjustments.append(
                    f"geocode: {geo['success']} ok,"
                    f" {geo.get('unfindable', 0)} unfindable,"
                    f" {geo.get('pending', 0)} pending"
                    f" (census {census_share:.0%})")
    e['_geo_prev'] = copy.deepcopy(geo)

    # ── Snapshot current state for rollback check next cycle ──
    if adjustments:
        e['_pre_rec_snapshot'] = {
            'def_snapshot': copy.deepcopy(d),
            'errors_at_snapshot': e.get('errors_this_cycle', 0),
            'enriched_at_snapshot': e.get('last_enriched_count', 0),
        }
        proc['history'].append(
            {'ts': now, 'op': 'REC', 'adjustments': adjustments})
        proc['history'].append(
            {'ts': now, 'op': 'DEF', 'params': copy.deepcopy(d)})
    return adjustments

PRIORITY_MAX = 25

async def load_priority_queue():
    """Load priority queue from local disk."""
    try:
        with open(PRIORITY_FILE) as f:
            data = json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        return {'queue': [], 'max_size': PRIORITY_MAX, 'updated_at': None}
    if not isinstance(data, dict):
        return {'queue': [], 'max_size': PRIORITY_MAX, 'updated_at': None}
    for item in data.get('queue', []):
        item.setdefault('status', 'pending')
    return data

async def save_priority_queue(pq, msg='priority queue update'):
    """Save priority queue to local disk atomically."""
    pq['updated_at'] = datetime.now(CT).isoformat()
    try:
        os.makedirs(LOCAL_STATE_DIR, exist_ok=True)
        tmp = PRIORITY_FILE + '.tmp'
        with open(tmp, 'w') as f:
            json.dump(pq, f, indent=2)
        os.replace(tmp, PRIORITY_FILE)
        return True
    except Exception as e:
        logger.warning(f"  priority queue save failed: {e}")
        return False


def get_pending_priority(pq):
    """Return list of docket strings with status 'pending'."""
    return [item['docket'] for item in pq.get('queue', [])
            if item.get('status') == 'pending']

def mark_priority_done(pq, docket):
    """Mark a priority item as done. Returns True if found."""
    for item in pq.get('queue', []):
        if item['docket'] == docket and item['status'] == 'pending':
            item['status'] = 'done'
            item['completed_at'] = datetime.now(CT).isoformat()
            return True
    return False

def mark_priority_error(pq, docket, error_msg=''):
    """Mark a priority item as errored so the user sees it failed."""
    for item in pq.get('queue', []):
        if item['docket'] == docket and item['status'] == 'pending':
            item['status'] = 'error'
            item['error'] = error_msg
            item['errored_at'] = datetime.now(CT).isoformat()
            return True
    return False

    'us_congressional_district': 5,
}


def clean_address_for_geocoding(address):
    """Strip double-spaces and normalize for geocoder input."""
    if not address:
        return ''
    addr = ' '.join(address.split())
    return addr.strip()


async def geocode_address(session, address):
    """Geocode using Census (primary), Nominatim (fallback).
    Tries multiple address formats. Returns dict with source info."""
    addr = clean_address_for_geocoding(address)
    if not addr:
        return None

    no_unit = re.sub(
        r'\s+(UNIT|APT|SUITE|STE|#)\s*\S*', '', addr,
        flags=re.IGNORECASE).strip()
    street = addr.split(',')[0].strip()
    street_no_unit = re.sub(
        r'\s+(UNIT|APT|SUITE|STE|#)\s*\S*', '', street,
        flags=re.IGNORECASE).strip()
    strategies = list(dict.fromkeys(
        [addr, no_unit, street_no_unit, street]))

    for si, q in enumerate(strategies):
        # Census
        try:
            async with session.get(
                    CENSUS_GEOCODER_URL,
                    params={'address': q,
                            'benchmark': 'Public_AR_Current',
                            'vintage': 'Current_Current',
                            'format': 'json'},
                    timeout=10) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    matches = (data.get('result', {})
                               .get('addressMatches', []))
                    if matches:
                        coords = matches[0]['coordinates']
                        # Extract census geography
                        geo_info = {}
                        geographies = matches[0].get(
                            'geographies', {})
                        tracts = geographies.get(
                            'Census Tracts', [])
                        if tracts:
                            t = tracts[0]
                            geo_info['census_tract'] = t.get(
                                'TRACT', '')
                            geo_info['census_geoid'] = t.get(
                                'GEOID', '')
                            geo_info['census_county'] = t.get(
                                'COUNTY', '')
                            geo_info['census_state'] = t.get(
                                'STATE', '')
                            geo_info['census_name'] = t.get(
                                'NAME', '')
                        return {
                            'lat': coords['y'],
                            'lon': coords['x'],
                            'display_name': matches[0].get(
                                'matchedAddress', addr),
                            'source': 'census',
                            'strategy': si + 1,
                            'census': geo_info,
                        }
        except Exception as e:
            logger.debug(f"  Census error ({q[:30]}): {e}")

        # Nominatim fallback
        try:
            async with session.get(
                    NOMINATIM_URL,
                    params={'q': q, 'format': 'json',
                            'limit': 1, 'countrycodes': 'us'},
                    headers={'User-Agent': NOMINATIM_USER_AGENT},
                    timeout=10) as resp:
                if resp.status == 200:
                    results = await resp.json()
                    if results:
                        hit = results[0]
                        return {
                            'lat': float(hit['lat']),
                            'lon': float(hit['lon']),
                            'display_name': hit.get(
                                'display_name', ''),
                            'source': 'nominatim',
                            'strategy': si + 1,
                        }
        except Exception as e:
            logger.debug(f"  Nominatim error ({q[:30]}): {e}")

        await asyncio.sleep(GEOCODE_DELAY_SECONDS)
    return None


async def lookup_all_districts(session, lat, lon):
    """Query Nashville GIS for all political districts at a point.
    Returns dict like {council_district: '5', council_rep: 'Sean Parker', ...}
    """
    districts = {}
    for key, layer_id in DISTRICT_LAYERS.items():
        url = f"{NASHVILLE_DISTRICTS_BASE}/{layer_id}/query"
        params = {
            'geometry': f'{lon},{lat}',
            'geometryType': 'esriGeometryPoint',
            'inSR': '4326',
            'spatialRel': 'esriSpatialRelIntersects',
            'outFields': 'DistrictName,Representative',
            'returnGeometry': 'false',
            'f': 'json',
        }
        try:
            async with session.get(url, params=params, timeout=10) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    features = data.get('features', [])
                    if features:
                        attrs = features[0].get('attributes', {})
                        dn = attrs.get('DistrictName', '')
                        if dn:
                            districts[key] = dn
                        rep = attrs.get('Representative', '')
                        if rep:
                            rep_key = key.replace('_district', '_rep')
                            districts[rep_key] = rep
        except Exception as e:
            logger.debug(f"  District lookup {key} error: {e}")
        await asyncio.sleep(DISTRICT_LOOKUP_DELAY_SECONDS)
    return districts


async def lookup_districts_for_case(session, case):
    """Re-run just the district/rep GIS lookup for a case that already has
    coordinates, without spending a fresh address geocode. Used by JOB 3.6
    to backfill council_rep (and the other layers) onto cases whose address
    already resolved fine but whose district lookup was incomplete."""
    lat, lon = case.get('latitude'), case.get('longitude')
    if lat is None or lon is None:
        return case
    districts = await lookup_all_districts(session, lat, lon)
    for key, val in districts.items():
        case[key] = val
    return case


async def enrich_case_with_location(session, case):
    address = case.get('address_formatted') or case.get('Address_1', '')
    if not address:
        return case

    # DEF: max attempts before marking unfindable
    max_attempts = 3

    # Skip unfindable unless enough time has passed (probe)
    if case.get('geocode_status') == 'unfindable':
        last = case.get('geocode_last_attempt', '')
        if last:
            try:
                days_ago = (datetime.now(CT)
                            - datetime.fromisoformat(last)
                            ).total_seconds() / 86400
                if days_ago < 7:
                    return case  # too soon to re-probe
            except Exception:
                pass
        logger.info(f"    Probing unfindable address: {address[:40]}")

    geo = await geocode_address(session, address)
    now = datetime.now(CT).isoformat()
    case['geocode_last_attempt'] = now
    attempts = case.get('geocode_attempts', 0) + 1
    case['geocode_attempts'] = attempts

    if not geo:
        if attempts >= max_attempts:
            case['geocode_status'] = 'unfindable'
            logger.info(f"    Marked unfindable after {attempts} attempts")
        else:
            case['geocode_status'] = 'no_result'
        return case

    case['latitude'] = geo['lat']
    case['longitude'] = geo['lon']
    case['geocoded_address'] = geo['display_name']
    case['geocode_status'] = 'success'
    case['geocode_source'] = geo.get('source', '')
    case['geocode_strategy'] = geo.get('strategy', 0)

    # Census tract data (from forward geocode or reverse lookup)
    census = geo.get('census', {})
    if census:
        for k, v in census.items():
            if v:
                case[k] = v
    elif not case.get('census_tract'):
        # Nominatim hit — reverse lookup for tract
        tract = await lookup_census_tract(
            session, geo['lat'], geo['lon'])
        for k, v in tract.items():
            if v:
                case[k] = v

    districts = await lookup_all_districts(
        session, geo['lat'], geo['lon'])
    for key, val in districts.items():
        case[key] = val
    if not districts:
        logger.debug(f"  No districts for {geo['lat']},{geo['lon']}")
    return case


async def lookup_census_tract(session, lat, lon):
    """Reverse lookup: lat/lon → census tract via Census geocoder."""
    try:
        async with session.get(
                CENSUS_REVERSE_URL,
                params={'x': lon, 'y': lat,
                        'benchmark': 'Public_AR_Current',
                        'vintage': 'Current_Current',
                        'format': 'json'},
                timeout=10) as resp:
            if resp.status == 200:
                data = await resp.json()
                geographies = (data.get('result', {})
                                .get('geographies', {}))
                tracts = geographies.get('Census Tracts', [])
                if tracts:
                    t = tracts[0]
                    return {
                        'census_tract': t.get('TRACT', ''),
                        'census_geoid': t.get('GEOID', ''),
                        'census_county': t.get('COUNTY', ''),
                        'census_state': t.get('STATE', ''),
                        'census_name': t.get('NAME', ''),
                    }
    except Exception as e:
        logger.debug(f"  Census reverse lookup error: {e}")
    return {}


async def geocode_all_cases(cases):
    cases_with_address = [
        c for c in cases
        if c.get('address_formatted') or c.get('Address_1')]
    if not cases_with_address:
        return cases
    logger.info(f"\n  GEOCODING {len(cases_with_address)} addresses"
                f" (1/sec rate limit)")
    async with aiohttp.ClientSession() as session:
        for i, case in enumerate(cases_with_address, 1):
            addr = (case.get('address_formatted')
                    or case.get('Address_1', ''))
            logger.info(f"  [{i}/{len(cases_with_address)}]"
                        f" {addr[:50]}")
            await enrich_case_with_location(session, case)
            await asyncio.sleep(GEOCODE_DELAY_SECONDS)
    geocoded = sum(
        1 for c in cases if c.get('geocode_status') == 'success')
    logger.info(f"  Geocoded {geocoded}/{len(cases_with_address)}"
                f" addresses")
    return cases

# --- Login frame finder (production-tested) ---
async def find_login_frame(page):
    """Find the frame containing login fields"""
    for attempt in range(5):
        await page.wait_for_timeout(MEDIUM_WAIT)
        
        for frame in page.frames:
            try:
                opercode = await frame.query_selector('input[name="OPERCODE"], input#OPERCODE')
                if opercode:
                    logger.info(f"✓ Found login frame")
                    return frame
            except:
                continue
        
        logger.info(f"Login frame search attempt {attempt + 1}/5...")
    
    return page.main_frame


# --- Search frame finder (production-tested) ---
async def find_search_frame(page):
    """Find the frame containing the search input after login.
    Detects 404/bad states and navigates to LOGIN_URL to recover."""
    for attempt in range(5):
        await page.wait_for_timeout(MEDIUM_WAIT)
        
        # Check for 404 or bad page state first
        try:
            current_url = page.url
            if 'LandingPage' in current_url or '404' in current_url or current_url.endswith('.html'):
                logger.warning(f"Detected bad URL ({current_url}), navigating to login...")
                await page.goto(LOGIN_URL, wait_until='domcontentloaded')
                await page.wait_for_timeout(LONG_WAIT)
                continue
            
            # Check page content for 404
            content = await page.content()
            if 'Not Found' in content and 'was not found on this server' in content:
                logger.warning("Detected 404 page, navigating to login...")
                await page.goto(LOGIN_URL, wait_until='domcontentloaded')
                await page.wait_for_timeout(LONG_WAIT)
                continue
        except:
            pass
        
        for frame in page.frames:
            try:
                text_inputs = await frame.query_selector_all('input[type="text"]')
                if text_inputs and len(text_inputs) > 10:
                    for inp in text_inputs:
                        if await inp.is_visible():
                            placeholder = await inp.get_attribute('placeholder') or ''
                            logger.info(f"Found search frame with input (placeholder: '{placeholder}')")
                            return frame
            except:
                continue
        
        if attempt < 4:
            logger.info(f"Search frame not ready, waiting... (attempt {attempt + 1}/5)")
    
    logger.warning("Could not find dedicated search frame, using main frame")
    return page.main_frame


# --- Login handler (production-tested) ---
async def do_login(page, worker_id=1):
    """Handle login process"""
    logger.info("=" * 60)
    logger.info(f"LOGGING IN (Worker {worker_id} as {ALIAS})")
    logger.info("=" * 60)
    
    await page.goto(LOGIN_URL, wait_until='domcontentloaded')
    await page.wait_for_timeout(LONG_WAIT)
    
    async def _handle_dialog(dialog):
        try:
            await dialog.accept()
        except Exception:
            pass  # Already handled
    
    page.on('dialog', lambda d: asyncio.ensure_future(_handle_dialog(d)))
    
    login_frame = await find_login_frame(page)
    
    # Fill username
    logger.info("Filling username...")
    try:
        username_field = await login_frame.query_selector('input#OPERCODE, input[name="OPERCODE"]')
        if username_field and await username_field.is_visible():
            await username_field.click()
            await page.wait_for_timeout(300)
            await username_field.fill('')
            await username_field.type(USERNAME, delay=50)
            logger.info("✓ Username entered")
    except Exception as e:
        await login_frame.evaluate(f"""
            () => {{
                const field = document.getElementById('OPERCODE') || document.querySelector('input[name="OPERCODE"]');
                if (field) {{
                    field.value = '{USERNAME}';
                    field.dispatchEvent(new Event('change', {{ bubbles: true }}));
                }}
            }}
        """)
        logger.info("✓ Username entered via JS")
    
    await page.wait_for_timeout(SHORT_WAIT)
    
    # Fill password
    logger.info("Filling password...")
    try:
        password_field = await login_frame.query_selector('input#PASSWD, input[name="PASSWD"]')
        if password_field and await password_field.is_visible():
            await password_field.click()
            await page.wait_for_timeout(300)
            await password_field.fill('')
            await password_field.type(PASSWORD, delay=50)
            logger.info("✓ Password entered")
    except Exception as e:
        await login_frame.evaluate(f"""
            () => {{
                const field = document.getElementById('PASSWD') || document.querySelector('input[name="PASSWD"]');
                if (field) {{
                    field.value = '{PASSWORD}';
                    field.dispatchEvent(new Event('change', {{ bubbles: true }}));
                }}
            }}
        """)
        logger.info("✓ Password entered via JS")
    
    await page.wait_for_timeout(SHORT_WAIT)
    
    # Click login button
    logger.info("Clicking login...")
    login_clicked = False
    
    try:
        login_btn = await login_frame.query_selector('button#LogInSub')
        if login_btn and await login_btn.is_visible():
            await login_btn.click()
            logger.info("✓ Clicked LogInSub button directly")
            login_clicked = True
    except Exception as e:
        logger.debug(f"Direct button click failed: {e}")
    
    if not login_clicked:
        try:
            result = await login_frame.evaluate("""
                () => {
                    if (typeof xSubmit === 'function') {
                        xSubmit();
                        return 'xSubmit';
                    }
                    const btn = document.getElementById('LogInSub');
                    if (btn) {
                        btn.click();
                        return 'getElementById';
                    }
                    const form = document.querySelector('form[name="main"]');
                    if (form) {
                        form.submit();
                        return 'formSubmit';
                    }
                    return false;
                }
            """)
            if result:
                logger.info(f"✓ Login triggered via JS in login frame: {result}")
                login_clicked = True
        except Exception as e:
            logger.debug(f"JS login failed: {e}")
    
    if not login_clicked:
        try:
            password_field = await login_frame.query_selector('input#PASSWD, input[name="PASSWD"]')
            if password_field:
                await password_field.press('Enter')
                logger.info("✓ Pressed Enter on password field")
                login_clicked = True
        except Exception as e:
            logger.debug(f"Enter key failed: {e}")
    
    await page.wait_for_timeout(MEDIUM_WAIT)
    
    if '#login' not in page.url:
        logger.info("✓ Login successful - URL changed")
    else:
        logger.warning("⚠ Still on login page, waiting longer...")
        await page.wait_for_timeout(LONG_WAIT)
    
    await page.wait_for_timeout(LONG_WAIT)
    return '#login' not in page.url

async def search_detainer_cases_by_date(page, start_date, end_date):
    """
    Search CaseLink for detainer cases filed between start_date and end_date.
    Uses Advanced Search with specific field names: P_26 (date from), P_27 (date to), P_31 (case type)
    """
    logger.info(f"Searching for DETAINER WARRANT cases from {start_date.strftime('%m/%d/%Y')} to {end_date.strftime('%m/%d/%Y')}")
    
    start_str = start_date.strftime('%m/%d/%Y')
    end_str = end_date.strftime('%m/%d/%Y')
    
    # Step 1: Open Advanced Search by calling Accordion function
    logger.info("Opening Advanced Search...")
    
    for frame in page.frames:
        try:
            result = await frame.evaluate("""
                () => {
                    // Method 1: Call Accordion function directly
                    if (typeof Accordion === 'function') {
                        Accordion('SECTION_1', 1);
                        return { success: true, method: 'Accordion' };
                    }
                    
                    // Method 2: Click the Advanced Search button/link
                    const elements = document.querySelectorAll('button, a, div, span, td');
                    for (const el of elements) {
                        const text = (el.innerText || '').toLowerCase();
                        if (text.includes('advanced search')) {
                            el.click();
                            return { success: true, method: 'click' };
                        }
                    }
                    
                    return { success: false };
                }
            """)
            
            if result and result.get('success'):
                logger.info(f"✓ Opened Advanced Search via {result.get('method')}")
                break
        except:
            continue
    
    await page.wait_for_timeout(MEDIUM_WAIT)
    
    # Step 2: Fill File Date From (P_26)
    logger.info(f"Setting File Date from: {start_str}...")
    
    for frame in page.frames:
        try:
            date_from_set = await frame.evaluate(f"""
                () => {{
                    const field = document.querySelector('input[name="P_26"]');
                    if (field) {{
                        field.value = '{start_str}';
                        field.dispatchEvent(new Event('input', {{ bubbles: true }}));
                        field.dispatchEvent(new Event('change', {{ bubbles: true }}));
                        return {{ success: true }};
                    }}
                    return {{ success: false }};
                }}
            """)
            
            if date_from_set and date_from_set.get('success'):
                logger.info(f"✓ Set File Date from (P_26): {start_str}")
                break
        except:
            continue
    
    await page.wait_for_timeout(SHORT_WAIT)
    
    # Step 3: Fill File Date To (P_27) - set to same date as from (today)
    logger.info(f"Setting File Date to: {end_str}...")
    
    for frame in page.frames:
        try:
            date_to_set = await frame.evaluate(f"""
                () => {{
                    const field = document.querySelector('input[name="P_27"]');
                    if (field) {{
                        field.value = '{end_str}';
                        field.dispatchEvent(new Event('input', {{ bubbles: true }}));
                        field.dispatchEvent(new Event('change', {{ bubbles: true }}));
                        return {{ success: true }};
                    }}
                    return {{ success: false }};
                }}
            """)
            
            if date_to_set and date_to_set.get('success'):
                logger.info(f"✓ Set File Date to (P_27): {end_str}")
                break
        except:
            continue
    
    await page.wait_for_timeout(SHORT_WAIT)
    
    # Step 4: Select Case Type = DETAINER WARRANT (P_31, value=2)
    logger.info("Selecting Case Type: DETAINER WARRANT...")
    
    for frame in page.frames:
        try:
            case_type_set = await frame.evaluate("""
                () => {
                    const select = document.querySelector('select[name="P_31"]');
                    if (select) {
                        select.value = '2';  // DETAINER WARRANT
                        select.dispatchEvent(new Event('change', { bubbles: true }));
                        return { success: true, value: '2' };
                    }
                    return { success: false };
                }
            """)
            
            if case_type_set and case_type_set.get('success'):
                logger.info("✓ Selected DETAINER WARRANT (P_31 = 2)")
                break
        except:
            continue
    
    await page.wait_for_timeout(SHORT_WAIT)
    
    
    # Step 5: Click Search button (WTKCB_20)
    logger.info("Clicking Search button...")
    search_clicked = False
    
    for frame in page.frames:
        try:
            # Method 1: Click the button directly
            clicked = await frame.evaluate("""
                () => {
                    const btn = document.querySelector('button[name="WTKCB_20"]');
                    if (btn) {
                        btn.click();
                        return { success: true, method: 'button click' };
                    }
                    return { success: false };
                }
            """)
            
            if clicked and clicked.get('success'):
                search_clicked = True
                logger.info(f"✓ Clicked Search button via {clicked.get('method')}")
                break
        except:
            continue
    
    # Fallback: Try PostBack function
    if not search_clicked:
        for frame in page.frames:
            try:
                clicked = await frame.evaluate("""
                    () => {
                        if (typeof PostBack === 'function') {
                            PostBack('', 'WTKCB_20', 'Search for Case(s)', 1, 0);
                            return { success: true, method: 'PostBack' };
                        }
                        return { success: false };
                    }
                """)
                
                if clicked and clicked.get('success'):
                    search_clicked = True
                    logger.info(f"✓ Clicked Search via {clicked.get('method')}")
                    break
            except:
                continue
    
    # Fallback: Find any Search button
    if not search_clicked:
        for frame in page.frames:
            try:
                clicked = await frame.evaluate("""
                    () => {
                        const buttons = document.querySelectorAll('button, input[type="button"]');
                        for (const btn of buttons) {
                            const text = (btn.innerText || btn.value || '').toLowerCase();
                            if (text.includes('search') && !text.includes('advanced')) {
                                btn.click();
                                return { success: true, method: 'text search' };
                            }
                        }
                        return { success: false };
                    }
                """)
                
                if clicked and clicked.get('success'):
                    search_clicked = True
                    logger.info(f"✓ Clicked Search via {clicked.get('method')}")
                    break
            except:
                continue
    
    if not search_clicked:
        logger.warning("⚠ Could not click Search button")
    
    await page.wait_for_timeout(LONG_WAIT * 2)  # Give more time for results to load
    
    # Check for "No Records Accessible" popup and dismiss it
    for frame in page.frames:
        try:
            popup_dismissed = await frame.evaluate("""
                () => {
                    // Look for the popup with "No Records Accessible" text
                    const allElements = document.querySelectorAll('*');
                    for (const el of allElements) {
                        if (el.innerText && el.innerText.includes('No Records Accessible')) {
                            // Find and click the OK button
                            const okBtn = document.querySelector('button');
                            if (okBtn && (okBtn.innerText || '').includes('OK')) {
                                okBtn.click();
                                return { found: true, dismissed: true };
                            }
                            // Try clicking any button in a dialog
                            const buttons = document.querySelectorAll('button');
                            for (const btn of buttons) {
                                if ((btn.innerText || '').trim() === 'OK') {
                                    btn.click();
                                    return { found: true, dismissed: true };
                                }
                            }
                            return { found: true, dismissed: false };
                        }
                    }
                    return { found: false };
                }
            """)
            
            if popup_dismissed and popup_dismissed.get('found'):
                logger.info("⚠ No records found for this date range (popup dismissed)")
                return []  # Return empty list - no cases found
        except:
            continue
    
    
    # Extract case numbers from results
    cases = await extract_cases_from_results(page)
    
    return cases


# --- Grid extraction with pagination (production-tested) ---
async def extract_cases_from_results(page):
    """Extract cases by clicking >> to paginate through results"""
    
    logger.info("Extracting case data from grid...")
    
    all_cases = {}  # Use dict with docket as key for deduplication
    
    # First, get the total count from the input next to "GO TO" button
    total_expected = 0
    main_frame = None
    
    for frame in page.frames:
        try:
            total_expected = await frame.evaluate("""
                () => {
                    const inputs = document.querySelectorAll('input');
                    for (const inp of inputs) {
                        const val = inp.value || '';
                        if (/^\\d+$/.test(val)) {
                            const num = parseInt(val);
                            if (num > 10 && num < 10000) {
                                const parent = inp.parentElement;
                                if (parent && parent.innerText.includes('GO TO')) {
                                    return num;
                                }
                            }
                        }
                    }
                    const text = document.body.innerText || '';
                    const match = text.match(/of\\s+(\\d+)\\s*Lines/i);
                    if (match) return parseInt(match[1]);
                    return 0;
                }
            """)
            if total_expected > 0:
                main_frame = frame
                logger.info(f"Total cases expected: {total_expected}")
                break
        except:
            continue
    
    if not main_frame:
        # Find the frame with the grid
        for frame in page.frames:
            try:
                has_grid = await frame.evaluate("() => document.querySelectorAll('input.list-input').length > 0")
                if has_grid:
                    main_frame = frame
                    break
            except:
                continue
    
    if not main_frame:
        logger.error("Could not find grid frame")
        return []
    
    max_iterations = 80  # Increased from 50
    iteration = 0
    stuck_count = 0
    last_count = 0
    consecutive_same = 0  # Track truly stuck vs slow loading
    
    while iteration < max_iterations:
        # Extract visible cases - try multiple times per iteration
        extraction_attempts = 0
        while extraction_attempts < 3:
            try:
                # Extract cases AND the current grid row range
                extract_result = await main_frame.evaluate("""
                    () => {
                        const cases = [];
                        const inputs = document.querySelectorAll('input.list-input');
                        const rows = {};
                        
                        for (const inp of inputs) {
                            const pfx = inp.getAttribute('pfx') || '';
                            const value = inp.value || '';
                            if (!pfx || !value) continue;
                            
                            const match = pfx.match(/^(P_\\d+)_(\\d+)$/);
                            if (!match) continue;
                            
                            const field = match[1];
                            const rowNum = match[2];
                            if (!rows[rowNum]) rows[rowNum] = {};
                            rows[rowNum][field] = value;
                        }
                        
                        for (const rowNum in rows) {
                            const row = rows[rowNum];
                            const docket = row['P_102'] || '';
                            if (!docket || !docket.match(/\\d{2}GT\\d+/)) continue;
                            
                            cases.push({
                                Docket_Number: docket,
                                Office: row['P_101'] || '',
                                Status: row['P_103'] || '',
                                File_Date: row['P_104'] || '',
                                Case_Type: row['P_105'] || 'DETAINER WARRANT',
                                Plaintiff_1: row['P_106'] || '',
                                Defendant_1: row['P_107'] || '',
                                Attorney: row['P_108'] || '',
                                Def_Attorney: row['P_109'] || ''
                            });
                        }
                        
                        // Get grid position info - the GM2A_1 input shows the first visible row number
                        let firstRow = 0, lastRow = 0;
                        const gm2a1 = document.querySelector('input[name="GM2A_1"]');
                        const gm2aInputs = document.querySelectorAll('input[name^="GM2A_"]');
                        if (gm2a1) firstRow = parseInt(gm2a1.value) || 0;
                        if (gm2aInputs.length > 0) {
                            const lastInput = gm2aInputs[gm2aInputs.length - 1];
                            lastRow = parseInt(lastInput.value) || 0;
                        }
                        
                        return { cases, firstRow, lastRow };
                    }
                """)
                
                cases = extract_result.get('cases', []) if extract_result else []
                first_row = extract_result.get('firstRow', 0) if extract_result else 0
                last_row = extract_result.get('lastRow', 0) if extract_result else 0
                
                if cases:
                    new_count = 0
                    for case in cases:
                        docket = case.get('Docket_Number')
                        if docket and docket not in all_cases:
                            all_cases[docket] = case
                            new_count += 1
                    if len(all_cases) > last_count:
                        # Log grid position when we get new cases
                        if iteration > 0 and new_count > 0:
                            logger.info(f"  Grid rows {first_row}-{last_row}: +{new_count} new cases")
                        break  # Got new cases, exit inner loop
                            
                extraction_attempts += 1
                if extraction_attempts < 3:
                    await page.wait_for_timeout(500)  # Brief wait between attempts
                            
            except Exception as e:
                logger.debug(f"Extract error: {e}")
                extraction_attempts += 1
        
        current_count = len(all_cases)
        
        # Log progress when we get new cases
        if current_count > last_count:
            logger.info(f"  Got {current_count}/{total_expected or '?'} unique cases")
            stuck_count = 0
            consecutive_same = 0
            last_count = current_count
        else:
            stuck_count += 1
            consecutive_same += 1
        
        # Check if we have all cases
        if total_expected > 0 and current_count >= total_expected:
            logger.info(f"✓ Got all {total_expected} cases")
            break
        
        # More tolerant stuck detection - give grid time to load
        # Only give up after 15 consecutive no-progress AND we've tried multiple times
        if stuck_count >= 15 and consecutive_same >= 10:
            logger.warning(f"⚠ No new cases after {stuck_count} attempts - got {current_count}/{total_expected}")
            break
        
        # Call GridBF('FORWPAGE') to navigate to next page of virtual grid
        # The grid reuses the same 20 row slots - values change as you navigate
        nav_success = False
        for frame in page.frames:
            try:
                nav_result = await frame.evaluate("""
                    () => {
                        // Try calling GridBF directly (CaseLink's navigation function)
                        if (typeof GridBF === 'function') {
                            GridBF('FORWPAGE', 0, 0, 0);
                            return { method: 'GridBF_FORWPAGE', success: true };
                        }
                        return { method: 'GridBF_not_found', success: false };
                    }
                """)
                
                if nav_result and nav_result.get('success'):
                    nav_success = True
                    if iteration == 0:
                        logger.info(f"Grid navigation method: {nav_result.get('method')}")
                    break
            except Exception as e:
                pass
        
        # If GridBF didn't work, try dispatching mousedown on the >> button
        if not nav_success:
            for frame in page.frames:
                try:
                    # Find and click the >> button using mousedown event
                    btn = await frame.query_selector('input[name="forward2A"]')
                    if btn:
                        await btn.dispatch_event('mousedown')
                        await page.wait_for_timeout(100)
                        await btn.dispatch_event('mouseup')
                        nav_success = True
                        if iteration == 0:
                            logger.debug("Grid navigation: mousedown on forward2A")
                        break
                except:
                    pass
        
        # Log if navigation completely failed
        if not nav_success and iteration == 0:
            logger.warning("⚠ Grid navigation not working - GridBF not found and no forward2A button")
        
        # Wait for grid data to update
        await page.wait_for_timeout(1500)
        iteration += 1
    
    result = list(all_cases.values())
    
    logger.info(f"✓ Total unique cases: {len(result)}" + 
                (f" (expected {total_expected})" if total_expected else ""))
    
    return result

#   Gvals[4] = judgment dates, [5] = names, [6] = judgment type, [7] = judge

async def extract_gvals(page):
    """Extract the Gvals array from the page containing pleadings data."""
    
    for frame in page.frames:
        try:
            result = await frame.evaluate("""
                () => {
                    if (typeof Gvals === 'undefined' || !Gvals) {
                        return null;
                    }
                    
                    return {
                        // Cost summary (indices 0-3)
                        costs: Gvals[0] ? Array.from(Gvals[0]) : null,
                        judgment_amount: Gvals[1] ? Array.from(Gvals[1]) : null,
                        interest: Gvals[2] ? Array.from(Gvals[2]) : null,
                        total: Gvals[3] ? Array.from(Gvals[3]) : null,
                        
                        // Pleadings data (indices 4-8) - PARALLEL ARRAYS
                        dates: Gvals[4] ? Array.from(Gvals[4]) : null,
                        codes: Gvals[5] ? Array.from(Gvals[5]) : null,
                        descriptions: Gvals[6] ? Array.from(Gvals[6]) : null,
                        details: Gvals[7] ? Array.from(Gvals[7]) : null,
                        image_paths: Gvals[8] ? Array.from(Gvals[8]) : null,
                        
                        // Party info (indices 10, 12)
                        plaintiff_arr: Gvals[10] ? Array.from(Gvals[10]) : null,
                        defendant_arr: Gvals[12] ? Array.from(Gvals[12]) : null,
                        
                        // Metadata
                        gvals_length: Gvals.length,
                        
                        // Line count from grid if visible
                        line_count: (() => {
                            const span = document.querySelector('span[id^="Gmvc"]');
                            if (span) {
                                const match = span.innerText.match(/of (\\d+) Lines/);
                                return match ? parseInt(match[1]) : null;
                            }
                            return null;
                        })()
                    };
                }
            """)
            
            if result:
                logger.info(f"✓ Extracted Gvals ({result.get('gvals_length', 0)} arrays)")
                return result
        except:
            continue
    
    return None


def parse_pleadings(gvals_data):
    """
    Convert parallel arrays from Gvals into list of pleading records.
    
    Gvals structure (parallel arrays - same index = same row):
        dates[i], codes[i], descriptions[i], details[i], image_paths[i]
    
    Returns list of dicts with date, code, description, detail, has_image
    """
    if not gvals_data:
        return []
    
    dates = gvals_data.get('dates') or []
    codes = gvals_data.get('codes') or []
    descriptions = gvals_data.get('descriptions') or []
    details = gvals_data.get('details') or []
    image_paths = gvals_data.get('image_paths') or []
    
    pleadings = []
    
    # Start at index 1 (index 0 is always empty in Gvals)
    max_len = max(len(dates), len(codes), len(descriptions), 1)
    
    for i in range(1, max_len):
        date = dates[i] if i < len(dates) else ''
        code = codes[i] if i < len(codes) else ''
        desc = descriptions[i] if i < len(descriptions) else ''
        detail = details[i] if i < len(details) else ''
        img_path = image_paths[i] if i < len(image_paths) else ''
        
        # Skip completely empty rows
        if not date and not code and not desc:
            continue
        
        pleadings.append({
            'date': date or '',
            'code': code or '',
            'description': desc or '',
            'detail': detail or '',
            'has_image': bool(img_path and img_path.strip())
        })
    
    return pleadings

    logger.info(f"Extracting pleadings for {case_number}...")

    if not gvals_data:

    }

def parse_judgments(gvals_data):
    """
    Parse judgment data from Gvals into structured list.
    
    Judgment table columns (from HTML):
    - P_61: Judg. Date
    - P_63: Name(s) 
    - P_64: Judgment (type/outcome)
    - P_66: Judge
    
    Gvals structure for Judgments tab typically:
    - Index 4: Judgment dates
    - Index 5: Names
    - Index 6: Judgment type
    - Index 7: Judge names
    """
    if not gvals_data:
        return []
    
    # Try to identify the arrays based on content patterns
    # Dates look like MM/DD/YYYY, Judgments contain POSSESSION/DISMISSED, etc.
    
    judgments = []
    line_count = gvals_data.get('line_count', 0)
    
    if line_count == 0:
        logger.info("  No judgment entries found")
        return []
    
    # The typical structure mirrors pleadings: parallel arrays at indices 4-7
    dates = gvals_data.get('arr_4') or []
    names = gvals_data.get('arr_5') or []
    judgment_types = gvals_data.get('arr_6') or []
    judges = gvals_data.get('arr_7') or []
    
    # Validate we have date-like data in arr_4
    # If not, try other indices
    if dates and len(dates) > 1:
        first_val = str(dates[1] if len(dates) > 1 else '')
        if not re.match(r'\d{2}/\d{2}/\d{4}', first_val):
            # Try to find dates in other arrays
            for i in range(4, 11):
                arr = gvals_data.get(f'arr_{i}') or []
                if arr and len(arr) > 1:
                    val = str(arr[1])
                    if re.match(r'\d{2}/\d{2}/\d{4}', val):
                        dates = arr
                        # Assume next arrays are names, types, judges
                        names = gvals_data.get(f'arr_{i+1}') or []
                        judgment_types = gvals_data.get(f'arr_{i+2}') or []
                        judges = gvals_data.get(f'arr_{i+3}') or []
                        break
    
    max_len = max(len(dates), len(names), len(judgment_types), len(judges), 1)
    
    for i in range(1, max_len):
        date = dates[i] if i < len(dates) else ''
        name = names[i] if i < len(names) else ''
        judgment_type = judgment_types[i] if i < len(judgment_types) else ''
        judge = judges[i] if i < len(judges) else ''
        
        # Skip empty rows
        if not date and not judgment_type:
            continue
        
        judgments.append({
            'judgment_date': date or '',
            'party_name': name or '',
            'judgment_type': judgment_type or '',
            'judge': judge or ''
        })
    
    return judgments

# After the grid scrape gives us a list of docket numbers, we iterate
# through each case to extract additional details:
#   1. Search for the case by docket number (P_21 field)
#   2. Click into the case detail view
#   3. Click the General tab
#   4. Click defendant's MORE INFO button (rightmost one)
#   5. Extract address from popup fields (P_211, P_212, P_213, P_214)
#   6. Dismiss popup and return to search
#   7. Geocode the address via Nominatim
#
# Each case gets a fresh search to avoid stale page state.

# --- Find content frame after navigating to a case ---
async def find_main_content_frame(page):
    """Find the main content frame after navigating to a case"""
    await page.wait_for_timeout(MEDIUM_WAIT)
    
    for frame in page.frames:
        try:
            content = await frame.content()
            if 'Defendant Name' in content or 'Plaintiff Name' in content or 'MORE INFORMATION' in content:
                logger.info("✓ Found main content frame")
                return frame
        except:
            continue
    
    return page.main_frame


# --- Search for a specific case by docket number ---
async def search_case(page, case_number):
    """Search for a case number"""
    logger.info(f"Searching for case: {case_number}")
    
    search_frame = await find_search_frame(page)
    
    search_filled = False
    
    text_inputs = await search_frame.query_selector_all('input[type="text"]')
    logger.info(f"Found {len(text_inputs)} text inputs in search frame")
    
    for inp in text_inputs:
        try:
            if await inp.is_visible():
                await inp.click()
                await page.wait_for_timeout(200)
                await inp.click(click_count=3)
                await page.wait_for_timeout(200)
                await page.keyboard.type(case_number, delay=75)
                search_filled = True
                logger.info(f"✓ Entered case number: {case_number}")
                break
        except Exception as e:
            logger.debug(f"Input failed: {e}")
            continue
    
    if not search_filled:
        logger.info("Trying JavaScript fallback for search...")
        for frame in page.frames:
            try:
                result = await frame.evaluate(f"""
                    () => {{
                        const inputs = document.querySelectorAll('input[type="text"]');
                        for (const input of inputs) {{
                            if (input.offsetParent !== null) {{
                                input.focus();
                                input.value = '{case_number}';
                                input.dispatchEvent(new Event('input', {{ bubbles: true }}));
                                input.dispatchEvent(new Event('change', {{ bubbles: true }}));
                                return true;
                            }}
                        }}
                        return false;
                    }}
                """)
                if result:
                    search_filled = True
                    search_frame = frame
                    logger.info("✓ Entered case number via JS")
                    break
            except:
                continue
    
    await page.wait_for_timeout(SHORT_WAIT)
    
    logger.info("Clicking Search button...")
    search_clicked = False
    
    try:
        buttons = await search_frame.query_selector_all('button, input[type="submit"], input[type="button"]')
        for btn in buttons:
            text = await btn.text_content() or ''
            value = await btn.get_attribute('value') or ''
            if 'search' in text.lower() or 'search' in value.lower():
                if await btn.is_visible():
                    await btn.click()
                    search_clicked = True
                    logger.info("✓ Clicked Search")
                    break
    except:
        pass
    
    if not search_clicked:
        await page.keyboard.press('Enter')
        logger.info("✓ Pressed Enter to search")
    
    await page.wait_for_timeout(LONG_WAIT)


# --- Click into a case from search results ---
async def click_into_case(page, case_number):
    """Click into the case from search results.
    
    CaseLink renders search results as a grid of <input class="list-input"> elements,
    NOT as regular text in <tr> rows. Case numbers are in input VALUES, not innerText.
    We need to find the input containing the docket number and trigger navigation.
    """
    logger.info("Clicking into case...")
    
    # Check if already on detail page
    for frame in page.frames:
        try:
            content = await frame.content()
            if case_number in content and ('Pleading Information' in content or 'Plaintiff Name' in content):
                logger.info("✓ Already on case detail page")
                return True
        except:
            continue
    
    # Method 1: Find the input with the case number and click its row
    for frame in page.frames:
        try:
            clicked = await frame.evaluate(f"""
                () => {{
                    // CaseLink uses input.list-input elements with pfx attributes
                    // Case number is in inputs like pfx="P_102_1" etc
                    const inputs = document.querySelectorAll('input.list-input');
                    for (const inp of inputs) {{
                        if (inp.value === '{case_number}') {{
                            // Found the docket number input - click it to navigate
                            inp.click();
                            return {{ success: true, method: 'input_click' }};
                        }}
                    }}
                    
                    // Fallback: try double-click on the input (some grids need this)
                    for (const inp of inputs) {{
                        if (inp.value === '{case_number}') {{
                            const dblClick = new MouseEvent('dblclick', {{ bubbles: true }});
                            inp.dispatchEvent(dblClick);
                            return {{ success: true, method: 'input_dblclick' }};
                        }}
                    }}
                    
                    return {{ success: false }};
                }}
            """)
            if clicked and clicked.get('success'):
                logger.info(f"✓ Clicked case via {clicked.get('method')}")
                await page.wait_for_timeout(LONG_WAIT)
                return True
        except:
            continue
    
    # Method 2: Try PostBack with the case number (CaseLink navigation pattern)
    for frame in page.frames:
        try:
            clicked = await frame.evaluate(f"""
                () => {{
                    // Some CaseLink grids use PostBack for row selection
                    if (typeof PostBack === 'function') {{
                        // Find the row index containing our case
                        const inputs = document.querySelectorAll('input.list-input');
                        for (const inp of inputs) {{
                            if (inp.value === '{case_number}') {{
                                const pfx = inp.getAttribute('pfx') || '';
                                const match = pfx.match(/P_\\d+_(\\d+)/);
                                if (match) {{
                                    const rowIdx = match[1];
                                    // Try clicking the row via PostBack
                                    PostBack('', 'WTKROW_' + rowIdx, '', 1, 0);
                                    return {{ success: true, method: 'PostBack_row', row: rowIdx }};
                                }}
                            }}
                        }}
                    }}
                    return {{ success: false }};
                }}
            """)
            if clicked and clicked.get('success'):
                logger.info(f"✓ Clicked case via {clicked.get('method')} (row {clicked.get('row')})")
                await page.wait_for_timeout(LONG_WAIT)
                return True
        except:
            continue
    
    # Method 3: Find rows containing the case number in any child input
    for frame in page.frames:
        try:
            clicked = await frame.evaluate(f"""
                () => {{
                    const rows = document.querySelectorAll('tr');
                    for (const row of rows) {{
                        // Check both innerText AND input values
                        const rowInputs = row.querySelectorAll('input');
                        let hasCaseNum = row.innerText.includes('{case_number}');
                        
                        if (!hasCaseNum) {{
                            for (const inp of rowInputs) {{
                                if (inp.value && inp.value.includes('{case_number}')) {{
                                    hasCaseNum = true;
                                    break;
                                }}
                            }}
                        }}
                        
                        if (hasCaseNum) {{
                            if (row.onclick) {{
                                row.click();
                                return {{ success: true, method: 'tr_onclick' }};
                            }}
                            const cell = row.querySelector('td');
                            if (cell) {{
                                cell.click();
                                return {{ success: true, method: 'td_click' }};
                            }}
                        }}
                    }}
                    return {{ success: false }};
                }}
            """)
            if clicked and clicked.get('success'):
                logger.info(f"✓ Clicked case via {clicked.get('method')}")
                await page.wait_for_timeout(LONG_WAIT)
                return True
        except:
            continue
    
    logger.warning(f"Could not click into case row for {case_number}")
    return False


# --- Click the General tab (where defendant info lives) ---
async def click_general_tab(page):
    """Click the General tab to access defendant information"""
    logger.info("Looking for General tab...")
    
    main_frame = await find_main_content_frame(page)
    
    tab_clicked = False
    
    tab_selectors = [
        'text=General',
        'a:has-text("General")',
        'button:has-text("General")',
        'td:has-text("General")',
        '[class*="tab"]:has-text("General")',
    ]
    
    for selector in tab_selectors:
        try:
            tab = await main_frame.query_selector(selector)
            if tab and await tab.is_visible():
                await tab.click()
                tab_clicked = True
                logger.info(f"✓ Clicked General tab via {selector}")
                break
        except:
            continue
    
    if not tab_clicked:
        try:
            result = await main_frame.evaluate("""
                () => {
                    const elements = document.querySelectorAll('a, td, button, span, div');
                    for (const el of elements) {
                        const text = el.innerText?.trim();
                        if (text === 'General') {
                            el.click();
                            return true;
                        }
                    }
                    return false;
                }
            """)
            if result:
                tab_clicked = True
                logger.info("✓ Clicked General tab via JS")
        except Exception as e:
            logger.debug(f"JS tab click failed: {e}")
    
    if not tab_clicked:
        for frame in page.frames:
            try:
                result = await frame.evaluate("""
                    () => {
                        const elements = document.querySelectorAll('a, td, button, span');
                        for (const el of elements) {
                            if (el.innerText?.trim() === 'General') {
                                el.click();
                                return true;
                            }
                        }
                        return false;
                    }
                """)
                if result:
                    tab_clicked = True
                    logger.info("✓ Clicked General tab in frame via JS")
                    break
            except:
                continue
    
    if tab_clicked:
        await page.wait_for_timeout(MEDIUM_WAIT)
    else:
        logger.warning("⚠ Could not find General tab")
    
    return tab_clicked


# --- Extract defendant address from MORE INFO popup ---
async def get_defendant_address(page, case_number):
    """Extract defendant address from the MORE INFORMATION popup.
    Returns empty address dict if info not available or times out.
    NEVER blocks - always returns within timeout."""
    logger.info(f"Getting defendant address for {case_number}")
    
    # Default empty address
    address = {
        'name': '',
        'address1': '',
        'address2': '',
        'city_state_zip': '',
        'phone': '',
        'full_address': ''
    }
    
    async def dismiss_all_popups():
        """Aggressively dismiss any popups - OK, Close, Cancel buttons"""
        for _ in range(3):  # Try multiple times
            # Press Enter first
            try:
                await page.keyboard.press('Enter')
                await page.wait_for_timeout(300)
            except:
                pass
            
            # Click any dismiss buttons
            for frame in page.frames:
                try:
                    await frame.evaluate("""
                        () => {
                            const btns = document.querySelectorAll('button, input[type="button"]');
                            for (const btn of btns) {
                                const text = (btn.innerText || btn.value || '').toUpperCase().trim();
                                if (text === 'OK' || text === 'CLOSE' || text === 'CANCEL') {
                                    btn.click();
                                }
                            }
                        }
                    """)
                except:
                    pass
            await page.wait_for_timeout(200)
    
    try:
        async def _get_address():
            nonlocal address
            
            # Click defendant MORE INFO button - find by looking for Defendant Name section
            button_clicked = False
            
            for frame_idx, frame in enumerate(page.frames):
                try:
                    clicked = await frame.evaluate("""
                        () => {
                            // Find the "Defendant Name" header
                            const allCells = document.querySelectorAll('td, th, div');
                            let defendantSection = null;
                            
                            for (const cell of allCells) {
                                const text = (cell.innerText || '').trim();
                                if (text === 'Defendant Name') {
                                    // Found it - get the parent table/section
                                    defendantSection = cell.closest('table');
                                    break;
                                }
                            }
                            
                            if (defendantSection) {
                                // Look for MORE INFO button in this section
                                const btns = defendantSection.querySelectorAll('button, input[type="button"]');
                                for (const btn of btns) {
                                    const text = (btn.innerText || btn.value || '').toUpperCase();
                                    if (text.includes('MORE') && text.includes('INFO')) {
                                        btn.click();
                                        return { success: true, method: 'defendant_section' };
                                    }
                                }
                            }
                            
                            // Fallback: find rightmost MORE INFO button
                            const allBtns = Array.from(document.querySelectorAll('button, input[type="button"]'));
                            const moreInfoBtns = allBtns.filter(b => {
                                const text = (b.innerText || b.value || '').toUpperCase();
                                return text.includes('MORE') && text.includes('INFO');
                            });
                            
                            if (moreInfoBtns.length === 0) {
                                return { success: false, reason: 'no MORE INFO buttons' };
                            }
                            
                            // Sort by X position, click rightmost (defendant is on right)
                            moreInfoBtns.sort((a, b) => {
                                return b.getBoundingClientRect().x - a.getBoundingClientRect().x;
                            });
                            
                            // Only click if it's clearly on the right side
                            const btn = moreInfoBtns[0];
                            const rect = btn.getBoundingClientRect();
                            const pageWidth = window.innerWidth || 1200;
                            
                            if (rect.x > pageWidth * 0.4) {  // Must be past 40% of page
                                btn.click();
                                return { success: true, method: 'rightmost', x: rect.x };
                            }
                            
                            return { success: false, reason: 'no button on right side' };
                        }
                    """)
                    
                    if clicked and clicked.get('success'):
                        logger.info(f"✓ Clicked defendant MORE INFO ({clicked.get('method', 'unknown')})")
                        button_clicked = True
                        await page.wait_for_timeout(MEDIUM_WAIT)
                        break
                        
                except Exception as e:
                    continue
            
            if not button_clicked:
                logger.info(f"No defendant MORE INFO button - skipping {case_number}")
                return address
            
            # Check for Invalid Password popup IMMEDIATELY
            await page.wait_for_timeout(500)
            for frame in page.frames:
                try:
                    has_error = await frame.evaluate("""
                        () => {
                            const text = document.body.innerText || '';
                            return text.includes('Invalid Password') || text.includes('Session Expired');
                        }
                    """)
                    if has_error:
                        logger.warning(f"⚠ Invalid Password/Session detected - dismissing")
                        await dismiss_all_popups()
                        return address  # Return empty, don't block
                except:
                    pass
            
            # Try to extract address from popup
            for frame in page.frames:
                try:
                    address_data = await frame.evaluate("""
                        () => {
                            const data = {};
                            
                            const nameField = document.querySelector('input[name="P_211"]');
                            const addr1Field = document.querySelector('input[name="P_212"]');
                            const addr2Field = document.querySelector('input[name="P_213"]');
                            const cszField = document.querySelector('input[name="P_214"]');
                            const phoneField = document.querySelector('input[name="P_27"]');
                            
                            if (nameField) data.name = nameField.value || nameField.getAttribute('saveval') || '';
                            if (addr1Field) data.address1 = addr1Field.value || addr1Field.getAttribute('saveval') || '';
                            if (addr2Field) data.address2 = addr2Field.value || addr2Field.getAttribute('saveval') || '';
                            if (cszField) data.city_state_zip = cszField.value || cszField.getAttribute('saveval') || '';
                            if (phoneField) data.phone = phoneField.value || phoneField.getAttribute('saveval') || '';
                            
                            return data;
                        }
                    """)
                    
                    if address_data and address_data.get('address1'):
                        address.update(address_data)
                        logger.info(f"✓ Extracted address: {address_data.get('address1')}")
                        break
                except:
                    continue
            
            # Build full address
            address['full_address'] = ' '.join(filter(None, [
                address.get('address1', ''),
                address.get('address2', ''),
                address.get('city_state_zip', '')
            ])).strip()
            
            # Always close popup
            await dismiss_all_popups()
            
            return address
        
        # Run with strict timeout (6 seconds max)
        result = await asyncio.wait_for(_get_address(), timeout=6.0)
        return result
        
    except asyncio.TimeoutError:
        logger.warning(f"⚠ Timeout getting address for {case_number} - dismissing popups and continuing")
        await dismiss_all_popups()
        return address
        
    except Exception as e:
        logger.warning(f"⚠ Error getting address for {case_number}: {e}")
        await dismiss_all_popups()
        return address


# --- Navigate back to search page ---
async def return_to_search(page, worker_id=1):
    """Navigate back to the main search page after processing a case.
    Uses LOGIN_URL which redirects to search when already logged in."""
    try:
        # Check current URL
        current_url = page.url
        logger.info(f"[Worker {worker_id}] return_to_search: current URL = {current_url}")
        
        # Always navigate to LOGIN_URL - it redirects to search if already logged in
        # This is more reliable than trying to detect states

        try:
            await page.goto(LOGIN_URL, wait_until='domcontentloaded')
            await page.wait_for_timeout(LONG_WAIT)
        except:
            pass


async def enrich_cases_with_details(page, cases):
    """
    Iterate through scraped cases, click into each one to extract
    defendant address, then geocode. This is the "walk back through
    the list" step.
    
    For each case:
      1. Navigate to search page
      2. Search for the docket number
      3. Click into the case
      4. Click General tab → defendant MORE INFO → extract address
      5. Return to search for the next case
    """
    if not cases:
        return cases
    
    logger.info(f"\n{'=' * 60}")
    logger.info(f"  ENRICHING {len(cases)} CASES WITH ADDRESSES")
    logger.info(f"{'=' * 60}")
    
    enriched_count = 0
    
    for i, case in enumerate(cases, 1):
        docket = case.get('Docket_Number', '')
        if not docket:
            continue
        
        # Skip if already has an address
        if case.get('address_formatted') or case.get('Address_1'):
            logger.info(f"  [{i}/{len(cases)}] {docket} — already has address, skipping")
            enriched_count += 1
            continue
        
        logger.info(f"\n  [{i}/{len(cases)}] {docket}")
        
        try:
            # Fresh login for each case - CaseLink requires this to
            # reliably load the search frame. This is how the production
            # scraper (process_single_case) works.
            await page.goto(LOGIN_URL, wait_until='domcontentloaded')
            await page.wait_for_timeout(LONG_WAIT)
            if not await do_login(page):
                logger.warning(f"    Login failed, skipping {docket}")
                continue
            await page.wait_for_timeout(MEDIUM_WAIT)
            
            # Search for this specific case
            await asyncio.wait_for(search_case(page, docket), timeout=15.0)
            
            # Click into the case
            if not await click_into_case(page, docket):
                logger.warning(f"    Could not click into {docket}")
                continue
            
            # Extract pleadings FIRST — case opens on Pleadings tab by default
            # and Gvals gets replaced when you click other tabs
            pleadings_data = await extract_pleadings(page, docket)
            
            # Extract judgments SECOND — click Judgments tab, read Gvals
            judgments_data = await extract_judgments(page, docket)
            if judgments_data.get('judgments'):
                pleadings_data['judgments'] = judgments_data['judgments']
                logger.info(f"    ✓ Got {len(judgments_data['judgments'])} judgments")
            
            # NOW click General tab for defendant address
            await click_general_tab(page)
            
            # Extract defendant address from MORE INFO popup
            address = await get_defendant_address(page, docket)
            
            if address.get('full_address'):
                case['Address_1'] = address.get('address1', '')
                case['Address_2'] = address.get('address2', '')
                case['city_state_zip'] = address.get('city_state_zip', '')
                case['address_formatted'] = address['full_address']
                case['defendant_name'] = address.get('name', '')
                enriched_count += 1
                logger.info(f"    ✓ Address: {address['full_address']}")
            else:
                logger.info(f"    No address found for {docket}")
            
            # Store pleadings and judgments
            if pleadings_data.get('pleadings'):
                case['pleadings'] = pleadings_data['pleadings']
            if pleadings_data.get('judgments'):
                case['judgments'] = pleadings_data['judgments']
        
        except asyncio.TimeoutError:
            logger.warning(f"    Timeout processing {docket}")
        except Exception as e:
            logger.warning(f"    Error processing {docket}: {e}")
    
    logger.info(f"\n  ✓ Enriched {enriched_count}/{len(cases)} cases with addresses")
    return cases

async def quick_status_check(page, docket):
    """Search by docket, read status from grid row 1, don't click in.
    Returns dict with docket/status/file_date or None."""
    search_frame = await find_search_frame(page)

    filled = False
    text_inputs = await search_frame.query_selector_all('input[type="text"]')
    for inp in text_inputs:
        try:
            if await inp.is_visible():
                await inp.click()
                await page.wait_for_timeout(200)
                await inp.click(click_count=3)
                await page.wait_for_timeout(200)
                await page.keyboard.type(docket, delay=75)
                filled = True
                break
        except Exception as e:
            logger.debug(f"  quick_status fill attempt failed: {e}")
            continue

    if not filled:
        # JS fallback
        for frame in page.frames:
            try:
                result = await frame.evaluate(f"""
                    () => {{
                        const inputs = document.querySelectorAll('input[type="text"]');
                        for (const input of inputs) {{
                            if (input.offsetParent !== null) {{
                                input.focus();
                                input.value = '{docket}';
                                input.dispatchEvent(new Event('input', {{ bubbles: true }}));
                                input.dispatchEvent(new Event('change', {{ bubbles: true }}));
                                return true;
                            }}
                        }}
                        return false;
                    }}
                """)
                if result:
                    filled = True
                    break
            except Exception as e:
                logger.debug(f"  quick_status JS fallback failed: {e}")
                continue

    if not filled:
        return None

    await page.wait_for_timeout(SHORT_WAIT)

    # Click search
    search_clicked = False
    try:
        buttons = await search_frame.query_selector_all(
            'button, input[type="submit"], input[type="button"]')
        for btn in buttons:
            text = await btn.text_content() or ''
            value = await btn.get_attribute('value') or ''
            if 'search' in text.lower() or 'search' in value.lower():
                if await btn.is_visible():
                    await btn.click()
                    search_clicked = True
                    break
    except Exception as e:
        logger.debug(f"  quick_status search click failed: {e}")
    if not search_clicked:
        await page.keyboard.press('Enter')

    await page.wait_for_timeout(LONG_WAIT)

    # Read first grid row only
    for frame in page.frames:
        try:
            result = await frame.evaluate("""
                () => {
                    const inputs = document.querySelectorAll('input.list-input');
                    const data = {};
                    for (const inp of inputs) {
                        const pfx = inp.getAttribute('pfx') || '';
                        const val = inp.value || '';
                        if (!pfx || !val) continue;
                        if (pfx === 'P_102_1') data.docket = val;
                        if (pfx === 'P_103_1') data.status = val;
                        if (pfx === 'P_104_1') data.file_date = val;
                        if (pfx === 'P_106_1') data.plaintiff = val;
                        if (pfx === 'P_107_1') data.defendant = val;
                    }
                    return Object.keys(data).length > 0 ? data : null;
                }
            """)
            if result and result.get('docket'):
                return result
        except Exception as e:
            logger.debug(f"  quick_status grid read failed: {e}")
            continue

    return None

# SWEEP DROP MERGE
# ============================================================================

def merge_sweep_drops(cases, now_iso, cycle_events, max_files=400, checked=None):
    """Fold the parallel sweep workers' grid observations into state.

    The sweep workers walk date windows backwards and read the results grid,
    which carries every docket's CURRENT status a few hundred rows at a time.
    They are pure observers -- this function is the only place their findings
    become state, which keeps a single writer over the case store even with
    several scrapers running.

    Drops carry no defendant names by design (the primary stores those
    encrypted; a plaintext copy in a drop file would be a new exposure for no
    gain). A docket nobody has seen before is created bare and flagged for
    enrichment, which fills the name in through the normal path.

    Returns (files, rows, changed, new).
    """
    if not os.path.isdir(SWEEP_DROP_DIR):
        return 0, 0, 0, 0
    try:
        names = sorted(n for n in os.listdir(SWEEP_DROP_DIR)
                       if n.endswith(".jsonl"))
    except OSError:
        return 0, 0, 0, 0
    if not names:
        return 0, 0, 0, 0

    os.makedirs(SWEEP_USED_DIR, exist_ok=True)
    files = rows = changed = new = 0

    # Grid columns that are safe to backfill onto an existing case. Defendant
    # name is deliberately absent.
    CARRY = [('Plaintiff_1', 'Plaintiff_1'), ('Attorney', 'Attorney'),
             ('Def_Attorney', 'Def_Attorney'), ('file_date', 'file_date'),
             ('office', 'office'), ('case_type', 'case_type')]

    for name in names[:max_files]:
        p = os.path.join(SWEEP_DROP_DIR, name)
        try:
            with open(p) as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        r = json.loads(line)
                    except Exception:
                        continue
                    if r.get("_meta"):
                        continue
                    docket = (r.get("docket") or "").strip()
                    if not docket:
                        continue
                    rows += 1
                    observed = r.get("observed_at") or now_iso
                    st = (r.get("status") or "").strip()

                    if docket not in cases:
                        cases[docket] = {
                            'status': st,
                            'file_date': r.get('file_date', ''),
                            'case_type': r.get('case_type', ''),
                            'office': r.get('office', ''),
                            'Plaintiff_1': r.get('Plaintiff_1', ''),
                            'Defendant_1': '',
                            'Attorney': r.get('Attorney', ''),
                            'Def_Attorney': r.get('Def_Attorney', ''),
                            'first_seen': observed,
                            'last_checked': observed,
                            'last_changed': observed,
                            'enriched': False,
                            'needs_enrichment': True,
                        }
                        new += 1
                        emit_event(cycle_events, observed, 'INS',
                                   docket=docket, subtype='sweep',
                                   data={'status': st,
                                         'file_date': r.get('file_date', ''),
                                         'src': r.get('src', '')})
                        continue

                    case = cases[docket]
                    if checked is not None:
                        checked.add(docket)   # -> this cycle's NUL op
                    if st and st != case.get('status', ''):
                        old = case.get('status', '')
                        case['status'] = st
                        case['last_changed'] = observed
                        # A status move is exactly when the detail page has
                        # something new on it, so queue the full drill.
                        case['needs_enrichment'] = True
                        changed += 1
                        emit_event(cycle_events, observed, 'SEG',
                                   docket=docket, subtype='status',
                                   data={'old': old, 'new': st,
                                         'src': r.get('src', '')})
                    # Fill gaps the original grid row never had, but never
                    # overwrite something already known.
                    for src_k, dst_k in CARRY:
                        v = r.get(src_k)
                        if v and not case.get(dst_k):
                            case[dst_k] = v
                    if observed > (case.get('last_checked') or ''):
                        case['last_checked'] = observed
            files += 1
            os.replace(p, os.path.join(SWEEP_USED_DIR, name))
        except Exception as e:
            logger.warning(f"  sweep merge: {name}: {type(e).__name__}: "
                           f"{str(e)[:120]}")

    # Consumed drops are ~60KB each and a full pass produces a few hundred, so
    # a generous retention quietly costs hundreds of MB on a disk that is
    # already at 81%. Keep a short tail for debugging, nothing more.
    try:
        used = sorted(os.listdir(SWEEP_USED_DIR))
        for stale in used[:-int(os.environ.get("CASELINK_KEEP_CONSUMED", "40"))]:
            os.unlink(os.path.join(SWEEP_USED_DIR, stale))
    except OSError:
        pass

    return files, rows, changed, new


def write_pending_index(cases, terminal):
    """Tell the sweep workers which date windows still hold OPEN cases.

    A window whose every case is CLOSED cannot change status, so re-reading it
    is wasted court traffic. With ~69.6k of 78.1k cases closed and the open
    ones concentrated in the last two years, this is the difference between
    walking all 345 windows and walking the ~90 that can still move.
    """
    try:
        import sweep_windows
    except Exception:
        return 0
    counts = {}
    for c in cases.values():
        if c.get('status') in terminal:
            continue
        idx = sweep_windows.index_for_file_date(c.get('file_date'))
        if idx is None:
            continue
        counts[idx] = counts.get(idx, 0) + 1
    try:
        os.makedirs(SWEEP_DIR, exist_ok=True)
        tmp = os.path.join(SWEEP_DIR, "pending_index.json.tmp")
        with open(tmp, "w") as f:
            json.dump({"generated_at": datetime.now(CT).isoformat(),
                       "open_total": sum(counts.values()),
                       "windows": {str(k): v for k, v in sorted(counts.items())}},
                      f, separators=(",", ":"))
        os.replace(tmp, os.path.join(SWEEP_DIR, "pending_index.json"))
        os.chmod(os.path.join(SWEEP_DIR, "pending_index.json"), 0o644)
    except Exception as e:
        logger.warning(f"  pending index write failed: {e}")
        return 0
    return len(counts)

async def run(mode='daemon', watch=False, backfill_to=None,
              rekey_old=None, rekey_new=None):

    # ── Hydrate state from the archive.org chain ──
    logger.info("Hydrating state from archive.org chain...")
    async with aiohttp.ClientSession() as boot_session:
        hydrated = await eo_chain.hydrate_state(boot_session)
    state = {'updated_at': None, 'cases': hydrated.get('cases') or {}}

    # Replay the LOCAL today.jsonl over the cold tier so a mid-day restart
    # loses nothing (the day is sealed to archive.org only at rollover).
    _today_ops = eo_chain._read_today_local()
    if _today_ops:
        _replayed = eo_chain.apply_transformations(state, _today_ops)
        state['cases'] = _replayed.get('cases') or state['cases']
        logger.info(f"  boot: replayed {len(_today_ops)} local today.jsonl ops")

    # Chain + today.jsonl store ENC: ciphertext for defendant fields; decrypt
    # for runtime. decrypt_name passes through any already-cleartext values.
    decrypt_state(state)

    # Track the head pointer locally so we can link the next block without
    # waiting for archive.org's search index to catch up (30-60 min lag).
    head = eo_chain.ChainHead(CHAIN_HEAD_FILE)
    if hydrated.get('head_id') and hydrated['head_id'] != head.data.get('head_id'):
        head.data['head_id']    = hydrated['head_id']
        head.data['next_index'] = hydrated['next_index']
        head.save()

    # Tuning params (DEF) ride in each block's header. Pull the last
    # snapshot from hydration; fall back to defaults.
    proc = default_processing()
    if hydrated.get('last_def_params'):
        proc['def'].update(hydrated['last_def_params'])
    if not proc.get('history'):
        proc['history'] = [{
            'ts': datetime.now(CT).isoformat(), 'op': 'DEF',
            'params': copy.deepcopy(proc['def']),
        }]

    params = proc['def']
    apply_tuning(params)
    # CaseLink's grid only ever emits PENDING or CLOSED for this case type --
    # DISMISSED/NONSUIT never appear, so the original terminal set retired
    # nothing and every one of the ~78k cases counted as "pending". That made
    # JOB 2's queue ~9x too big and sorted stalest-first, i.e. it would have
    # spent weeks re-checking closed 2020 cases before reaching a live one.
    # CLOSED is terminal for the per-docket queue; a case that reopens is still
    # caught, because the backward sweep re-reads the grid for every case
    # regardless of the status we hold.
    terminal = set(params.get('terminal_statuses', ['DISMISSED', 'NONSUIT']))
    terminal.update(['CLOSED', 'DISMISSED', 'NONSUIT'])

    if backfill_to:
        proc['eva']['backfill_target'] = backfill_to

    # ── Rekey mode ──
    if mode == 'rekey':
        if not rekey_old or not rekey_new:
            logger.error("--rekey requires --old-key and --new-key")
            return
        n = len(state.get('cases', {}))
        logger.info(f"Re-encrypting {n} cases in memory "
                    f"(chain stores scrubbed cases — nothing to re-upload)...")
        encrypt_state(state, password=rekey_new)
        logger.info("Done. Update ENCRYPTION_PASSWORD in config to the new key.")
        return

    cases = state.get('cases', {})
    logger.info(f"\n{'=' * 60}")
    logger.info(f"  Nashville CaseLink Eviction Scraper")
    logger.info(f"  State: {len(cases)} cases, "
                f"{sum(1 for c in cases.values() if c.get('status') not in terminal)} pending")

    # Serve the full hydrated baseline immediately, before the long first
    # enrichment cycle, so the webhook has data within seconds of boot.
    try:
        eo_chain.write_snapshot(cases)
        logger.info(f"  boot: wrote initial snapshot ({len(cases)} cases) "
                    f"-> {eo_chain.SNAPSHOT_FILE}")
    except Exception as e:
        logger.error(f"  boot: initial snapshot failed: {e}")
    logger.info(f"  Backfill cursor: {proc['eva'].get('backfill_cursor', 'not started')}")
    logger.info(f"  Mode: {'single cycle' if mode == 'once' else 'daemon'}")
    logger.info(f"  Browser: {'visible' if watch else 'headless'}")
    logger.info(f"  Waits: {params.get('waits', {})}")
    logger.info(f"{'=' * 60}\n")

    async with async_playwright() as pw:
        browser = await pw.chromium.launch(
            headless=not watch, slow_mo=300 if watch else 0)
        context = await browser.new_context(viewport={'width': 1280, 'height': 900})
        page = await context.new_page()
        page.set_default_timeout(PAGE_TIMEOUT)

        try:
            while True:
                cycle_start = datetime.now()
                now_ct = datetime.now(CT)
                now_iso = now_ct.isoformat()
                errors = 0
                new_count = 0
                changed_count = 0
                checked_count = 0
                enriched_count = 0
                cycle_events = []  # EO event log for this cycle
                cycle_checked_dockets = set()  # for batched NUL at cycle end
                prev_cases_snapshot = copy.deepcopy(state.get('cases', {}))

                # Login
                if not await do_login(page):
                    logger.error("Login failed")
                    errors += 1
                    if mode == 'once':
                        break
                    await asyncio.sleep(60)
                    continue

                await find_search_frame(page)

                # ────────────────────────────────────────────────
                # JOB 0: Merge parallel sweep observations
                # ────────────────────────────────────────────────
                # Runs before discovery so anything the sweep flagged for
                # enrichment is already in the pool when JOB 3 picks its batch.
                try:
                    sf, sr, sc, sn = merge_sweep_drops(
                        cases, now_iso, cycle_events,
                        checked=cycle_checked_dockets)
                    if sf:
                        logger.info(f"\n── JOB 0: Merged {sf} sweep drop(s): "
                                    f"{sr} observations, {sc} status changes, "
                                    f"{sn} new ──")
                        changed_count += sc
                        new_count += sn
                        state['cases'] = cases
                        state['updated_at'] = now_iso
                except Exception as e:
                    logger.warning(f"  sweep merge failed: {e}")

                # Refresh the map of which windows still hold open cases, so
                # the sweep workers stop re-reading weeks that are fully closed.
                try:
                    _pi = write_pending_index(cases, terminal)
                    _open = sum(1 for v in cases.values()
                                if v.get('status') not in terminal)
                    logger.info(f"  pending index: {_open} open cases across "
                                f"{_pi} window(s)")
                except Exception as e:
                    logger.warning(f"  pending index failed: {e}")

                # ────────────────────────────────────────────────
                # JOB 1: Discovery (today sweep)
                # ────────────────────────────────────────────────
                discovery_last = proc['eva'].get('discovery_last')
                hours_since = 999
                if discovery_last:
                    try:
                        hours_since = (now_ct - datetime.fromisoformat(
                            discovery_last)).total_seconds() / 3600
                    except Exception:
                        pass

                if hours_since > params.get('discovery_interval_hours', 20):
                    # Gap-aware window: heal downtime gaps by sweeping from the
                    # day after the chain's last sealed block forward; capped 14d.
                    # Gap-aware window anchored on the newest filing we hold:
                    # sweep from there to today so downtime gaps heal, capped 21d.
                    d_end = now_ct
                    d_start = now_ct - timedelta(days=2)
                    _newest = None
                    for _c in cases.values():
                        _fd = (_c.get('file_date') or '').strip()
                        if not _fd:
                            continue
                        try:
                            _d = datetime.strptime(_fd, '%m/%d/%Y').replace(tzinfo=CT)
                        except Exception:
                            continue
                        if _newest is None or _d > _newest:
                            _newest = _d
                    if _newest is not None and _newest < d_start:
                        d_start = max(_newest, now_ct - timedelta(days=21))
                    logger.info(f"\n── JOB 1: Discovery "
                                f"({d_start.strftime('%m/%d')} - "
                                f"{d_end.strftime('%m/%d')}) ──")

                    grid_cases = await search_detainer_cases_by_date(
                        page, d_start, d_end)

                    for gc in grid_cases:
                        docket = gc.get('Docket_Number', '')
                        if not docket:
                            continue
                        cycle_checked_dockets.add(docket)
                        if docket not in cases:
                            cases[docket] = grid_case_to_entry(gc, now_iso)
                            new_count += 1
                            emit_event(cycle_events, now_iso, 'INS',
                                       docket=docket, subtype='discovery',
                                       data={
                                           'status': gc.get('Status', ''),
                                           'file_date': gc.get('File_Date', ''),
                                           'plaintiff': gc.get('Plaintiff_1', ''),
                                           'attorney': gc.get('Attorney', ''),
                                       })
                        else:
                            old_status = cases[docket].get('status', '')
                            new_status = gc.get('Status', '')
                            if new_status and new_status != old_status:
                                cases[docket]['status'] = new_status
                                cases[docket]['last_changed'] = now_iso
                                cases[docket]['needs_enrichment'] = True
                                changed_count += 1
                                emit_event(cycle_events, now_iso, 'SEG',
                                           docket=docket, subtype='status',
                                           data={'old': old_status,
                                                 'new': new_status})
                            cases[docket]['last_checked'] = now_iso

                    proc['eva']['discovery_last'] = now_iso
                    logger.info(f"  {len(grid_cases)} in grid, "
                                f"{new_count} new, {changed_count} changed")
                    await page.wait_for_timeout(MEDIUM_WAIT)

                    # Intermediate commit so discoveries aren't lost
                    if new_count > 0:
                        state['cases'] = cases
                        state['updated_at'] = now_iso
                        try:
                            eo_chain.write_snapshot(cases)
                            logger.info(f"  discovery: refreshed snapshot (+{new_count} new)")
                        except Exception as e:
                            logger.error(f"  discovery: snapshot refresh failed: {e}")
                else:
                    logger.info(f"\n── JOB 1: Discovery skipped "
                                f"({hours_since:.0f}h ago) ──")

                # ────────────────────────────────────────────────
                # JOB 2: Pending status checks (lightweight)
                # ────────────────────────────────────────────────
                # DEF: skip when enrichment backlog dominates OR
                #      REC has disabled it (budget=0)
                unenriched_count = sum(
                    1 for v in cases.values()
                    if not v.get('enriched') or v.get('needs_enrichment'))
                job2_skip_threshold = min(25, params.get(
                    'job2_skip_when_unenriched_above', 25))
                job2_consec_fail_limit = params.get(
                    'job2_consecutive_fail_limit', 3)
                budget = params.get('pending_budget', 50)

                j2_eva = proc['eva'].setdefault('job2', {})

                if budget == 0:
                    logger.info("\n── JOB 2: Disabled by REC"
                                " (0 budget) ──")
                elif unenriched_count > job2_skip_threshold:
                    logger.info(f"\n── JOB 2: Skipped ({unenriched_count}"
                                f" unenriched > {job2_skip_threshold}"
                                f" threshold) ──")
                else:
                    pending = sorted(
                        [(k, v) for k, v in cases.items()
                         if v.get('status') not in terminal
                         and v.get('last_checked', '') < now_iso],
                        key=lambda x: x[1].get('last_checked', '1970'))
                    batch = pending[:budget]

                    if batch:
                        logger.info(f"\n── JOB 2: Checking {len(batch)}/"
                                    f"{len(pending)} pending ──")

                        j2_consec_fails = 0

                        for docket, case in batch:
                            elapsed = (datetime.now()
                                       - cycle_start).total_seconds()
                            if elapsed > 900:
                                logger.info(
                                    f"  Time budget at {checked_count}"
                                    f" checks")
                                break

                            # Circuit breaker → hard reset → move on
                            if j2_consec_fails >= job2_consec_fail_limit:
                                logger.warning(
                                    f"  {j2_consec_fails} consecutive"
                                    f" fails — hard reset, moving to"
                                    f" Job 3")
                                await hard_reset(page)
                                break

                            try:
                                if not await hard_reset(page):
                                    errors += 1
                                    j2_consec_fails += 1
                                    case['last_checked'] = now_iso
                                    continue

                                result = await asyncio.wait_for(
                                    quick_status_check(page, docket),
                                    timeout=15.0)
                                checked_count += 1
                                cycle_checked_dockets.add(docket)
                                j2_eva['cycle_attempts'] = (
                                    j2_eva.get('cycle_attempts', 0) + 1)

                                if result and result.get('status'):
                                    j2_consec_fails = 0
                                    j2_eva['cycle_successes'] = (
                                        j2_eva.get(
                                            'cycle_successes', 0) + 1)
                                    old_st = case.get('status', '')
                                    new_st = result['status']
                                    if new_st != old_st:
                                        case['status'] = new_st
                                        case['last_changed'] = now_iso
                                        case['needs_enrichment'] = True
                                        changed_count += 1
                                        proc['eva']['pending_stats'][
                                            'hits'] = (
                                            proc['eva']['pending_stats']
                                            .get('hits', 0) + 1)
                                        logger.info(
                                            f"  {docket}:"
                                            f" {old_st} → {new_st}")
                                        emit_event(
                                            cycle_events, now_iso, 'SEG',
                                            docket=docket, subtype='status',
                                            data={'old': old_st,
                                                  'new': new_st})
                                    else:
                                        logger.info(
                                            f"  {docket}: still"
                                            f" {old_st}")
                                else:
                                    j2_consec_fails += 1
                                    logger.info(
                                        f"  {docket}: no result")

                                # Always mark checked
                                case['last_checked'] = now_iso
                                proc['eva']['pending_stats'][
                                    'checks'] = (
                                    proc['eva']['pending_stats']
                                    .get('checks', 0) + 1)

                            except asyncio.TimeoutError:
                                logger.warning(
                                    f"  Timeout checking {docket}")
                                errors += 1
                                j2_consec_fails += 1
                                case['last_checked'] = now_iso
                            except Exception as e:
                                logger.warning(
                                    f"  Error checking {docket}: {e}")
                                errors += 1
                                j2_consec_fails += 1
                                case['last_checked'] = now_iso

                # ────────────────────────────────────────────────
                # JOB 3: Enrichment (full drill per case)
                #   Priority queue items go first, then normal.
                # ────────────────────────────────────────────────
                pq = await load_priority_queue()
                priority_dockets = get_pending_priority(pq)
                pq_changed = False

                # Build priority batch: dockets from the queue that
                # exist in cases (even if already enriched — the user
                # requested a re-scrape by adding them).
                priority_batch = []
                for pd in priority_dockets:
                    if pd in cases:
                        # Force re-enrichment for priority items
                        cases[pd]['needs_enrichment'] = True
                        cases[pd]['enriched'] = False
                        priority_batch.append((pd, cases[pd]))
                    else:
                        # Docket not discovered yet — mark as error
                        mark_priority_error(
                            pq, pd,
                            'docket not found in cases (not yet discovered)')
                        pq_changed = True

                if priority_batch:
                    logger.info(f"\n  ★ Priority queue: {len(priority_batch)}"
                                f" items to enrich first")

                # ── Split unenriched into RECENT (≤14 days) vs OLD ──
                # Recent = filed in last 14 days (by file_date, not
                # first_seen). These ALWAYS get enriched first,
                # hardcoded, regardless of what processing.json says.
                RECENT_CUTOFF_DAYS = 14
                RECENT_MIN_BUDGET = 40  # hardcoded floor
                recent_cutoff_dt = (now_ct - timedelta(
                    days=RECENT_CUTOFF_DAYS)).replace(tzinfo=None)

                def parse_file_date(fd):
                    """Parse MM/DD/YYYY file_date to datetime."""
                    try:
                        return datetime.strptime(fd, '%m/%d/%Y')
                    except Exception:
                        return None

                all_unenriched = [
                    (k, v) for k, v in cases.items()
                    if not v.get('enriched') or v.get('needs_enrichment')]

                recent_unenriched = sorted(
                    [(k, v) for k, v in all_unenriched
                     if (parse_file_date(v.get('file_date', ''))
                         and parse_file_date(v['file_date'])
                         >= recent_cutoff_dt)],
                    key=lambda x: parse_file_date(
                        x[1].get('file_date', '01/01/1970'))
                    or datetime.min,
                    reverse=True)

                recent_docket_set = {k for k, _ in recent_unenriched}
                old_unenriched = sorted(
                    [(k, v) for k, v in all_unenriched
                     if k not in recent_docket_set],
                    key=lambda x: parse_file_date(
                        x[1].get('file_date', '01/01/1970'))
                    or datetime.min,
                    reverse=True)

                needs_enrich = all_unenriched  # for logging total

                # Idle-cycle boost: discovery found nothing new this
                # cycle (new_count == 0, set by JOB 1 above), so nothing is
                # competing with the backlog for budget -- spend more of
                # the cycle catching up on enrichment/addresses instead of
                # the token default. Bounded by the existing elapsed>1500s
                # cutoff either way. Added 2026-09-04 alongside the 24/7
                # schedule change.
                IDLE_ENRICHMENT_BUDGET = 300
                base_budget = max(RECENT_MIN_BUDGET,
                                   params.get('enrichment_budget', 40))
                e_budget = IDLE_ENRICHMENT_BUDGET if new_count == 0 else base_budget

                # Priority queue items first, then recent, then old
                remaining_budget = max(0, e_budget - len(priority_batch))
                pq_docket_set = {pd for pd, _ in priority_batch}

                recent_batch = [
                    (k, v) for k, v in recent_unenriched
                    if k not in pq_docket_set
                ][:remaining_budget]

                old_budget = max(0, remaining_budget - len(recent_batch))
                old_batch = [
                    (k, v) for k, v in old_unenriched
                    if k not in pq_docket_set
                ][:old_budget]

                # ── Tier 4: cases with NO address that no flag can reach ──
                # 49,849 cases carry enriched=True / needs_enrichment=False and
                # yet have no address at all, and none of them is marked
                # 'unfindable' -- so they were never actually attempted, they
                # were marked done in bulk. The selection above keys entirely
                # off those two flags, so nothing could ever pick them up: 64%
                # of the corpus had no address and no route to getting one.
                #
                # Rather than flip the flag on 50k records at once (which would
                # bloat every diff and let ancient filings crowd out this
                # week's), the backlog is computed here as a last tier. Recent
                # and flagged work always wins; this only spends what is left.
                # Open cases come first, then newest filings.
                addr_budget = max(0, remaining_budget - len(recent_batch)
                                  - len(old_batch))
                addr_batch = []
                if addr_budget > 0:
                    taken = pq_docket_set | {k for k, _ in recent_batch} \
                            | {k for k, _ in old_batch}
                    addressless = [
                        (k, v) for k, v in cases.items()
                        if k not in taken
                        and not (v.get('Address_1') or v.get('address_formatted'))
                        and v.get('address_status') != 'unfindable'
                        and (v.get('address_attempts') or 0) < 3
                    ]
                    addressless.sort(key=lambda x: (
                        x[1].get('status') in terminal,          # open first
                        -(parse_file_date(x[1].get('file_date', ''))
                          or datetime.min).toordinal(),          # newest first
                    ))
                    addr_batch = addressless[:addr_budget]
                    if addr_batch:
                        logger.info(
                            f"  ⌖ {len(addr_batch)} from the addressless "
                            f"backlog ({len(addressless)} remain, "
                            f"{sum(1 for _, v in addressless if v.get('status') not in terminal)} open)")

                e_batch = priority_batch + recent_batch + old_batch + addr_batch
                priority_docket_set = pq_docket_set

                if recent_batch:
                    logger.info(f"  ⚡ {len(recent_batch)} recent cases"
                                f" (filed ≤{RECENT_CUTOFF_DAYS}d) queued"
                                f" first out of {len(recent_unenriched)}"
                                f" recent unenriched")

                if e_batch:
                    logger.info(f"\n── JOB 3: Enriching {len(e_batch)}/"
                                f"{len(needs_enrich)} cases ──")

                    j3_consec_fails = 0
                    for docket, case in e_batch:
                        elapsed = (datetime.now() - cycle_start).total_seconds()
                        if elapsed > 1500:
                            logger.info(f"  Time budget at {enriched_count}")
                            break

                        # If we've failed 5 straight, hard reset first
                        if j3_consec_fails >= 5:
                            logger.warning(
                                f"  {j3_consec_fails} consecutive"
                                f" enrichment fails — hard reset")
                            if not await hard_reset(page):
                                logger.error("  Hard reset failed, ending Job 3")
                                break
                            j3_consec_fails = 0

                        try:
                            if not await hard_reset(page):
                                errors += 1
                                j3_consec_fails += 1
                                continue

                            proc['eva']['enrichment_stats']['attempted'] = (
                                proc['eva']['enrichment_stats'].get(
                                    'attempted', 0) + 1)

                            await asyncio.wait_for(
                                search_case(page, docket), timeout=15.0)

                            if not await click_into_case(page, docket):
                                errors += 1
                                j3_consec_fails += 1
                                continue

                            pld = await extract_pleadings(page, docket)
                            if pld.get('pleadings'):
                                case['pleadings'] = pld['pleadings']

                            # The Judgments tab click cost ~4.1s per case and
                            # returned nothing new: every Gvals array is
                            # already populated when the case opens, and
                            # parse_judgments reads the same arrays 4-7 that
                            # parse_pleadings just read -- which is why the two
                            # lists always came out the same length. Derive it
                            # from the payload we already have. Output is
                            # byte-identical to the tab-click path.
                            _g = pld.get('_gvals') or {}
                            jud_rows = parse_judgments({
                                'arr_4': _g.get('dates'),
                                'arr_5': _g.get('codes'),
                                'arr_6': _g.get('descriptions'),
                                'arr_7': _g.get('details'),
                                'line_count': _g.get('line_count') or 0,
                            }) if _g else []
                            if jud_rows:
                                case['judgments'] = jud_rows

                            # Only go after the address when we don't have one.
                            # The General tab click plus the defendant MORE INFO
                            # popup is ~9s, and an address does not change once
                            # recorded -- so on a re-drill triggered by a status
                            # change it is pure cost.
                            had_addr = bool(case.get('Address_1')
                                            or case.get('address_formatted'))
                            if had_addr:
                                addr = {}
                                got_addr = True
                            else:
                                await click_general_tab(page)
                                addr = await get_defendant_address(page, docket)
                                got_addr = bool(addr.get('full_address'))
                            if got_addr and addr.get('full_address'):
                                case['Address_1'] = addr.get('address1', '')
                                case['Address_2'] = addr.get('address2', '')
                                case['city_state_zip'] = addr.get(
                                    'city_state_zip', '')
                                case['address_formatted'] = addr['full_address']
                                case['defendant_name'] = addr.get('name', '')
                                case['address_status'] = 'success'
                                case['enriched'] = True
                                case['needs_enrichment'] = False
                            elif had_addr:
                                # Re-drill of a case we already had an address
                                # for: events refreshed, address left alone.
                                case['enriched'] = True
                                case['needs_enrichment'] = False
                            else:
                                # Always count the attempt. Tier-4 cases are
                                # selected by "has no address", not by a flag,
                                # so without this a case the court genuinely has
                                # no address for would be retried forever.
                                addr_attempts = case.get('address_attempts', 0) + 1
                                case['address_attempts'] = addr_attempts
                                case['address_last_attempt'] = now_iso
                                if addr_attempts >= 3:
                                    case['address_status'] = 'unfindable'
                                    case['enriched'] = True
                                    case['needs_enrichment'] = False
                                else:
                                    case['address_status'] = 'no_result'
                                    case['enriched'] = False
                                    case['needs_enrichment'] = True

                            case['last_checked'] = now_iso
                            case['last_changed'] = now_iso
                            if got_addr:
                                enriched_count += 1
                                proc['eva']['enrichment_stats']['succeeded'] = (
                                    proc['eva']['enrichment_stats'].get(
                                        'succeeded', 0) + 1)
                            j3_consec_fails = 0
                            logger.info(f"  ✓ {docket}" if got_addr else
                                        f"  ~ {docket} (no address, attempt {case['address_attempts']})")
                            cycle_checked_dockets.add(docket)

                            # Mark priority item done
                            if docket in priority_docket_set:
                                mark_priority_done(pq, docket)
                                pq_changed = True
                                logger.info(f"  ★ Priority item {docket} done")

                            emit_event(cycle_events, now_iso, 'CON',
                                       docket=docket, subtype='enrich',
                                       data={
                                           'address': case.get(
                                               'address_formatted', ''),
                                           'pleadings_count': len(
                                               case.get('pleadings', [])),
                                       })

                            # Geocode immediately
                            if ((case.get('address_formatted')
                                 or case.get('Address_1'))
                                    and not case.get('geocode_status')):
                                await geocode_all_cases([case])
                                if case.get('geocode_status') == 'success':
                                    emit_event(
                                        cycle_events, now_iso, 'CON',
                                        docket=docket, subtype='geocode',
                                        data={
                                            'lat': case.get('latitude'),
                                            'lon': case.get('longitude'),
                                        })
                                if case.get('council_district'):
                                    emit_event(
                                        cycle_events, now_iso, 'CON',
                                        docket=docket, subtype='districts',
                                        data={k: case[k] for k in [
                                            'council_district',
                                            'council_rep',
                                            'state_house_district',
                                            'us_congressional_district',
                                        ] if case.get(k)})

                            # State stays in memory; cycle-end block emit
                            # captures all changes.
                            state['cases'] = cases
                            state['updated_at'] = now_iso

                        except asyncio.TimeoutError:
                            logger.warning(f"  Timeout enriching {docket}")
                            errors += 1
                            j3_consec_fails += 1
                            if docket in priority_docket_set:
                                mark_priority_error(pq, docket, 'timeout')
                                pq_changed = True
                        except Exception as e:
                            logger.warning(f"  Error enriching {docket}: {e}")
                            errors += 1
                            j3_consec_fails += 1
                            if docket in priority_docket_set:
                                mark_priority_error(
                                    pq, docket, str(e)[:120])
                                pq_changed = True

                # Save priority queue if anything changed
                if pq_changed:
                    done_count = sum(
                        1 for i in pq['queue'] if i['status'] == 'done')
                    # Prune completed items older than 24h
                    cutoff = (datetime.now(CT)
                              - timedelta(hours=24)).isoformat()
                    pq['queue'] = [
                        i for i in pq['queue']
                        if not (i['status'] == 'done'
                                and i.get('completed_at', '') < cutoff)]
                    await save_priority_queue(
                        pq,
                        f"priority: {done_count} done,"
                        f" {len(get_pending_priority(pq))} pending")

                # ────────────────────────────────────────────────
                # JOB 3.5: Re-geocode (fix past failures + missing districts)
                # ────────────────────────────────────────────────
                # No budget limit — these are fast API calls, not scraping.
                # Rate limit (1/sec) is in geocode_all_cases itself.
                # A case with council_district but no council_rep and no
                # stored lat/lon has to go through the full geocode again --
                # there's no coordinate to hand lookup_all_districts directly
                # (see JOB 3.6 below for the cheap path when lat/lon exists).
                # These are the oldest rows: enriched back when
                # lookup_all_districts only had 3 layers and never asked for
                # Representative at all.
                needs_geo = [
                    v for v in cases.values()
                    if (v.get('address_formatted') or v.get('Address_1'))
                    and (v.get('geocode_status') == 'no_result'
                         or (v.get('geocode_status') == 'success'
                             and (not v.get('council_district')
                                  or not v.get('census_tract')
                                  or (not v.get('council_rep')
                                      and v.get('latitude') is None)))
                         or not v.get('geocode_status'))
                ]
                # Also probe unfindable cases older than 7 days
                for v in cases.values():
                    if v.get('geocode_status') == 'unfindable':
                        last = v.get('geocode_last_attempt', '')
                        if last:
                            try:
                                days = (datetime.now(CT)
                                        - datetime.fromisoformat(last)
                                        ).total_seconds() / 86400
                                if days >= 7:
                                    needs_geo.append(v)
                            except Exception:
                                pass
                if needs_geo:
                    elapsed = (datetime.now()
                               - cycle_start).total_seconds()
                    if elapsed < 1200:
                        logger.info(
                            f"\n── JOB 3.5: Re-geocoding"
                            f" {len(needs_geo)} cases ──")
                        await geocode_all_cases(needs_geo)
                        # State stays in memory; cycle-end block captures.
                        state['cases'] = cases
                        state['updated_at'] = now_iso
                        geo_ok = sum(
                            1 for c in needs_geo
                            if c.get('geocode_status') == 'success')
                        districts_ok = sum(
                            1 for c in needs_geo
                            if c.get('council_district'))
                        logger.info(
                            f"  re-geocode: {geo_ok}/{len(needs_geo)}"
                            f" geocoded, {districts_ok} w/districts")

                # ────────────────────────────────────────────────
                # JOB 3.6: Backfill council_rep (and the other district
                # rep/layer fields) for cases that already have coordinates.
                # JOB 3.5 only re-selects a case for the address geocoder if
                # council_district or census_tract is missing -- a case that
                # already has council_district but never got a
                # Representative back (a partial lookup_all_districts
                # failure, or a row enriched before that field existed) has
                # neither, so it was never revisited and stayed blank
                # forever. No address geocode needed here, just the 5 GIS
                # layer calls against the coordinate already on file.
                # ────────────────────────────────────────────────
                needs_rep = [
                    v for v in cases.values()
                    if v.get('latitude') is not None
                    and v.get('longitude') is not None
                    and not v.get('council_rep')
                ]
                # First run against the live backlog is ~48k cases. Left
                # uncapped that's a single unbroken multi-hour burst of 5
                # requests/case against a small county ArcGIS server -- the
                # kind of load that plausibly caused the partial failures
                # this job exists to fix. Capped and newest-first instead, so
                # every cycle makes bounded progress and the freshest cases
                # (the ones anyone is actually looking at) clear first.
                DISTRICT_BACKFILL_BATCH = 2000
                if len(needs_rep) > DISTRICT_BACKFILL_BATCH:
                    # file_date is MM/DD/YYYY -- sorting the raw string
                    # would put 01/2026 before 12/2020. parse_file_date
                    # (defined above, JOB 3) gives real chronological order;
                    # unparseable dates sort last, not first.
                    needs_rep.sort(
                        key=lambda v: parse_file_date(v.get('file_date', ''))
                                      or datetime.min,
                        reverse=True)
                    needs_rep = needs_rep[:DISTRICT_BACKFILL_BATCH]
                if needs_rep:
                    elapsed = (datetime.now() - cycle_start).total_seconds()
                    if elapsed < 1200:
                        logger.info(
                            f"\n── JOB 3.6: Backfilling council/rep for"
                            f" {len(needs_rep)} cases ──")
                        async with aiohttp.ClientSession() as session:
                            for case in needs_rep:
                                await lookup_districts_for_case(
                                    session, case)
                        state['cases'] = cases
                        state['updated_at'] = now_iso
                        rep_ok = sum(
                            1 for c in needs_rep if c.get('council_rep'))
                        logger.info(
                            f"  district backfill: {rep_ok}/{len(needs_rep)}"
                            f" got a rep")

                # ────────────────────────────────────────────────
                # JOB 4: Backfill (extend coverage backward)
                # ────────────────────────────────────────────────
                elapsed = (datetime.now() - cycle_start).total_seconds()
                remaining = max(0, 1500 - elapsed)
                bf_cursor = proc['eva'].get('backfill_cursor')
                bf_target = proc['eva'].get('backfill_target', '2020-01-01')

                # JOB 4 is retired. It walked date windows backwards exactly
                # as the sweep workers now do, but discarded the status of every
                # case it already held (`if dk not in cases`), so it only ever
                # inserted. Leaving both running would have two processes
                # searching the same date ranges against the same court system.
                # The sweep supersedes it: same walk, and it records status
                # changes instead of dropping them. Set CASELINK_ENABLE_JOB4=1
                # to bring it back if the sweep is ever taken out of service.
                job4_enabled = os.environ.get('CASELINK_ENABLE_JOB4') == '1'
                if job4_enabled and remaining > 300 and bf_cursor != 'DONE':
                    if bf_cursor is None:
                        bf_cursor = (now_ct - timedelta(days=3)).strftime(
                            '%Y-%m-%d')

                    cursor_dt = datetime.strptime(bf_cursor, '%Y-%m-%d')
                    target_dt = datetime.strptime(bf_target, '%Y-%m-%d')

                    if cursor_dt > target_dt:
                        month_key = cursor_dt.strftime('%Y-%m')
                        density = params.get('filing_density', {}).get(
                            month_key,
                            params.get('filing_density', {}).get('default', 40))
                        window = max(1, min(7, int(
                            params.get('target_grid_results', 700)
                            / max(1, density))))

                        bf_end = cursor_dt - timedelta(days=1)
                        bf_start = max(target_dt,
                                       bf_end - timedelta(days=window - 1))

                        logger.info(f"\n── JOB 4: Backfill "
                                    f"{bf_start.strftime('%m/%d/%Y')} - "
                                    f"{bf_end.strftime('%m/%d/%Y')} ──")
                        try:
                            if not await hard_reset(page):
                                errors += 1
                            else:
                                await find_search_frame(page)
                                grid_cases = (
                                    await search_detainer_cases_by_date(
                                        page, bf_start, bf_end))
                                bf_new = 0
                                for gc in grid_cases:
                                    dk = gc.get('Docket_Number', '')
                                    if dk and dk not in cases:
                                        cases[dk] = grid_case_to_entry(
                                            gc, now_iso)
                                        bf_new += 1

                                actual = len(grid_cases) / max(1, window)
                                params.setdefault(
                                    'filing_density', {})[month_key] = round(
                                        actual, 1)

                                if len(grid_cases) > 900:
                                    params['filing_density'][month_key] = (
                                        actual * 1.5)
                                    proc['history'].append({
                                        'ts': now_iso, 'op': 'REC',
                                        'adjustments': [
                                            f"grid overflow ({len(grid_cases)})"
                                            f" for {month_key}"],
                                    })

                                new_cursor = bf_start.strftime('%Y-%m-%d')
                                proc['eva']['backfill_cursor'] = new_cursor
                                new_count += bf_new
                                logger.info(f"  {len(grid_cases)} in grid, "
                                            f"{bf_new} new")

                                if bf_start <= target_dt:
                                    proc['eva']['backfill_cursor'] = 'DONE'
                                    logger.info("  ✓ Backfill complete!")

                        except Exception as e:
                            logger.warning(f"  Backfill error: {e}")
                            errors += 1

                # ────────────────────────────────────────────────
                # COMMIT: emit one EO block to archive.org
                # ────────────────────────────────────────────────
                state['cases'] = cases
                state['updated_at'] = now_iso

                total_pending = sum(
                    1 for c in cases.values()
                    if c.get('status') not in terminal)
                total_enriched = sum(
                    1 for c in cases.values() if c.get('enriched'))

                commit_msg = (f"+{new_count} new {changed_count} chg "
                              f"{checked_count} chk {enriched_count} enr "
                              f"{errors} err | {len(cases)} total")

                # EVA — cycle summary
                eva = proc['eva']
                eva['last_cycle'] = now_iso
                eva['total_cases'] = len(cases)
                eva['total_pending'] = total_pending
                eva['total_enriched'] = total_enriched
                eva['errors_this_cycle'] = errors
                eva['last_enriched_count'] = enriched_count

                # Geocode EVA snapshot
                geo_success = sum(1 for c in cases.values()
                                  if c.get('geocode_status') == 'success')
                geo_unfindable = sum(1 for c in cases.values()
                                     if c.get('geocode_status') == 'unfindable')
                geo_pending = sum(1 for c in cases.values()
                                  if c.get('geocode_status') == 'no_result')
                census_count = sum(1 for c in cases.values()
                                   if c.get('geocode_source') == 'census')
                nominatim_count = sum(1 for c in cases.values()
                                      if c.get('geocode_source') == 'nominatim')
                eva['geocode'] = {
                    'success': geo_success,
                    'unfindable': geo_unfindable,
                    'pending': geo_pending,
                    'census': census_count,
                    'nominatim': nominatim_count,
                }
                if errors == 0:
                    eva['consecutive_clean'] = eva.get(
                        'consecutive_clean', 0) + 1
                else:
                    eva['consecutive_clean'] = 0

                proc['history'].append({
                    'ts': now_iso, 'op': 'EVA',
                    'new': new_count, 'changed': changed_count,
                    'checked': checked_count, 'enriched': enriched_count,
                    'errors': errors,
                    'elapsed': round(
                        (datetime.now() - cycle_start).total_seconds()),
                })

                # REC — run tuning adjustments, may mutate proc['def']
                adj = run_rec(proc)
                if adj:
                    apply_tuning(params)
                    logger.info(f"  REC: {', '.join(adj)}")

                # Trim history
                if len(proc['history']) > 200:
                    proc['history'] = proc['history'][-200:]

                proc['updated_at'] = now_iso

                # EVA op — always emitted, even on a no-op cycle
                emit_event(cycle_events, now_iso, 'EVA',
                           data={
                               'new': new_count,
                               'changed': changed_count,
                               'checked': checked_count,
                               'enriched': enriched_count,
                               'errors': errors,
                               'total_cases': len(cases),
                               'elapsed': round(
                                   (datetime.now()
                                    - cycle_start).total_seconds()),
                           })
                if adj:
                    emit_event(cycle_events, now_iso, 'REC',
                               data={
                                   'adjustments': adj,
                                   'def_snapshot': copy.deepcopy(proc['def']),
                               })

                # Build the transformation list. transformations_from_diff
                # produces INS for new dockets, DEF/SEG/CON for changed ones,
                # and a batched NUL op for dockets checked-but-unchanged.
                # EVA and REC ride along as extra ops.
                transformations = eo_chain.transformations_from_diff(
                    prev_cases      = prev_cases_snapshot,
                    new_cases       = cases,
                    checked_dockets = cycle_checked_dockets,
                    extra_ops       = [ev for ev in cycle_events
                                       if ev.get('op') in ('EVA', 'REC')],
                )

                # ── Local hot tier + served snapshot (every cycle) ──
                # Per-cycle behaviour: accumulate this cycle's ops into the
                # local today.jsonl and refresh the snapshot n8n serves. The
                # archive.org chain gets exactly ONE block per day, sealed at
                # the America/Chicago rollover below.
                substantive = (new_count + changed_count + enriched_count) > 0

                today_ct = datetime.now(CT).strftime('%Y-%m-%d')
                open_day = head.data.get('open_day')
                rolled_over_ok = True   # cleanly sitting in today_ct's day?

                # Daily rollover: seal the just-completed prior day as ONE
                # block BEFORE this cycle's (new-day) ops are appended.
                if open_day is not None and today_ct != open_day:
                    prior_ops = [o for o in eo_chain._read_today_local()
                                 if o.get('op') != 'SIG']
                    if prior_ops:
                        eo_chain._encrypt_ops_inplace(prior_ops)  # ciphertext to archive
                        seal_jsonl, seal_header = eo_chain.build_block_jsonl(
                            block_type      = 'daily',
                            block_index     = head.data['next_index'],
                            prev_archive_id = head.data['head_id'],
                            prev_hash       = head.data['head_hash'],
                            transformations = prior_ops,
                            block_date      = open_day,
                            extra_header    = {
                                'def_params':      copy.deepcopy(proc['def']),
                                'eva_snapshot':    copy.deepcopy(eva),
                                'scraper_version': 'DER_5',
                                'commit_msg':      f'daily seal {open_day}',
                            },
                        )
                        seal_ok = False
                        try:
                            async with aiohttp.ClientSession() as up_session:
                                resp = await eo_chain.upload_block(
                                    up_session, seal_jsonl, seal_header)
                            archive_id = (resp.get('archive') or {}).get('identifier')
                            if archive_id:
                                head.advance(archive_id, seal_jsonl,
                                             block_date=open_day)
                                seal_ok = True
                                logger.info(
                                    f"  chain: daily block {open_day} -> "
                                    f"{archive_id} ({len(prior_ops)} ops)")
                            else:
                                logger.error("  chain: daily upload returned no id")
                        except Exception as e:
                            logger.error(
                                f"  chain: daily seal failed (kept in "
                                f"today.jsonl, retry next cycle): {e}")

                        if not seal_ok:
                            # Hold the prior day; do not advance into the new
                            # day. Next cycle retries with the same ops.
                            rolled_over_ok = False
                        else:
                            # 10-day aggregate fold (full encrypted state).
                            try:
                                async with aiohttp.ClientSession() as agg_session:
                                    recent_blocks = await eo_chain.search_chain(
                                        agg_session,
                                        subject=eo_chain.CHAIN_SUBJECT,
                                        rows=500)
                                    if eo_chain.should_aggregate(recent_blocks):
                                        logger.info("  chain: 10-day window, "
                                                    "building aggregate")
                                        enc_cases = copy.deepcopy(cases)
                                        encrypt_state({'cases': enc_cases})
                                        agg_jsonl, agg_header = (
                                            eo_chain.build_aggregate_block(
                                                block_index     = head.data['next_index'],
                                                prev_archive_id = head.data['head_id'],
                                                prev_hash       = head.data['head_hash'],
                                                resolved_state  = enc_cases,
                                                folded_block_ids = [
                                                    b['identifier']
                                                    for b in recent_blocks],
                                                operator_totals = {},
                                                def_params      = copy.deepcopy(
                                                    proc['def']),
                                            ))
                                        agg_resp = await eo_chain.upload_block(
                                            agg_session, agg_jsonl, agg_header)
                                        agg_id = (agg_resp.get('archive') or {}).get(
                                            'identifier')
                                        if agg_id:
                                            head.advance(
                                                agg_id, agg_jsonl,
                                                block_date=agg_header['date'],
                                                is_aggregate=True)
                                            logger.info(
                                                f"  chain: aggregate -> {agg_id}")
                            except Exception as e:
                                logger.error(f"  chain: aggregate fold failed: {e}")

                    if rolled_over_ok:
                        # Open a fresh day on disk.
                        eo_chain._write_today_local(
                            [eo_chain.day_open_marker(today_ct)])
                        head.data['open_day'] = today_ct
                        head.save()
                elif open_day is None:
                    # First cycle after deploy: open today's hot tier (keep any
                    # ops a prior restart already wrote).
                    if not eo_chain._read_today_local():
                        eo_chain._write_today_local(
                            [eo_chain.day_open_marker(today_ct)])
                    head.data['open_day'] = today_ct
                    head.save()

                # Append this cycle's ops to the (current-day) hot tier and
                # refresh the served snapshot. On a failed rollover we hold the
                # baseline so nothing is dropped — next cycle re-diffs.
                if rolled_over_ok:
                    if substantive and transformations:
                        eo_chain._append_today_local(transformations)
                    if substantive:
                        prev_cases_snapshot = copy.deepcopy(cases)
                eo_chain.write_snapshot(cases)

                # Summary
                elapsed_sec = (datetime.now() - cycle_start).total_seconds()
                logger.info(f"\n{'=' * 60}")
                logger.info(f"  CYCLE COMPLETE ({elapsed_sec:.0f}s)")
                logger.info(f"  Total: {len(cases)} "
                            f"({total_pending} pending, "
                            f"{total_enriched} enriched)")
                logger.info(f"  This cycle: +{new_count} new, "
                            f"{changed_count} chg, {enriched_count} enr, "
                            f"{errors} err")
                logger.info(f"  Backfill: "
                            f"{proc['eva'].get('backfill_cursor', 'n/a')}")
                logger.info(f"{'=' * 60}")

                if mode == 'once':
                    break

                # Smart sleep: only wait when caught up
                unenriched = sum(
                    1 for c in cases.values()
                    if not c.get('enriched') or c.get('needs_enrichment'))
                backfill_done = proc['eva'].get('backfill_cursor') == 'DONE'

                if unenriched > 0 or not backfill_done:
                    logger.info(f"\n  Work remaining ({unenriched} unenriched"
                                f"{', backfill active' if not backfill_done else ''}"
                                f") — next cycle in 10s")
                    await asyncio.sleep(10)
                else:
                    sleep_sec = max(0, DAEMON_INTERVAL - elapsed_sec)
                    next_time = datetime.now() + timedelta(seconds=sleep_sec)
                    logger.info(f"\nCaught up. Next cycle at "
                                f"{next_time.strftime('%H:%M:%S')}")
                    await asyncio.sleep(sleep_sec)

        except KeyboardInterrupt:
            logger.info("\nStopped by user")
        finally:
            await browser.close()
            logger.info(f"Final state: {len(cases)} cases")


CREDS_FILE = Path.home() / '.caselink_scraper'

def load_saved_creds():
    if CREDS_FILE.exists():
        try:
            data = json.loads(CREDS_FILE.read_text())
            return data
        except Exception:
            pass
    return None

def save_creds(username, password, enc_password):
    CREDS_FILE.write_text(json.dumps({
        'username': username,
        'password': password,
        'encryption_password': enc_password,
    }))
    CREDS_FILE.chmod(0o600)
    logger.info(f"  Credentials saved to {CREDS_FILE}")


def load_credentials(args):
    """Load credentials from env vars → creds file → interactive prompt.
    Env vars (CASELINK_USER, CASELINK_PASS, CASELINK_ENC_KEY) take priority
    so the scraper can run unattended under systemd."""
    env_user = os.environ.get('CASELINK_USER', '').strip()
    env_pass = os.environ.get('CASELINK_PASS', '').strip()
    env_enc  = os.environ.get('CASELINK_ENC_KEY', '').strip()

    if env_user and env_pass and env_enc:
        logger.info("  Credentials loaded from environment")
        return env_user, env_pass, env_enc

    saved = load_saved_creds()
    if saved and '--new-creds' not in args:
        logger.info(f"  Credentials loaded from {CREDS_FILE}")
        return (saved['username'], saved['password'],
                saved.get('encryption_password', ''))

    # Interactive fallback (won't work under systemd)
    if not sys.stdin.isatty():
        logger.error("  No credentials found and stdin is not a terminal.\n"
                      "  Set CASELINK_USER, CASELINK_PASS, CASELINK_ENC_KEY "
                      "in /mnt/intelechia/caselink/.env\n"
                      "  or run interactively once to save a creds file.")
        sys.exit(1)

    user = input("  CaseLink username: ").strip()
    pw   = getpass.getpass("  CaseLink password: ")
    enc  = getpass.getpass("  Encryption password: ")
    save_creds(user, pw, enc)
    return user, pw, enc


def main():
    global USERNAME, PASSWORD, ENCRYPTION_PASSWORD, ALIAS

    args = sys.argv[1:]
    watch = '--watch' in args

    logger.info(f"\n{'=' * 60}")
    logger.info("  Nashville CaseLink Eviction Scraper")
    logger.info(f"{'=' * 60}\n")

    USERNAME, PASSWORD, ENCRYPTION_PASSWORD = load_credentials(args)

    # Address this account by alias from here on, and scrub any account name
    # that still reaches a log record (exception text, page dumps): this log is
    # republished verbatim on a public SSE endpoint.
    try:
        import caselink_accounts
        _accts = caselink_accounts.load_accounts()
        caselink_accounts.install_redaction(_accts)
        ALIAS = caselink_accounts.alias_for(USERNAME, _accts)
        logger.info(f"  Account alias: {ALIAS}")
    except Exception as _e:
        logger.warning(f"  account alias/redaction unavailable: "
                       f"{type(_e).__name__}")

    if '--rekey' in args:
        old_key = new_key = None
        for i, arg in enumerate(args):
            if arg == '--old-key' and i + 1 < len(args):
                old_key = args[i + 1]
            elif arg == '--new-key' and i + 1 < len(args):
                new_key = args[i + 1]
        if not old_key:
            old_key = getpass.getpass("  Old encryption key: ")
        if not new_key:
            new_key = getpass.getpass("  New encryption key: ")
        ENCRYPTION_PASSWORD = old_key
        asyncio.run(run(mode='rekey', rekey_old=old_key, rekey_new=new_key))
        save_creds(USERNAME, PASSWORD, new_key)
        return

    mode = 'once' if '--once' in args else 'daemon'

    backfill_to = None
    for i, arg in enumerate(args):
        if arg == '--backfill-to' and i + 1 < len(args):
            backfill_to = args[i + 1]

    asyncio.run(run(mode=mode, watch=watch, backfill_to=backfill_to))


if __name__ == '__main__':
    main()
