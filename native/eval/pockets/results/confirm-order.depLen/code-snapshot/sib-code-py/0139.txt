ALERTS_SCHEMA = """
CREATE TABLE IF NOT EXISTS alerts (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  slug TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  description TEXT,
  kind TEXT NOT NULL,
  terms_json TEXT,
  matched_matter_ids_json TEXT,
  corpus_size INTEGER,
  created_at TEXT,
  active INTEGER DEFAULT 1
);
"""

# Topic tags (custom alerts) added after the alerts table's first deploy —
# every existing row predates ownership entirely, which is exactly why it
# defaults an old row to private and unowned rather than public: a tag
# nobody explicitly marked shared never becomes visible to anyone else
# just because this migration ran.
ALERTS_ADDED_COLUMNS = [
    ("owner_account_id", "TEXT"),
    ("visibility", "TEXT NOT NULL DEFAULT 'private'"),
    ("config_json", "TEXT"),
]


def _ensure_alerts_table(con: sqlite3.Connection) -> None:
    con.executescript(ALERTS_SCHEMA)
    have = {r[1] for r in con.execute("PRAGMA table_info(alerts)").fetchall()}
    for name, decl in ALERTS_ADDED_COLUMNS:
        if name not in have:
            con.execute(f"ALTER TABLE alerts ADD COLUMN {name} {decl}")
    con.commit()


# Two separate anonymous-voting signals. matter_tag_votes asks "does THIS
# tag really apply to THIS matter" — raw signal for sorting real matches
# from false positives on a pattern-matched flag. alert_quality_votes asks
# a different question entirely: "is this whole tag's matching any good" —
# a vote on someone's tag as a whole, not on any one match. voter_id is a
# random per-browser id (see LS_VOTER_KEY client-side), unrelated to the
# device key the zero-knowledge account system uses, so voting never links
# back to a person's notes/account activity.
VOTES_SCHEMA = """
CREATE TABLE IF NOT EXISTS matter_tag_votes (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  alert_slug TEXT NOT NULL,
  matter_id INTEGER NOT NULL,
  voter_id TEXT NOT NULL,
  vote INTEGER NOT NULL,
  created_at TEXT,
  UNIQUE(alert_slug, matter_id, voter_id)
);
CREATE INDEX IF NOT EXISTS idx_matter_tag_votes_lookup ON matter_tag_votes(alert_slug, matter_id);

CREATE TABLE IF NOT EXISTS alert_quality_votes (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  alert_slug TEXT NOT NULL,
  voter_id TEXT NOT NULL,
  vote INTEGER NOT NULL,
  created_at TEXT,
  UNIQUE(alert_slug, voter_id)
);
"""


def _ensure_votes_tables(con: sqlite3.Connection) -> None:
    con.executescript(VOTES_SCHEMA)
    con.commit()


def _tag_vote_counts(con: sqlite3.Connection, alert_slug: str, matter_id: int) -> dict:
    rows = con.execute(
        "SELECT vote, COUNT(*) c FROM matter_tag_votes WHERE alert_slug=? AND matter_id=? GROUP BY vote",
        (alert_slug, matter_id),
    ).fetchall()
    return {"up": sum(r["c"] for r in rows if r["vote"] == 1),
            "down": sum(r["c"] for r in rows if r["vote"] == -1)}


def _quality_vote_counts(con: sqlite3.Connection, alert_slug: str) -> dict:
    rows = con.execute(
        "SELECT vote, COUNT(*) c FROM alert_quality_votes WHERE alert_slug=? GROUP BY vote",
        (alert_slug,),
    ).fetchall()
    return {"up": sum(r["c"] for r in rows if r["vote"] == 1),
            "down": sum(r["c"] for r in rows if r["vote"] == -1)}


def _slugify(name: str) -> str:
    s = re.sub(r"[^a-z0-9]+", "-", name.strip().lower()).strip("-")
    return s or "alert"


def matter_url(matter_id: int, matter_guid: str | None) -> str:
    # The WebAPI's MatterId/MatterGuid are NOT the public site's ID/GUID —
    # those are a completely separate ID space on Legistar's front end
    # (confirmed: same matter, two unrelated ID/GUID pairs). gateway.aspx
    # is Legistar's own permalink resolver: it takes the WebAPI's MatterId
    # and 302s to the real, correct LegislationDetail.aspx URL.
    return f"https://nashville.legistar.com/gateway.aspx?m=l&id={matter_id}"


EVIDENCE_LABELS = {
    "core": "technology",
    "vendor": "vendor",
    "guard": "statutory qualifier phrase",
    "sensor": "data/sensor term",
    "declared_use": "Metro's own declared use",
}


def humanize_evidence(raw: list[str]) -> list[str]:
    """One chip per concept. The lexicon carries several spellings of the
    same thing ("facial recognition"/"facial", "uas"/"uav"/"unmanned
    aircraft", "body camera"/"body worn"), and listing them all reads as
    more evidence than there is. Terms that trigger the same clause of
    §13.08.080 are the same finding, so they collapse to one."""
    out, seen_concepts, seen_labels = [], set(), set()
    for e in raw:
        if e == "CERTIFICATION:self-declared-yes":
            continue  # surfaced separately as its own badge
        if ":" in e:
            kind, term = e.split(":", 1)
            term = term.split(" (w=")[0]
            kind = kind.lower()
            label = EVIDENCE_LABELS.get(kind, kind)
            concept = clause_for(term) if kind == "core" else ""
            if concept:
                if concept in seen_concepts:
                    continue
                seen_concepts.add(concept)
            # "capable of (oversight language)" reads like broken English —
            # quoting the phrase makes clear it's literal document wording
            # the classifier watches for, not a claim about the matter.
            shown = f'"{term.replace("_", " ")}"' if kind == "guard" else term.replace("_", " ")
            text = f"{shown} ({label})"
        else:
            text = e.replace("_", " ")
        if text.lower() in seen_labels:
            continue
        seen_labels.add(text.lower())
        out.append(text)
    return out


def self_certified(raw: list[str]) -> bool:
    return "CERTIFICATION:self-declared-yes" in raw


# The production rule (scorer.py's fuse()) is one point in a much larger
# space: "flagged" = at least one CORE technology term, OR a named VENDOR,
# OR a self-certified "yes" on the procurement form. That's a real,
# specific choice among several defensible ones — requiring a vendor name
# is a different, stricter bet than counting bare data/sensor language
# (which the production rule deliberately excludes, after "surveillance"
# and "lpr" alone produced false positives). Nothing here reruns OCR or
# the reader; it only re-aggregates the SAME stored per-term hits under a
# different rule, which is honest because the underlying evidence really
# does support more than one reading.
METHODOLOGY_KINDS = {
    "core": "Names a specific surveillance technology (camera, ALPR, drone, facial recognition, …)",
    "vendor": "Names a known surveillance vendor (Motorola, Flock, Axon, …)",
    "cert": "The filer self-certified \"yes\" to the procurement form's surveillance question",
    "declared_use": "Metro's own answer to the intake form's \"what is this cooperative being used "
                     "to purchase\" question names a specific surveillance technology — treated as "
                     "certain, the same as a \"yes\" self-certification",
    "sensor": "Uses only generic data/sensor language (collect, record, monitor, …) — excluded by default: too broad alone",
}
# The actual lexicon behind each kind — not the shortened "camera, ALPR, …"
# examples in the description above, the real, complete list src/classify.py
# tests against. "cert" and "declared_use" have no term list of their own:
# each is a specific document ANSWER (a yes/no checkbox; a declared-use
# sentence checked against the same core list), not an independent word
# search.
METHODOLOGY_TERMS = {
    "core": sorted(t.replace("_", " ") for t in CORE_TERMS),
    "vendor": sorted(t.replace("_", " ") for t in VENDOR_TERMS),
    "sensor": sorted(t.replace("_", " ") for t in SENSOR_TERMS),
    "cert": [],
    "declared_use": [],
}
DEFAULT_REQUIRE = "core,vendor,cert,declared_use"


def _evidence_kinds(raw_evidence: list[str]) -> set[str]:
    kinds = set()
    for e in raw_evidence:
        if e == "CERTIFICATION:self-declared-yes":
            kinds.add("cert")
        elif ":" in e:
            kinds.add(e.split(":", 1)[0].lower())
    return kinds


def _anchored(raw_evidence: list[str], require: set[str]) -> bool:
    """Would this matter be flagged under THIS methodology? ANY of the
    required kinds present is enough — the same OR the production rule
    uses, just over a caller-chosen set of kinds instead of the fixed one."""
    return bool(_evidence_kinds(raw_evidence) & require)


FEED_SORTS = {
    "score": "s.score DESC",
    "date_desc": "COALESCE(m.enactment_date, m.passed_date, m.agenda_date, m.intro_date) DESC",
    "date_asc": "COALESCE(m.enactment_date, m.passed_date, m.agenda_date, m.intro_date) ASC",
}


@app.get("/api/feed")
def api_feed(limit: int = Query(100, le=500), sort: str = Query("date_desc"),
              q: str = Query(""), require: str = Query(DEFAULT_REQUIRE),
              min_score: float = Query(0)):
    con = db()
    # Newest first by default. Score-first buried this week's filings under a
    # four-year backlog that scores higher only because those contracts are
    # longer; someone opening the feed wants to know what just happened.
    order = FEED_SORTS.get(sort, FEED_SORTS["date_desc"])
    q = q.strip()
    require_set = {k.strip().lower() for k in require.split(",") if k.strip()}
    # "error" rows are unreadable documents, not a methodology's business;
    # every other scanned matter — including ones the DEFAULT rule left as
    # "background" — is fair game once the caller picks a looser rule.
    where, params = ["s.label != 'error'"], []
    if q:
        where.append("(s.file LIKE ? OR s.title LIKE ? OR s.headline LIKE ? OR s.evidence_json LIKE ?)")
        params += [f"%{q}%"] * 4
    rows = con.execute(
        f"""SELECT s.matter_id, s.file, s.title, s.label, s.score, s.evidence_json,
                  s.archives_json, s.sidecar_archive_json, s.scanned_at,
                  s.headline, s.slots_json,
                  m.matter_guid, m.status_name, m.intro_date, m.agenda_date,
                  m.passed_date, m.enactment_date
           FROM scan_results s
           LEFT JOIN matters m ON m.matter_id = s.matter_id
           WHERE {' AND '.join(where)}
           ORDER BY {order}""",
        params,
    ).fetchall()
    con.close()
    out = []
    for r in rows:
        raw_evidence = json.loads(r["evidence_json"] or "[]")
        if not _anchored(raw_evidence, require_set) or (r["score"] or 0) < min_score:
            continue
        if len(out) >= limit:
            break
        # A direct source link is real infrastructure (Legistar's own
        # hosting, populated straight from the attachment at scan time);
        # the Wayback capture is a best-effort bonus that often fails
        # silently. Requiring archiveUrl here used to throw away every
        # direct link too the moment archiving didn't succeed — which was
        # most of the time — leaving no way to reach the source at all.
        archives = [a for a in json.loads(r["archives_json"] or "[]") if a.get("sourceUrl") or a.get("archiveUrl")]
        sidecar = json.loads(r["sidecar_archive_json"]) if r["sidecar_archive_json"] else None
        status = describe_status(r["status_name"], r["intro_date"], r["agenda_date"],
                                  r["passed_date"], r["enactment_date"])
        out.append({
            "matterId": r["matter_id"],
            "file": r["file"],
            "title": r["title"],
            "headline": r["headline"],
            "slots": json.loads(r["slots_json"] or "{}"),
            "score": r["score"],
            "evidence": humanize_evidence(raw_evidence),
            "selfCertified": self_certified(raw_evidence),
            "status": status["label"],
            "statusDate": status["date"],
            "url": matter_url(r["matter_id"], r["matter_guid"]),
            "archives": [{"name": a["name"], "url": a.get("archiveUrl") or a.get("sourceUrl"),
                          "isArchive": bool(a.get("archiveUrl"))} for a in archives],
            "sidecarUrl": sidecar["itemUrl"] if sidecar else None,
        })
    return JSONResponse(out)


SORTS = {
    "relevance": "(s.score IS NOT NULL) DESC, s.score DESC, m.agenda_date DESC",
    "score": "s.score DESC",
    "date_desc": "m.agenda_date DESC",
    "date_asc": "m.agenda_date ASC",
    "title": "m.title ASC",
}


@app.get("/api/search")
def api_search(
    q: str = Query("", min_length=0),
    department: str = Query(""),
    status: str = Query(""),
    sponsor: str = Query(""),
    body: str = Query(""),
    dept: str = Query(""),
    type_name: str = Query(""),
    flagged: str = Query(""),
    sort: str = Query("relevance"),
    limit: int = Query(50, le=200),
):
    """`body` is the council body that handled it (a committee, or the
    Council itself) — `dept` is the Metro DEPARTMENT the matter names,
    which lives in the title, not in body_name. They are different
    questions and the general browse needs both. `department` stays as an
    alias for `body` so older links keep working."""
    q = q.strip()
    body = body or department
    if not any([q, body, status, sponsor, dept, type_name, flagged]):
        return JSONResponse([])
    con = db()
    clauses, params = [], []
    if sponsor:
        clauses.append("m.matter_id IN (SELECT matter_id FROM sponsorships WHERE name = ?)")
        params.append(sponsor)
    if q:
        clauses.append("(m.title LIKE ? OR m.file LIKE ?)")
        params += [f"%{q}%", f"%{q}%"]
    if body:
        clauses.append("m.body_name = ?")
        params.append(body)
    if dept:
        forms = departments_mod.surface_forms(dept) or [dept]
        clauses.append("(" + " OR ".join(["m.title LIKE ?"] * len(forms)) + ")")
        params += [f"%{f}%" for f in forms]
    if type_name:
        clauses.append("m.type_name = ?")
        params.append(type_name)
    if flagged == "1":
        clauses.append("s.label = 'surveillance'")
    if status:
        clauses.append("m.status_name = ?")
        params.append(status)
    order = SORTS.get(sort, SORTS["relevance"])
    rows = con.execute(
        f"""SELECT m.matter_id, m.matter_guid, m.file, m.title, m.status_name,
                   m.type_name, m.body_name,
                   m.intro_date, m.agenda_date, m.passed_date, m.enactment_date,
                   s.label, s.score,
                   COALESCE(s.headline, m.headline) AS headline, m.clean_title
            FROM matters m
            LEFT JOIN scan_results s ON s.matter_id = m.matter_id
            WHERE {' AND '.join(clauses) if clauses else '1=1'}
            ORDER BY {order}
            LIMIT ?""",
        (*params, limit),
    ).fetchall()
    con.close()
    out = []
    for r in rows:
        st = describe_status(r["status_name"], r["intro_date"], r["agenda_date"],
                              r["passed_date"], r["enactment_date"])
        out.append({
            "matterId": r["matter_id"],
            "file": r["file"],
            "title": r["title"],
            "headline": r["headline"],
            "cleanTitle": r["clean_title"],
            "type": r["type_name"],
            "body": r["body_name"],
            "status": st["label"],
            "statusDate": st["date"],
            "agendaDate": r["agenda_date"],
            "flagged": r["label"] == "surveillance",
            "score": r["score"],
            "url": matter_url(r["matter_id"], r["matter_guid"]),
        })
    return JSONResponse(out)


@app.get("/api/filters")
def api_filters():
    con = db()
    departments = [r["body_name"] for r in con.execute(
        "SELECT DISTINCT body_name FROM matters WHERE body_name IS NOT NULL AND body_name != '' ORDER BY body_name"
    ).fetchall()]
    statuses = [r["status_name"] for r in con.execute(
        "SELECT DISTINCT status_name FROM matters WHERE status_name IS NOT NULL AND status_name != '' ORDER BY status_name"
    ).fetchall()]
    try:
        sponsors = [r["name"] for r in con.execute(
            "SELECT name, COUNT(*) c FROM sponsorships GROUP BY name ORDER BY name"
        ).fetchall()]
    except sqlite3.OperationalError:
        sponsors = []   # participation not fetched yet
    con.close()
    return JSONResponse({"departments": departments, "statuses": statuses,
                         "sponsors": sponsors})


_DEPT_INDEX: dict | None = None


def _department_index(con: sqlite3.Connection) -> dict:
    """How many matters name each canonical Metro department in their
    title. Computed in ONE pass over all ~8,400 titles rather than 59
    separate LIKE scans, then cached for the process — the underlying set
    only changes when a scan reseeds, and a stale count on a browse tile
    is cheaper than a full rescan on every page load."""
    global _DEPT_INDEX
    if _DEPT_INDEX is not None:
        return _DEPT_INDEX
    counts: dict[str, int] = {}
    for r in con.execute("SELECT title FROM matters WHERE title IS NOT NULL").fetchall():
        for d in departments_mod.find_in(r["title"]):
            counts[d] = counts.get(d, 0) + 1
    _DEPT_INDEX = counts
    return counts


# Metro parks an "Indefinitely Deferred" matter by setting its agenda date
# to a far-future placeholder — 60 matters currently sit on 2027-08-17.
# Sorting a "recent activity" feed by raw agenda date therefore puts a
# three-year-old deferred bill at the very top, above this week's real
# business, which is exactly backwards. "Recent" means dates that have
# actually arrived; "upcoming" means a real scheduled date, which a
# deferral placeholder is not.
DEFERRED_STATUS = "Indefinitely Deferred"


@app.get("/api/recent")
def api_recent(limit: int = Query(40, le=200), type_name: str = Query("")):
    """The newest matters of ANY kind that have actually happened — the
    general feed, not the surveillance one. Surveillance is a topic tag on
    top of this, not the thing this site is."""
    con = db()
    today = datetime.date.today().isoformat()
    where = ["COALESCE(m.enactment_date, m.passed_date, m.agenda_date, m.intro_date) IS NOT NULL",
             "date(COALESCE(m.enactment_date, m.passed_date, m.agenda_date, m.intro_date)) <= date(?)"]
    params: list = [today]
    if type_name:
        where.append("m.type_name = ?")
        params.append(type_name)
    rows = con.execute(
        f"""SELECT m.matter_id, m.matter_guid, m.file, m.title, m.type_name, m.body_name,
                   m.status_name, m.intro_date, m.agenda_date, m.passed_date, m.enactment_date,
                   s.label, s.score,
                   COALESCE(s.headline, m.headline) AS headline, m.clean_title
            FROM matters m LEFT JOIN scan_results s ON s.matter_id = m.matter_id
            WHERE {' AND '.join(where)}
            ORDER BY COALESCE(m.enactment_date, m.passed_date, m.agenda_date, m.intro_date) DESC
            LIMIT ?""",
        (*params, limit),
    ).fetchall()
    entities = {b["matterId"]: b for b in _matter_brief(con, [r["matter_id"] for r in rows])}
    con.close()
    out = []
    for r in rows:
        st = describe_status(r["status_name"], r["intro_date"], r["agenda_date"],
                              r["passed_date"], r["enactment_date"])
        b = entities.get(r["matter_id"], {})
        out.append({
            "matterId": r["matter_id"], "file": r["file"], "title": r["title"],
            "headline": r["headline"], "cleanTitle": r["clean_title"],
            "type": r["type_name"], "body": r["body_name"],
            "status": st["label"], "statusDate": st["date"],
            "flagged": r["label"] == "surveillance", "score": r["score"],
            "url": matter_url(r["matter_id"], r["matter_guid"]),
            "sponsors": b.get("sponsors") or [], "entities": b.get("entities") or [],
        })
    return JSONResponse(out)


@app.get("/api/upcoming")
def api_upcoming(limit: int = Query(25, le=100)):
    """What's actually scheduled ahead — the other half of tracking. The
    deferral placeholder date is excluded by status, so a bill parked in
    2027 never shows up here pretending to be next month's business."""
    con = db()
    today = datetime.date.today().isoformat()
    rows = con.execute(
        """SELECT m.matter_id, m.matter_guid, m.file, m.title, m.type_name, m.body_name,

           FROM matters m LEFT JOIN scan_results s ON s.matter_id = m.matter_id
           WHERE m.agenda_date IS NOT NULL
             AND date(m.agenda_date) > date(?)
             AND COALESCE(m.status_name, '') != ?

           LIMIT ?""",
        (today, DEFERRED_STATUS, limit),
    ).fetchall()
    entities = {b["matterId"]: b for b in _matter_brief(con, [r["matter_id"] for r in rows])}
    con.close()
    out = []
    for r in rows:
        b = entities.get(r["matter_id"], {})
        out.append({
            "matterId": r["matter_id"], "file": r["file"], "title": r["title"],
            "headline": r["headline"], "cleanTitle": r["clean_title"],
            "type": r["type_name"], "body": r["body_name"],
            "status": r["status_name"], "statusDate": r["agenda_date"],
            "flagged": r["label"] == "surveillance", "score": r["score"],
            "url": matter_url(r["matter_id"], r["matter_guid"]),
            "sponsors": b.get("sponsors") or [], "entities": b.get("entities") or [],

@app.get("/api/browse")
def api_browse():
    """The dimensions anything in Metro Nashville can be sliced by —
    what kind of thing it is, which body handled it, which department it
    names, who sponsored it. Each comes back with a real count, so a tile
    that would open onto nothing never gets shown as if it would."""
    con = db()
    types = [{"name": r["type_name"], "count": r["c"]} for r in con.execute(
        "SELECT type_name, COUNT(*) c FROM matters WHERE type_name IS NOT NULL AND type_name != ''"
        " GROUP BY type_name ORDER BY c DESC").fetchall()]
    bodies = [{"name": r["body_name"], "count": r["c"]} for r in con.execute(
        "SELECT body_name, COUNT(*) c FROM matters WHERE body_name IS NOT NULL AND body_name != ''"
        " GROUP BY body_name ORDER BY c DESC").fetchall()]
    statuses = [{"name": r["status_name"], "count": r["c"]} for r in con.execute(
        "SELECT status_name, COUNT(*) c FROM matters WHERE status_name IS NOT NULL AND status_name != ''"
        " GROUP BY status_name ORDER BY c DESC").fetchall()]
    depts = sorted(
        ({"name": k, "count": v} for k, v in _department_index(con).items()),
        key=lambda d: -d["count"])
    try:
        sponsors = [{"name": r["name"], "count": r["c"]} for r in con.execute(
            "SELECT name, COUNT(*) c FROM sponsorships GROUP BY name ORDER BY c DESC").fetchall()]
    except sqlite3.OperationalError:
        sponsors = []
    total_flagged = con.execute(
        "SELECT COUNT(*) c FROM scan_results WHERE label='surveillance'"
    ).fetchone()["c"]
    con.close()
    return JSONResponse({"types": types, "bodies": bodies, "statuses": statuses,
                         "departments": depts, "sponsors": sponsors,
                         "totalFlagged": total_flagged})


@app.get("/api/methodology")
def api_methodology():
    """Plain-language description of every kind currently readable off of a
    scan, plus how many of the 234 already-read matters each one alone
    would flag — so "adjust the methodology" has real numbers behind it,
    not just checkboxes that feel like they do something."""
    con = db()
    rows = con.execute(
        "SELECT evidence_json, score FROM scan_results WHERE label != 'error'"
    ).fetchall()
    con.close()
    counts = {k: 0 for k in METHODOLOGY_KINDS}
    for r in rows:
        kinds = _evidence_kinds(json.loads(r["evidence_json"] or "[]"))
        for k in kinds & set(METHODOLOGY_KINDS):
            counts[k] += 1
    return JSONResponse({
        "total": len(rows),
        "defaultRequire": DEFAULT_REQUIRE.split(","),
        "kinds": [{"key": k, "description": d, "matterCount": counts[k],
                  "terms": METHODOLOGY_TERMS.get(k, [])}
                  for k, d in METHODOLOGY_KINDS.items()],
        "note": ("A matter is \"flagged\" when it has at least one hit in ANY "
                 "of the required kinds — the same OR rule production uses, "
                 "just over whichever kinds you choose. Nothing is re-read; "
                 "this re-aggregates the same stored per-term hits."),
    })


_BODY_GATE_CACHE: dict = {}


def _noninformative_bodies(con: sqlite3.Connection) -> set[str]:
    """Which deciding-body names are too common across the WHOLE corpus to
    tell you anything as an "entity involved" in one specific matter —
    measured, not asserted: Tukey's fence (1.5x IQR above the 75th
    percentile) over the real distribution of how many matters each body
    decides. Checked live: "Metropolitan Council" decides 6,383 of ~8,440
    matters (76%) — naming it as an "entity" on a matter page is exactly
    as uninformative as a word that matches half the corpus; the fence
    catches it (and "Planning and Zoning Committee," "Rules, Confirmations,
    and Public Elections Committee") without anyone hand-picking those
    three names. Cached — this is a corpus-wide property, recomputed only
    when the matter count actually changes."""
    total = con.execute("SELECT COUNT(*) c FROM matters WHERE body_name IS NOT NULL").fetchone()["c"]
    if _BODY_GATE_CACHE.get("total") == total:
        return _BODY_GATE_CACHE["excluded"]
    counts = sorted(r["c"] for r in con.execute(
        "SELECT COUNT(*) c FROM matters WHERE body_name IS NOT NULL GROUP BY body_name").fetchall())
    if len(counts) < 4:
        excluded: set[str] = set()
    else:
        q1, q3 = counts[len(counts) // 4], counts[3 * len(counts) // 4]
        fence = q3 + 1.5 * (q3 - q1)
        excluded = {r["body_name"] for r in con.execute(
            "SELECT body_name, COUNT(*) c FROM matters WHERE body_name IS NOT NULL "
            "GROUP BY body_name HAVING c > ?", (fence,)).fetchall()}
    _BODY_GATE_CACHE.clear()
    _BODY_GATE_CACHE.update(total=total, excluded=excluded)
    return excluded


def _matter_brief(con: sqlite3.Connection, matter_ids: list[int]) -> list[dict]:
    """The shared row shape every tag's matched-matters list renders from —
    Surveillance included, now that it's a tag like any other rather than
    a page with its own bespoke fields. A matter that's been deep-read
    (by any methodology, not just whichever tag you're looking at it
    through) carries its score/evidence/self-cert/archives here too, so
    the SAME template shows that richness wherever it applies instead of
    gating it behind one specific tag's UI."""
    if not matter_ids:
        return []
    placeholders = ",".join("?" for _ in matter_ids)
    rows = con.execute(
        f"""SELECT m.matter_id, m.matter_guid, m.file, m.title, m.body_name, m.status_name,
                   m.type_name, m.intro_date, m.agenda_date, m.passed_date, m.enactment_date,
                   COALESCE(s.headline, m.headline) AS headline, m.clean_title,
                   s.score, s.label, s.evidence_json, s.archives_json
            FROM matters m LEFT JOIN scan_results s ON s.matter_id = m.matter_id
            WHERE m.matter_id IN ({placeholders})""",
        matter_ids,
    ).fetchall()
    sponsors_by_matter: dict[int, list[str]] = {}
    try:
        for r in con.execute(
            f"SELECT matter_id, name FROM sponsorships WHERE matter_id IN ({placeholders}) "
            "ORDER BY sequence", matter_ids
        ).fetchall():
            sponsors_by_matter.setdefault(r["matter_id"], []).append(r["name"])
    except sqlite3.OperationalError:
        pass  # table not populated on a DB seeded before fetch_participation.py ran
    noninformative_bodies = _noninformative_bodies(con)
    out = []
    for r in rows:
        status = describe_status(r["status_name"], r["intro_date"], r["agenda_date"],
                                  r["passed_date"], r["enactment_date"])
        sponsors = sponsors_by_matter.get(r["matter_id"], [])
        # DMD-gated: a sponsor's own name is always real signal (who
        # specifically put their name on THIS matter is never "too
        # common" the way an overused word is), but the deciding body only
        # counts as an "entity involved" when it clears the measured fence
        # above — "Metropolitan Council" names nothing, "Budget and
        # Finance Committee" does.
        entities = list(sponsors)
        if r["body_name"] and r["body_name"] not in noninformative_bodies:
            entities.append(r["body_name"])
        entry = {
            "matterId": r["matter_id"], "file": r["file"], "title": r["title"],
            "headline": r["headline"], "cleanTitle": r["clean_title"],
            "department": r["body_name"], "body": r["body_name"], "type": r["type_name"],
            "status": status["label"], "statusDate": status["date"],
            "url": matter_url(r["matter_id"], r["matter_guid"]),
            "flagged": r["label"] == "surveillance",
            "sponsors": sponsors, "entities": entities,
        }
        if r["score"] is not None:
            raw_evidence = json.loads(r["evidence_json"] or "[]")
            entry.update({
                "score": r["score"],
                "evidence": humanize_evidence(raw_evidence),
                "selfCertified": self_certified(raw_evidence),
                "archives": [{"name": a["name"], "url": a.get("archiveUrl") or a.get("sourceUrl"),
                             "isArchive": bool(a.get("archiveUrl"))}
                             for a in json.loads(r["archives_json"] or "[]")
                             if a.get("sourceUrl") or a.get("archiveUrl")],
            })
        out.append(entry)
    return out


def _matters_for_departments(con: sqlite3.Connection, dept_names: list[str]) -> list[int]:
    """Every matter whose TITLE names one of these Metro departments, under
    any surface form (alias or canonical). `matters.body_name` is a
    different field entirely — the council BODY that decided the matter
    (a committee, or "Metropolitan Council") — never the department it's
    for, so it can't answer this question at all."""
    forms: list[str] = []
    for d in dept_names:
        forms.extend(departments_mod.surface_forms(d))
    if not forms:
        return []
    clause = " OR ".join(["title LIKE ?"] * len(forms))
    params = [f"%{f}%" for f in forms]
    rows = con.execute(f"SELECT matter_id FROM matters WHERE {clause}", params).fetchall()
    return [r["matter_id"] for r in rows]


def _matters_for_phrases(con: sqlite3.Connection, phrases: list[str]) -> dict[str, list[int]]:
    """Every matter whose TITLE contains a given phrase, checked against
    the FULL 8,000+ matter corpus — unlike alerts_engine.corpus_signal,
    which only covers the few hundred matters deep-read for surveillance
    and would silently under-report a general topic like "safety" or
    "police" as almost nonexistent just because deep-reading was scoped to
    something else entirely. Word-bounded (via the same _bounded() regex
    title_null_signal uses below), on purpose: what gets counted here has
    to be the exact same thing the null test measured, or "distinguishing"
    would be answering a question this function doesn't actually ask."""
    rows = con.execute("SELECT matter_id, title FROM matters WHERE title IS NOT NULL").fetchall()
    out: dict[str, list[int]] = {p: [] for p in phrases}
    patterns = {p: alerts_engine._bounded(p) for p in phrases}
    for r in rows:
        for p, pat in patterns.items():
            if pat.search(r["title"]):
                out[p].append(r["matter_id"])
    return out


NULL_POOL_SIZE = 150

# Same reasoning as alerts_engine._NULL_CACHE: the null distribution only
# depends on the corpus of titles, never on what a caller is testing, so
# it's wasted work to rebuild it (and resample+rate 300 terms) on every
# chat turn or tag creation. Keyed on a cheap COUNT — titles are only ever
# added/updated by the daily scan pipeline, never edited mid-request.
_TITLE_NULL_CACHE: dict = {}


def _title_null_distributions(con: sqlite3.Connection) -> tuple[list[float], list[float], list[str], int]:
    total = con.execute("SELECT COUNT(*) c FROM matters WHERE title IS NOT NULL").fetchone()["c"]
    if _TITLE_NULL_CACHE.get("total") == total and "uni" in _TITLE_NULL_CACHE:
        c = _TITLE_NULL_CACHE
        return c["uni"], c["bi"], c["titles"], total
    titles = [r["title"] for r in con.execute(
        "SELECT title FROM matters WHERE title IS NOT NULL").fetchall()]
    if not total:
        _TITLE_NULL_CACHE.clear()
        _TITLE_NULL_CACHE.update(total=0, uni=[], bi=[], titles=[])
        return [], [], [], 0

    def rate(term: str) -> float:
        pat = alerts_engine._bounded(term)
        return sum(1 for t in titles if pat.search(t)) / total

    all_grams: set[str] = set()
    for t in titles:
        all_grams |= alerts_engine._grams(t)
    uni_pool = [g for g in all_grams if " " not in g]
    bi_pool = [g for g in all_grams if " " in g]

    rng = random.Random(0)

    def null_dist(pool: list[str]) -> list[float]:
        sample = rng.sample(pool, min(NULL_POOL_SIZE, len(pool))) if pool else []
        return sorted(rate(t) for t in sample)

    uni_null, bi_null = null_dist(uni_pool), null_dist(bi_pool)
    _TITLE_NULL_CACHE.clear()
    _TITLE_NULL_CACHE.update(total=total, uni=uni_null, bi=bi_null, titles=titles,
                             poolUni=len(uni_pool), poolBi=len(bi_pool))
    return uni_null, bi_null, titles, total


def title_null_signal(con: sqlite3.Connection, phrases: list[str]) -> dict:
    """The same idea as alerts_engine.null_signal (is a term's match rate
    actually unusual, or just an ordinary word this corpus uses a lot) but
    run over EVERY matter's title, not just the deep-read subset — the
    right corpus for a general-purpose topic tag meant to span all of
    Metro's business, not only what was previously scanned for
    surveillance. The null pool comes from real title vocabulary, sampled
    and matched the identical way a real term would be; "distinguishing"
    is Tukey's fence (1.5x IQR above the null's own 75th percentile) on
    that MEASURED distribution, not a hand-picked cutoff. DMD, per
    Bateson: a difference that makes a difference, not just any
    difference from zero."""
    uni_null, bi_null, titles, total = _title_null_distributions(con)
    if not total or not phrases:
        return {"corpusSize": total, "terms": []}

    def rate(term: str) -> float:
        pat = alerts_engine._bounded(term)
        return sum(1 for t in titles if pat.search(t)) / total

    out = []
    for term in phrases:
        r = rate(term)
        dist = bi_null if " " in term else uni_null
        if dist:
            pct = round(100 * bisect.bisect_left(dist, r) / len(dist), 1)
            q1, q3 = dist[len(dist) // 4], dist[3 * len(dist) // 4]
            fence = q3 + 1.5 * (q3 - q1)
            distinguishing = r > fence
        else:
            pct, distinguishing = None, None
        out.append({"term": term, "matchRate": round(r, 4), "matchCount": round(r * total),
                     "nullPercentile": pct, "distinguishing": distinguishing})
    return {"corpusSize": total, "nullPoolUnigrams": _TITLE_NULL_CACHE.get("poolUni", 0),
            "nullPoolBigrams": _TITLE_NULL_CACHE.get("poolBi", 0), "terms": out}


def _tag_overlap(con: sqlite3.Connection, matched_ids: list[int], *,
                  exclude_slug: str | None = None) -> list[dict]:
    """How much a candidate tag's matches overlap with every OTHER active
    tag, including the builtin Surveillance one — flagged when more than
    half of the SMALLER tag's matches are also in the other tag (plain
    overlap coefficient, not a tuned statistic: "over half of it is the
    same matters" is self-explanatory to a reader without needing a null).
    A high-overlap pair isn't necessarily a problem — two tags CAN
    legitimately describe much of the same reality — but hiding it would
    let someone think they've built two independent filters when they
    have not."""
    mine = set(matched_ids)
    if not mine:
        return []
    others = [{"slug": "surveillance", "name": "Surveillance",
               "ids": {r["matter_id"] for r in con.execute(
                   "SELECT matter_id FROM scan_results WHERE label='surveillance'").fetchall()}}]
    _ensure_alerts_table(con)
    for r in con.execute("SELECT slug, name, matched_matter_ids_json FROM alerts WHERE active=1").fetchall():
        if r["slug"] == exclude_slug:
            continue
        others.append({"slug": r["slug"], "name": r["name"],
                        "ids": set(json.loads(r["matched_matter_ids_json"] or "[]"))})
    out = []
    for o in others:
        if not o["ids"] or o["slug"] == exclude_slug:
            continue
        shared = mine & o["ids"]
        if not shared:
            continue
        smaller = min(len(mine), len(o["ids"]))
        coefficient = len(shared) / smaller
        if coefficient > 0.5:
            out.append({"slug": o["slug"], "name": o["name"], "sharedCount": len(shared),
                        "overlapCoefficient": round(coefficient, 2), "otherTagSize": len(o["ids"])})
    out.sort(key=lambda x: -x["overlapCoefficient"])
    return out


@app.post("/api/alerts/chat")
def alerts_chat(payload: dict = Body(...)):
    """One turn of "describe what you want, get a working filter." No
    model runs SERVER-SIDE — the only one in this whole stack is the
    optional on-device model the browser itself may run (see the
    Topic Tags page), which proposes `hintTerms` here rather than deciding
    anything: every phrase, whether it came from the raw message,
    eoreader7's own reading of it, or a local model's guess, gets the
    exact same treatment — checked against real matters, kept only if it
    actually occurs. A department name is the one thing that's ever an
    exact match rather than a checked guess, because it's checked directly
    against every matter's title."""
    message = (payload.get("message") or "").strip()
    hint_terms = [t.strip() for t in (payload.get("hintTerms") or [])
                  if isinstance(t, str) and 1 < len(t.strip()) < 80][:12]
    if not message:
        return JSONResponse({"error": "say what you want to watch for"}, status_code=400)
    con = db()
    dept_hits = departments_mod.find_in(message)
    for t in hint_terms:
        for d in departments_mod.find_in(t):
            if d not in dept_hits:
                dept_hits.append(d)
    if dept_hits:
        matter_ids = _matters_for_departments(con, dept_hits)
        overlap = _tag_overlap(con, matter_ids)
        sample = _matter_brief(con, matter_ids[:8])
        con.close()
        depts = ", ".join(dept_hits)
        reply = (f"That names {len(dept_hits)} Metro department"
                 f"{'s' if len(dept_hits) != 1 else ''} we already track exactly: "
                 f"{depts}. {len(matter_ids)} matters name "
                 f"{'one of them' if len(dept_hits) != 1 else 'it'} in their title — checked "
                 f"across every matter on file, not just the ones we've deep-read for "
                 f"surveillance.")
        if overlap:
            top = overlap[0]
            reply += (f" Heads up: {round(top['overlapCoefficient'] * 100)}% of that also "
                      f"overlaps with your existing \"{top['name']}\" tag.")
        return JSONResponse({
            "reply": reply, "kind": "department", "terms": dept_hits,
            "matchedMatterIds": matter_ids, "matchCount": len(matter_ids),
            "corpusSize": None, "sample": sample, "overlap": overlap,
        })
    con.close()
    preview = alerts_engine.build_alert_preview(message, extra_phrases=hint_terms)
    kept = [t["term"] for t in preview["terms"]]
    n = len(preview["matchedMatterIds"])
    con = db()
    sample = _matter_brief(con, preview["matchedMatterIds"][:8])
    # DMD: is each surviving phrase's match rate actually unusual for this
    # corpus, or just an ordinary word the corpus happens to use a lot —
    # checked automatically on every chat turn now, not just on request.
    dmd = alerts_engine.null_signal(kept) if kept else {"terms": []}
    overlap = _tag_overlap(con, preview["matchedMatterIds"])
    con.close()
    if kept:
        shown = ", ".join(f'"{t}"' for t in kept[:6])
        more = f", plus {len(kept) - 6} more" if len(kept) > 6 else ""
        reply = (f"Reading that the way we read a title, the phrases that stand out are "
                 f"{shown}{more}. {n} of the {preview['corpusSize']} matters we've already "
                 f"read in depth mention at least one of them.")
        if n == 0:
            reply += " None of them showed up yet, though — try naming a vendor or department instead."
        noisy = [t["term"] for t in dmd["terms"] if t["distinguishing"] is False]
        if noisy:
            shown_noisy = ", ".join(f'"{t}"' for t in noisy[:4])
            reply += (f" Worth knowing: {shown_noisy} match{'es' if len(noisy) == 1 else ''} about "
                      f"as often as an ordinary word does in this corpus — not wrong, just not "
                      f"distinguishing much on its own.")
        if overlap:
            top = overlap[0]
            reply += (f" And {round(top['overlapCoefficient'] * 100)}% of these matters are "
                      f"also in your existing \"{top['name']}\" tag.")
    else:
        reply = ("I couldn't pull anything specific out of that — try naming a "
                 "technology, a vendor, or a department, or say more about it.")
    return JSONResponse({
        "reply": reply, "kind": "lexicon", "terms": kept,
        "droppedTerms": preview["droppedTerms"],
        "matchedMatterIds": preview["matchedMatterIds"], "matchCount": n,
        "corpusSize": preview["corpusSize"], "sample": sample,
        "dmdSignal": dmd["terms"], "overlap": overlap,
    })


@app.post("/api/alerts/refine")
def alerts_refine(payload: dict = Body(...)):
    """Step two of the wizard: "here are examples that definitely have
    what I mean" — a handful of matters the person picked out by hand,
    already known to be right by keyword. This contrasts those against an
    ordinary background sample to find the vocabulary they share that the
    original description never said — the shape underneath the keywords,
    which can then catch matters the literal phrase search never would
    have. Nothing generative runs here either; it's the same measured
    contrast as build_alert_preview, just against real hand-picked
    examples instead of a corpus-wide occurrence count."""
    golden_ids = [int(i) for i in (payload.get("goldenIds") or []) if str(i).isdigit()]
    known_ids = set(int(i) for i in (payload.get("knownMatterIds") or []) if str(i).isdigit())
    if not golden_ids:
        return JSONResponse({"error": "pick at least one confirmed example"}, status_code=400)
    shadow = alerts_engine.golden_shadow(golden_ids)
    shadow_terms = [t["term"] for t in shadow["terms"]]
    matched = alerts_engine.matches_for_terms(shadow_terms) if shadow_terms else []
    newly_found = sorted(set(matched) - known_ids - set(golden_ids))
    con = db()
    sample = _matter_brief(con, newly_found[:8])
    con.close()
    if shadow_terms:
        shown = ", ".join(f'"{t["term"]}"' for t in shadow["terms"][:6])
        reply = (f"Contrasting {shadow['goldenCount']} confirmed example"
                 f"{'s' if shadow['goldenCount'] != 1 else ''} against "
                 f"{shadow['backgroundCount']} ordinary matters, the vocabulary that "
                 f"actually sets them apart is {shown}. That shape alone turns up "
                 f"{len(newly_found)} matter{'s' if len(newly_found) != 1 else ''} "
                 f"the original wording never mentioned.")
    else:
        reply = ("Those examples didn't share anything that isn't just as common "
                 "everywhere else — nothing new to add beyond the original wording.")
    return JSONResponse({
        "reply": reply, "kind": "lexicon", "terms": shadow_terms,
        "shadowDetail": shadow["terms"], "matchedMatterIds": newly_found,
        "matchCount": len(newly_found), "sample": sample,
    })


def _methodology_matches(con: sqlite3.Connection, require: list[str], min_score: float) -> list[int]:
    """The same aggregation /api/feed runs live, frozen into a saved tag's
    match list at save time — a duplicated ruleset with different dials,
    not a link back to the live feed, so it can drift from the default
    rule's own results going forward."""
    require_set = {k.strip().lower() for k in require if k.strip()}
    rows = con.execute(
        "SELECT matter_id, evidence_json, score FROM scan_results WHERE label != 'error'"
    ).fetchall()
    return [r["matter_id"] for r in rows
            if _anchored(json.loads(r["evidence_json"] or "[]"), require_set)
            and (r["score"] or 0) >= min_score]


@app.post("/api/alerts")
def create_alert(payload: dict = Body(...)):
    name = (payload.get("name") or "").strip()
    kind = payload.get("kind") or "lexicon"
    terms = [t for t in (payload.get("terms") or []) if isinstance(t, str) and t.strip()]
    description = (payload.get("description") or "").strip()
    owner_account_id = payload.get("ownerAccountId")
    if owner_account_id and not ID_RE.match(owner_account_id):
        owner_account_id = None
    config = None
    if kind == "methodology":
        require = [k for k in (payload.get("require") or []) if isinstance(k, str)]
        min_score = float(payload.get("minScore") or 0)
        if not name or not require:
            return JSONResponse({"error": "need a name and at least one kind switched on"},
                                status_code=400)
        terms = require  # shown on the tag's own page as which kinds it requires
        config = {"require": require, "minScore": min_score}
    elif not name or not terms:
        return JSONResponse({"error": "need a name and at least one term"}, status_code=400)
    con = db()
    _ensure_alerts_table(con)
    dmd_signal = None
    per_term: dict[str, list[int]] | None = None
    if kind == "department":
        per_term = {t: _matters_for_departments(con, [t]) for t in terms}
        matched = sorted(set().union(*per_term.values())) if per_term else []
        corpus_size = None
    elif kind == "methodology":
        matched = _methodology_matches(con, config["require"], config["minScore"])
        corpus_size = None
    elif kind == "phrase":
        # A general topic (not surveillance-specific) — matched against
        # every matter's title, not just the few hundred deep-read for
        # surveillance. DMD (null-significance) runs automatically here:
        # every phrase-kind tag gets checked against a random-vocabulary
        # null before it's saved, same as the chat wizard does.
        per_term = _matters_for_phrases(con, terms)
        matched = sorted(set().union(*per_term.values())) if per_term else []
        corpus_size = None
        dmd_signal = title_null_signal(con, terms)
        config = {"dmdSignal": dmd_signal["terms"]}
    else:
        signal = alerts_engine.corpus_signal(terms)
        per_term = {t["term"]: t["matterIds"] for t in signal["terms"]}
        matched = signal["matchedMatterIds"]
        corpus_size = signal["corpusSize"]
    overlap = _tag_overlap(con, matched)
    if per_term is not None:
        config = config or {}
        # Per-matter attribution — which term(s) actually caused THIS
        # match — so a tag page can disclose why a given matter is on it,
        # not just that it is. Now that there are several tags, not one,
        # that's the difference between a filter you can trust and one
        # you have to take on faith.
        config["termMatches"] = {t: ids for t, ids in per_term.items() if ids}
    if kind == "phrase":
        config["overlap"] = overlap
    slug = _slugify(name)
    now = datetime.datetime.utcnow().isoformat()
    try:
        con.execute(
            """INSERT INTO alerts (slug, name, description, kind, terms_json,
                                   matched_matter_ids_json, corpus_size, created_at, active,
                                   owner_account_id, visibility, config_json)
               VALUES (?,?,?,?,?,?,?,?,1,?,'private',?)""",
            (slug, name, description, kind, json.dumps(terms),
             json.dumps(matched), corpus_size, now, owner_account_id,
             json.dumps(config) if config else None),
        )
        con.commit()
    except sqlite3.IntegrityError:
        con.close()
        return JSONResponse({"error": f'a topic tag named "{name}" already exists — try another name'},
                            status_code=409)
    con.close()
    return JSONResponse({"slug": slug, "matchCount": len(matched),
                         "dmdSignal": dmd_signal["terms"] if dmd_signal else None,
                         "overlap": overlap})


def _alert_out(con: sqlite3.Connection, r: sqlite3.Row, *, include_terms: bool = True) -> dict:
    matched = json.loads(r["matched_matter_ids_json"] or "[]")
    _ensure_votes_tables(con)
    out = {
        "slug": r["slug"], "name": r["name"], "kind": r["kind"],
        "description": r["description"], "matchCount": len(matched),
        "visibility": r["visibility"] or "private", "adjustable": False,
        "qualityVotes": _quality_vote_counts(con, r["slug"]),
    }
    if include_terms:
        out["terms"] = json.loads(r["terms_json"] or "[]")
    return out


@app.get("/api/alerts")
def list_alerts(account_id: str = Query("")):
    """YOUR topic tags — the builtin Surveillance one, plus any custom tag
    you own. Ownership is the account_id your browser already holds (the
    same one that decrypts your notes); a tag with no owner or a different

        # off scan_results, not a frozen snapshot — but from here down it
        # renders through the exact same template and fields as any other

            "dmdSignal": None, "overlap": None,
        })
    _ensure_alerts_table(con)
    r = con.execute("SELECT * FROM alerts WHERE slug=?", (slug,)).fetchone()
    if not r:
        con.close()
        return JSONResponse({"error": "not found"}, status_code=404)

        con.close()
        return JSONResponse({"error": "not found"}, status_code=404)
    matched = json.loads(r["matched_matter_ids_json"] or "[]")
    matters = _matter_brief(con, matched)

    quality_votes = _quality_vote_counts(con, slug)

    con.close()

    # put each matter here, so the tag page can disclose why, per matter.

        m["matchedOn"] = [t for t, ids in term_matches.items() if m["matterId"] in ids]
    return JSONResponse({
        "slug": r["slug"], "name": r["name"], "kind": r["kind"],
        "description": r["description"], "terms": json.loads(r["terms_json"] or "[]"),
        "corpusSize": r["corpus_size"], "matters": matters,
        "visibility": r["visibility"] or "private", "isOwner": bool(is_owner),

    })

@app.post("/api/subscribe")
def api_subscribe(body: dict = Body(...)):
    email = (body.get("email") or "").strip().lower()
    min_score = float(body.get("minScore") or 0)
    account_id = body.get("accountId")
    if account_id and not ID_RE.match(account_id):
        account_id = None
    if not EMAIL_RE.match(email):
        return JSONResponse({"error": "not a valid email"}, status_code=400)
    con = db()
    token = secrets.token_urlsafe(24)
    try:
        # account_id links this subscription to a device/handle account so a
        # real send can later be logged into THAT account's own encrypted
        # blob (see pending_digest_events) — never overwritten with NULL on
        # a re-subscribe, since a later anonymous re-subscribe of the same
        # email shouldn't sever a link that was already established.
        con.execute(
            """INSERT INTO subscribers (email, min_score, active, unsubscribe_token, account_id, created_at)
               VALUES (?, ?, 1, ?, ?, ?)
               ON CONFLICT(email) DO UPDATE SET min_score=excluded.min_score, active=1,
                 account_id=COALESCE(excluded.account_id, subscribers.account_id)""",
            (email, min_score, token, account_id, datetime.datetime.utcnow().isoformat()),
        )
        con.commit()
    finally:
        con.close()
    return JSONResponse({"ok": True})

@app.put("/api/account/{account_id}")
def api_account_put(account_id: str, body: dict = Body(...)):
    if not ID_RE.match(account_id):
        return JSONResponse({"error": "bad id"}, status_code=400)
    con = db()
    existing = _account_row(con, account_id)
    expected = body.get("expectedVersion")
    if existing and expected is not None and existing["version"] != expected:
        con.close()
        return JSONResponse({"error": "version conflict", "version": existing["version"]}, status_code=409)

    handle_hash = body.get("handleHash")
    if handle_hash:
        if not ID_RE.match(handle_hash):
            con.close()
            return JSONResponse({"error": "bad handleHash"}, status_code=400)
        taken = con.execute(
            "SELECT account_id FROM handle_index WHERE handle_hash = ?", (handle_hash,)
        ).fetchone()
        if taken and taken["account_id"] != account_id:
            con.close()
            return JSONResponse({"error": "handle already in use"}, status_code=409)
        con.execute(
            "INSERT INTO handle_index (handle_hash, account_id, created_at) VALUES (?, ?, ?) "
            "ON CONFLICT(handle_hash) DO NOTHING",
            (handle_hash, account_id, datetime.datetime.utcnow().isoformat()),
        )

    new_version = (existing["version"] if existing else 0) + 1
    con.execute(
        """INSERT INTO account_blobs (account_id, salt, wrapped_dek_device, wrapped_dek_password,
                                       blob, version, updated_at)
           VALUES (?,?,?,?,?,?,?)
           ON CONFLICT(account_id) DO UPDATE SET
             salt=COALESCE(excluded.salt, account_blobs.salt),
             wrapped_dek_device=COALESCE(excluded.wrapped_dek_device, account_blobs.wrapped_dek_device),
             wrapped_dek_password=COALESCE(excluded.wrapped_dek_password, account_blobs.wrapped_dek_password),
             blob=excluded.blob, version=excluded.version, updated_at=excluded.updated_at""",
        (
            account_id, body.get("salt"), body.get("wrappedDekDevice"), body.get("wrappedDekPassword"),
            body.get("blob"), new_version, datetime.datetime.utcnow().isoformat(),
        ),
    )
    con.commit()
    con.close()
    return JSONResponse({"ok": True, "version": new_version})


def _ensure_pending_table(con: sqlite3.Connection) -> None:
    con.executescript("""
        CREATE TABLE IF NOT EXISTS pending_digest_events (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          account_id TEXT NOT NULL,
          ts TEXT, kind TEXT, matter_ids_json TEXT, created_at TEXT
        );
        CREATE INDEX IF NOT EXISTS idx_pending_digest_account ON pending_digest_events(account_id);
    """)
    con.commit()

@app.get("/api/digest-preview")
def digest_preview(min_score: float = Query(0)):
    """A retrospective, honestly scoped: it groups matters CURRENTLY
    flagged under today's classifier by their real historical agenda/intro
    date, so it can say "here are the calendar days you'd have gotten a
    digest, going by when these matters actually reached the council" —
    it does NOT claim to reconstruct what the classifier would have said
    on any past day, since only the latest scan state is ever kept."""
    con = db()
    rows = con.execute(
        """SELECT s.matter_id, s.file, s.title, s.score,
                  COALESCE(m.agenda_date, m.intro_date) AS d
           FROM scan_results s LEFT JOIN matters m ON m.matter_id = s.matter_id
           WHERE s.label = 'surveillance' AND s.score >= ? AND
                 COALESCE(m.agenda_date, m.intro_date) IS NOT NULL
           ORDER BY d DESC""",
        (min_score,),
    ).fetchall()
    con.close()
    by_date: dict[str, list] = {}
    for r in rows:
        day = (r["d"] or "")[:10]
        if not day:
            continue
        by_date.setdefault(day, []).append({"matterId": r["matter_id"], "file": r["file"],
                                             "title": r["title"], "score": r["score"]})
    days = [{"date": d, "matters": items} for d, items in sorted(by_date.items(), reverse=True)]
    return JSONResponse({"minScore": min_score, "totalDays": len(days), "days": days[:60]})

def describe_status(status_name: str | None, intro_date: str | None, agenda_date: str | None,
                     passed_date: str | None, enactment_date: str | None) -> dict:
    """Legistar's own status_name is often uninformative (a Resolution's
    status can literally be the word "Resolution"), and a bare "agenda"
    date reads as upcoming even for something decided years ago. Prefer
    the real passed/enactment dates Legistar provides but we weren't
    capturing until now."""
    if enactment_date:
        return {"label": "Enacted", "date": enactment_date}
    if passed_date:
        return {"label": "Passed", "date": passed_date}
    if status_name in ("Withdrawn", "Indefinitely Deferred", "Failed", "Deferred", "Disapproved"):
        return {"label": status_name, "date": agenda_date or intro_date}
    if agenda_date:
        return {"label": "Scheduled", "date": agenda_date}
    return {"label": status_name or "Introduced", "date": intro_date}


def _matter_detail(matter_id: int) -> dict | None:
    con = db()
    m = con.execute("SELECT * FROM matters WHERE matter_id = ?", (matter_id,)).fetchone()
    if not m:
        con.close()
        return None
    s = con.execute("SELECT * FROM scan_results WHERE matter_id = ?", (matter_id,)).fetchone()
    try:
        p = con.execute("SELECT * FROM participation WHERE matter_id = ?",
                        (matter_id,)).fetchone()
    except sqlite3.OperationalError:
        p = None   # table not created yet on a database seeded before this
    con.close()
    status = describe_status(m["status_name"], m["intro_date"], m["agenda_date"],
                              m["passed_date"], m["enactment_date"])
    out = {
        "matterId": m["matter_id"],
        "file": m["file"],
        "title": m["title"],

function escapeHtml(s) {
  return (s || '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

  if (s === 'enacted') return 'st-enacted';

  if (s === 'scheduled') return 'st-scheduled';
  if (['withdrawn','failed','disapproved','deferred','indefinitely deferred'].includes(s)) return 'st-dead';

}

// Every tag, Surveillance included, does the same thing when clicked:
// ADDS its full matched set into the Coming up/Recent activity union below
// (several can be on at once, OR'd together — checking a second tag can
// only grow what's shown, never shrink it, the same as adding a second
// keyword to an OR search). Each tag's full list — not just the default
// page's small top-N sample — is fetched once and cached, which is what
// makes this genuinely additive instead of just filtering an already-
// truncated default view.
async function toggleTagFilter(slug, btn) {
  if (activeTagSlugs.has(slug)) {
    activeTagSlugs.delete(slug);
    btn.classList.remove('chip-on');
    renderGeneralSections();
    return;
  }
  activeTagSlugs.add(slug);
  btn.classList.add('chip-on');
  if (!tagMatchCache[slug]) {
    btn.disabled = true;
    try {
      const d = await (await fetch('/api/alerts/' + slug)).json();
      tagMatchCache[slug] = d.matters || [];
    } catch (e) { tagMatchCache[slug] = []; }
    btn.disabled = false;
  }
  renderGeneralSections();
}

// The union across every active tag, deduped by matter — each row carries
// which tag(s) actually matched it, for the chips column below.
function unionForActiveTags() {
  const byId = new Map();
  for (const slug of activeTagSlugs) {
    for (const m of (tagMatchCache[slug] || [])) {
      if (!byId.has(m.matterId)) byId.set(m.matterId, {...m, matchedTags: []});
      byId.get(m.matterId).matchedTags.push({slug, name: TAG_NAMES[slug] || slug});
    }
  }
  return [...byId.values()];
}

}

function generalMain(m) { return m.headline || m.cleanTitle || m.title || m.file; }

function sortGeneral(rows, key, dir) {
  const val = m => key === 'main' ? generalMain(m) : (key === 'statusDate' ? (m.statusDate || m.agendaDate) : m[key]);
  const sorted = [...rows].sort((a, b) => {
    const av = val(a), bv = val(b);
    if (av == null && bv == null) return 0;
    if (av == null) return 1;
    if (bv == null) return -1;
    return String(av).localeCompare(String(bv));
  });
  return dir === 'asc' ? sorted : sorted.reverse();
}

async function aesEncrypt(keyBytes, plaintext) {
  const key = await crypto.subtle.importKey('raw', keyBytes, 'AES-GCM', false, ['encrypt']);
  const iv = crypto.getRandomValues(new Uint8Array(12));

async function aesDecrypt(keyBytes, packed) {
  const key = await crypto.subtle.importKey('raw', keyBytes, 'AES-GCM', false, ['decrypt']);
  const [ivB64, ctB64] = packed.split('.');
  const pt = await crypto.subtle.decrypt({name: 'AES-GCM', iv: new Uint8Array(unb64(ivB64))}, key, unb64(ctB64));
  return td.decode(pt);
}

async function saveAccount(extra) {
  const deviceKeyBytes = new Uint8Array(unb64(localStorage.getItem(LS_DEVICE_KEY)));
  const wrappedDekDevice = await wrapKey(await deviceWrapKeyFrom(deviceKeyBytes), ACCOUNT.dek);
  const blob = await aesEncrypt(ACCOUNT.dek, JSON.stringify(ACCOUNT.notes));
  const body = Object.assign({blob, wrappedDekDevice, expectedVersion: ACCOUNT.version || undefined}, extra || {});
  const r = await fetch('/api/account/' + ACCOUNT.accountId, {
    method: 'PUT', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(body),
  });
  const d = await r.json();
  if (d.version) ACCOUNT.version = d.version;
  return d;
}

async function initAccount() {
  try {
    let deviceKeyB64 = localStorage.getItem(LS_DEVICE_KEY);
    if (!deviceKeyB64) {
      deviceKeyB64 = b64(crypto.getRandomValues(new Uint8Array(32)));
      localStorage.setItem(LS_DEVICE_KEY, deviceKeyB64);
    }
    const deviceKeyBytes = new Uint8Array(unb64(deviceKeyB64));
    ACCOUNT.accountId = await sha256hex(deviceKeyB64);

    const r = await fetch('/api/account/' + ACCOUNT.accountId);
    const d = await r.json();
    const wrapper = await deviceWrapKeyFrom(deviceKeyBytes);
    if (d.found && d.wrappedDekDevice) {
      ACCOUNT.dek = await unwrapKey(wrapper, d.wrappedDekDevice);
      ACCOUNT.version = d.version;
      try { ACCOUNT.notes = JSON.parse(await aesDecrypt(ACCOUNT.dek, d.blob)); } catch { ACCOUNT.notes = {}; }
    } else {
      ACCOUNT.dek = crypto.getRandomValues(new Uint8Array(32));
      ACCOUNT.version = 0;
      ACCOUNT.notes = {};
      await saveAccount();
    }
    ACCOUNT.ready = true;
    setAccountStatus(`Notes saved to this browser, encrypted — we can't read them.
      <a id="link-setpw">Set a password to use them on another device</a> ·
      <a id="link-login">Already have a handle?</a>`);
    wireAccountLinks();
    renderDigestHistory();
    syncDigestLog();
  } catch (e) {
    setAccountStatus('Notes are unavailable in this browser (private/incognito mode may block this).');
  }
}

function statusClass(l) {
  const s = (l||'').toLowerCase();
  if (s === 'enacted') return 'st-enacted';
  if (s === 'passed') return 'st-passed';
  if (s === 'scheduled') return 'st-scheduled';
  if (['withdrawn','failed','disapproved','deferred','indefinitely deferred'].includes(s)) return 'st-dead';
  return 'st-open';
}

function b64(buf) { return btoa(String.fromCharCode(...new Uint8Array(buf))); }
async function sha256hex(s) {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(s));
  return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,'0')).join('');
}
// Anonymous and decoupled from the zero-knowledge account/notes system —
// this id exists only to keep one browser from voting on the same matter
// more than once, never to identify anyone.
async function myVoterId() {
  let k = localStorage.getItem('legwatch_voter_id');
  if (!k) { k = b64(crypto.getRandomValues(new Uint8Array(32))); localStorage.setItem('legwatch_voter_id', k); }
  return sha256hex(k);
}

function wireTagVote(m) {
  if (!m.flagged) return;
  const box = document.getElementById('tag-vote');
  if (!box) return;
  box.style.display = 'flex';
  const upBtn = document.getElementById('tv-up'), downBtn = document.getElementById('tv-down');
  const paint = (tv, mine) => {
    document.getElementById('tv-up-n').textContent = tv.up;
    document.getElementById('tv-down-n').textContent = tv.down;
    upBtn.classList.toggle('on', mine === 1);
    downBtn.classList.toggle('on', mine === -1);
  };
  let mine = null;
  try { mine = JSON.parse(localStorage.getItem('legwatch_tag_vote:__MATTER_ID__') || 'null'); } catch (e) {}
  paint(m.tagVotes || {up: 0, down: 0}, mine);
  [upBtn, downBtn].forEach(btn => btn.addEventListener('click', async () => {
    const v = Number(btn.dataset.v);
    const voterId = await myVoterId();
    upBtn.disabled = downBtn.disabled = true;
    const retract = mine === v;
    const r = await fetch('/api/matters/__MATTER_ID__/vote' + (retract ? '?voter_id=' + voterId + '&alert_slug=surveillance' : ''), {
      method: retract ? 'DELETE' : 'POST',
      headers: {'Content-Type': 'application/json'},
      body: retract ? undefined : JSON.stringify({voterId, vote: v, alertSlug: 'surveillance'}),
    });
    upBtn.disabled = downBtn.disabled = false;
    if (!r.ok) return;
    mine = retract ? null : v;
    try {
      if (mine === null) localStorage.removeItem('legwatch_tag_vote:__MATTER_ID__');
      else localStorage.setItem('legwatch_tag_vote:__MATTER_ID__', JSON.stringify(mine));
    } catch (e) {}
    paint(await r.json(), mine);
  }));
}
