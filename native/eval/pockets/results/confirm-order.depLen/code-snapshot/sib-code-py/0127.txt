_LOOKUP = {d.lower(): d for d in DEPARTMENTS}
_LOOKUP.update(ALIASES)
# Longest first so "nashville fire department" wins over "fire department".
_ORDERED = sorted(_LOOKUP.items(), key=lambda kv: -len(kv[0]))


def resolve(fragment: str) -> str:
    """Canonical department for a fragment of title text, or "" if none."""
    if not fragment:
        return ""
    f = re.sub(r"[^a-z0-9 ]", " ", fragment.lower())
    f = re.sub(r"\s+", " ", f).strip()
    if f in _LOOKUP:
        return _LOOKUP[f]
    for key, canon in _ORDERED:
        if len(key) >= 4 and key in f:
            return canon
    return ""


def spans_in(text: str) -> list[tuple[int, int, str]]:
    """Where each department is named, as (start, end, canonical name).

    Offsets index the ORIGINAL text, so punctuation is flattened in place
    rather than collapsed — collapsing runs of whitespace would shift every
    offset after the first comma and make the spans lie. Longest key wins,
    and overlapping matches are dropped, so "Metropolitan Nashville
    Department of Emergency Communications" is one span rather than that
    plus a nested "Emergency Communications"."""
    flat = re.sub(r"[^A-Za-z0-9]", " ", text or "").lower()
    taken: list[tuple[int, int, str]] = []
    for key, canon in _ORDERED:
        if len(key) < 4:
            continue
        start = 0
        while True:
            i = flat.find(key, start)
            if i < 0:
                break
            j = i + len(key)
            start = j
            # whole words only — "fire department" must not match inside
            # "campfire department"
            if (i > 0 and flat[i - 1].isalnum()) or (j < len(flat) and flat[j].isalnum()):
                continue
            if any(i < b and a < j for a, b, _ in taken):
                continue
            taken.append((i, j, canon))
    return sorted(taken)


def surface_forms(canonical: str) -> list[str]:
    """Every string that resolves TO this canonical department — the
    canonical name itself plus any alias — so a title search can match
    however the title actually phrased it ("MNPD" as often as "Metropolitan
    Nashville Police Department"). A title never carries `matters.body_name`
    — that column is the deciding council BODY (a committee, or "Metropolitan
    Council"), a different field entirely — so this is what a department
    filter actually has to search."""
    forms = [canonical]
    for alias, canon in ALIASES.items():
        if canon == canonical:
            forms.append(alias)
    return forms


def find_in(text: str) -> list[str]:
    """Every department named in a span of text, most specific first."""
    t = re.sub(r"[^a-z0-9 ]", " ", (text or "").lower())
    t = re.sub(r"\s+", " ", t)
    out = []
    for key, canon in _ORDERED:
        if len(key) >= 4 and key in t and canon not in out:
            out.append(canon)
    return out
