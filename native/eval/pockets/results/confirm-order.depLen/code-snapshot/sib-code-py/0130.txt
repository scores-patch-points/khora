"""Sidecar-reading archival via the Khora relay (n8n -> archive.org S3
upload). Small, our-own-data payloads only — PDFs go through archive.py's
Wayback crawl instead; this relay round-trips bytes through JSON and hits
size limits well under a typical PDF (confirmed: a 20MB attachment 500'd,
a 3KB sidecar reading didn't)."""
from __future__ import annotations

import base64
import hashlib
import json
import uuid

import requests

WEBHOOK = "https://n8n.intelechia.com/webhook/khora-archive"


def archive_sidecar(subject: str, payload: dict, *, timeout: int = 30) -> dict | None:
    """Upload `payload` as a JSON sidecar under a stable per-subject
    archive.org item (khora-<sha256(subject)>) — repeated calls for the same
    subject land in the same item, one file per call. Returns
    {'itemId', 'itemUrl'} on a clean 2xx from the relay, or None on any
    failure. Never raises; never blocks the pipeline."""
    subject_hash = hashlib.sha256(subject.encode()).hexdigest()
    blob = json.dumps(payload, indent=2).encode()
    body = {
        "subject_id": subject_hash,
        "subject_hash": subject_hash,
        "file_id": str(uuid.uuid4()),
        "blob_sha256": hashlib.sha256(blob).hexdigest(),
        "encrypted_blob": base64.b64encode(blob).decode(),
    }
    try:
        r = requests.post(WEBHOOK, json=body, timeout=timeout)
    except requests.RequestException:
        return None
    if r.status_code != 200:
        return None
    item_id = f"khora-{subject_hash}"
    return {"itemId": item_id, "itemUrl": f"https://archive.org/details/{item_id}"}
