#!/usr/bin/env python3
import json, re, sys, time, urllib.parse, urllib.request, os
INV="/Users/mlacy/Documents/3.0/NPJ/tools/archive/inventory.json"
OUT="/Users/mlacy/Documents/3.0/NPJ/tools/archive/cdx.json"
d=json.load(open(INV)); srcs=d["sources"]
def real(au): return bool(re.search(r"web\.archive\.org/web/\d{14}", au or ""))
missing=[(u,r) for u,r in srcs.items() if not real(r.get("archive_url"))]
print("checking", len(missing), "urls against CDX", file=sys.stderr)
res={}
def cdx(u):
    q="https://web.archive.org/cdx/search/cdx?url="+urllib.parse.quote(u,safe="")+"&output=json&fl=timestamp,original,statuscode&filter=statuscode:200&limit=5&sort=reverse"
    try:
        req=urllib.request.Request(q, headers={"User-Agent":"npj-archivist/1.0"})
        with urllib.request.urlopen(req, timeout=30) as r:
            data=json.load(r)
        if not data or len(data)<2: return None
        # rows[0] is header
        ts=data[1][0]
        return "https://web.archive.org/web/"+ts+"/"+u
    except urllib.error.HTTPError as e:
        if e.code==429: time.sleep(5); return cdx(u)
        return None
    except Exception:
        return None
for i,(u,r) in enumerate(missing):
    au=cdx(u)
    res[u]=au
    print(("FOUND " if au else "none  ")+u[:100], file=sys.stderr)
    if i and i%10==0: print(" ... %d/%d"%(i,len(missing)), file=sys.stderr)
    time.sleep(1.0)
json.dump(res, open(OUT,"w"), indent=1)
found=sum(1 for v in res.values() if v)
print("already on archive.org:", found, "/", len(missing), file=sys.stderr)
