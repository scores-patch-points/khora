# fake_kind: modattr (non-existent attribute of a core stdlib module), bare (non-existent bare function), module (non-existent module), method (non-existent method on an instance: not statically resolvable)

def add(id, label, kind, code): S.append((id, label, kind, code.strip("\n")))
