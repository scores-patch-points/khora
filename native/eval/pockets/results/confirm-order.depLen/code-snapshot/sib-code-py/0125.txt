"""Surveillance classifier over eoreader7 output.

The classifier scores a document's eoreader7 surface keys (the noun-phrase
chunks the reader surfaces as grammatical subjects/objects) plus the raw text
against a surveillance lexicon. Using the eoreader7 referent surfaces rather
than raw keyword counts is the point: a term must be an actual referent of the
document (something the legislation is ABOUT), which filters out background
noise like "LPRF parks grant" or ceremonial mentions.

Scoring model (weights tuned on the golden/background corpus in
reports/eo_matrix.json):
  - CORE: surveillance technology nouns          weight 3.0
  - SENSOR: collection/capture vocabulary        weight 1.0
  - GUARDRAIL: restriction/oversight grammar     weight 1.2
  - VENDOR: named surveillance vendors           weight 2.0
  - CODE: MCL 13.08.080 anchor                   weight 2.5
  - density bonus: graph density vs a baseline
Verdict: surveillance if score >= threshold.
"""

@dataclass
class Verdict:
    file: str
    score: float
    label: str  # "surveillance" | "background"
    evidence: list = field(default_factory=list)


def _norm(term: str) -> str:
    return term.strip().lower().replace(" ", "_").replace("-", "_")


def surface_matches(surfaces: list[str], lexicon: list[str]) -> list[str]:
    """Return lexicon terms found as (substring of) any surface key."""
    hits = []
    for s in surfaces:
        sn = _norm(s)
        for t in lexicon:
            if t in sn:
                hits.append(t)
    return hits


def classify_surfaces(
    surfaces: list[str],
    *,
    title: str = "",
    text: str = "",
    relation_edges: int = 0,
    n_entries: int = 0,
    file: str = "",
) -> Verdict:
    evidence: list[str] = []
    score = 0.0
    core_hits: list[str] = []
    vendor_hits: list[str] = []

    def add(weight: float, term: str):
        nonlocal score
        score += weight
        evidence.append(f"{term} (w={weight})")

    for t in surface_matches(surfaces, CORE_TERMS):
        add(3.0, f"CORE:{t}")
        core_hits.append(t)
    for t in surface_matches(surfaces, SENSOR_TERMS):
        add(1.0, f"SENSOR:{t}")
    for t in surface_matches(surfaces, GUARDRAIL_TERMS):
        add(1.2, f"GUARD:{t}")
    for t in surface_matches(surfaces, VENDOR_TERMS):
        add(2.0, f"VENDOR:{t}")
        vendor_hits.append(t)

    joined = (title + " " + text).lower().replace("-", " ")
    code_hit = bool(CODE_ANCHOR_RE.search(joined))
    if code_hit:
        add(2.5, "CODE:13.08.080")

    # density bonus: surveillance legislation is denser (more relations per entry)
    if n_entries > 300:
        score += 0.5 * min(1.0, (n_entries - 300) / 200)
        evidence.append(f"density:nGraphEntries={n_entries}")

    # negative correction: strong background lexicon overrides marginal scores
    neg_hits = [t for t in NEGATIVE_LEXICON if t in joined]
    if neg_hits and score < 6:
        for t in neg_hits:
            evidence.append(f"negative:{t}")
        score *= 0.5

    # CORE-ANCHOR RULE: sensor/guardrail/density vocabulary alone is not
    # surveillance (zoning, beer, and procurement docs share it). Nor is the
    # 13.08.080 code citation by itself — confirmed live: EVERY Nashville
    # cooperative-purchasing contract carries a standard certification page
    # ("Does the contract contain any good/service relative to surveillance,
    # as described in MCL 13.08.080?"), regardless of what's actually being
    # purchased (UPS courier services, Dell computers, waste equipment all
    # false-positived on exactly this boilerplate). A document must
    # reference a concrete surveillance technology (CORE term) or a named
    # surveillance vendor to be flagged; the code citation and bare
    # "surveillance" still count as supporting evidence once something
    # concrete has already anchored it.
    anchored = bool(core_hits) or bool(vendor_hits)
    label = "surveillance" if anchored and score >= 5.0 else "background"
    if not anchored:
        evidence.append("no_core_anchor")
    return Verdict(file=file, score=round(score, 2), label=label, evidence=evidence)


def classify_title(title: str, *, file: str = "") -> Verdict:
    """Lightweight title-only pre-filter. Uses the same lexicons but matches
    against the raw title string (titles are too short for eoreader7). Not a
    substitute for the surface-based pass — it is the fast screen that decides
    which matters warrant the expensive eoreader7 deep pass."""
    t = _norm(title).replace("_", " ")
    evidence: list[str] = []
    score = 0.0
    core_hits = [x for x in CORE_TERMS if _norm(x).replace("_", " ") in t or x in _norm(title)]
    vendor_hits = [x for x in VENDOR_TERMS if _norm(x).replace("_", " ") in t or x in _norm(title)]
    for x in core_hits:
        score += 3.0
        evidence.append(f"CORE:{x}")
    for x in vendor_hits:
        score += 2.0
        evidence.append(f"VENDOR:{x}")
    for x in GUARDRAIL_TERMS:
        if _norm(x).replace("_", " ") in t:
            score += 1.0
            evidence.append(f"GUARD:{x}")
    if CODE_ANCHOR_RE.search(title):
        score += 2.5
        evidence.append("CODE:13.08.080")
    anchored = bool(core_hits) or bool(vendor_hits) or bool(CODE_ANCHOR_RE.search(title))
    # 3.0 = a single CORE technology term in the title (e.g. "surveillance",
    # "drone", "camera"). This is the recall screen: anything plausible is
    # forwarded to the deep pass, which is the authority.
    label = "surveillance" if anchored and score >= 3.0 else "background"
    if not anchored:
        evidence.append("no_core_anchor")
    return Verdict(file=file, score=round(score, 2), label=label, evidence=evidence)
